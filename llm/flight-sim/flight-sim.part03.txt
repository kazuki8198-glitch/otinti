const RXQ = { q: [] };
function rxPush(fn, data) { RXQ.q.push([fn, data]); }
function rxRun() {
  const t0 = performance.now(), B = game.touch ? 3 : 6;
  while (RXQ.q.length) { const [fn, d] = RXQ.q.shift(); fn(d); if (performance.now() - t0 > B) break; }
}

  if (typeof markReceive === 'function') markReceive(t, r.marks);
  // the aerodrome's outline (no trees on an airfield): rings of local metres
  t.aero = null;
  if (r.aero && r.aero.byteLength) { const F = new Float32Array(r.aero), rings = []; for (let i = 0; i < F.length;) { const n = F[i]; rings.push(F.subarray(i + 1, i + 1 + n * 2)); i += 1 + n * 2; } t.aero = rings; }

}
// a grid over a tile's road segments (12 m cells), to ask quickly whether a point is on a road
function cityRoadGrid(t) {
  const R_ = t.roads; t.rgrid = null; if (!R_ || !R_.pts.length) return;
  const G = 12, nx = Math.ceil((t.W || 2500) / G) + 2, nz = Math.ceil((t.Hm || 2500) / G) + 2, cells = new Map(), P = R_.pts;
  for (let l = 0; l < R_.cls.length; l++) for (let k = R_.off[l]; k < R_.off[l + 1] - 1; k++) {
    const m = R_.wid[l] / 2 + 1;
    const x0 = Math.floor((Math.min(P[k * 3], P[k * 3 + 3]) - m) / G), x1 = Math.floor((Math.max(P[k * 3], P[k * 3 + 3]) + m) / G);
    const z0 = Math.floor((Math.min(P[k * 3 + 1], P[k * 3 + 4]) - m) / G), z1 = Math.floor((Math.max(P[k * 3 + 1], P[k * 3 + 4]) + m) / G);
    for (let i = x0; i <= x1; i++) for (let j = z0; j <= z1; j++) { const key = (j + 1) * nx + i + 1; let c = cells.get(key); if (!c) cells.set(key, c = []); c.push(l, k); }
  }
  t.rgrid = { G, nx, nz, cells };
}
// is a world point on a mapped road or inside a mapped building? (the Japan city layer)
function cityBlocked(wx, wz) {
  if (!CITY.ready || !JW.online) return false;
  const [la, lo] = worldToGeo(wx, wz), n = 2 ** 14, x = Math.floor((lo + 180) / 360 * n), y = Math.floor(mercY(la) * n);
  const t = CITY.tiles.get(x + '/' + y); if (!t || t.state !== 2) return false;
  const X = (lo - t.lonW) * t.kxN, Z = -(la - t.latN) * KZ;
  if (t.rgrid) {
    const g = t.rgrid, c = g.cells.get((Math.floor(Z / g.G) + 1) * g.nx + Math.floor(X / g.G) + 1), R_ = t.roads, P = R_.pts;
    if (c) for (let i = 0; i < c.length; i += 2) {
      const l = c[i], k = c[i + 1], ax = P[k * 3], az = P[k * 3 + 1], dx = P[k * 3 + 3] - ax, dz = P[k * 3 + 4] - az, L2 = dx * dx + dz * dz || 1;
      const s = clamp(((X - ax) * dx + (Z - az) * dz) / L2, 0, 1);
      if (Math.hypot(X - ax - dx * s, Z - az - dz * s) < R_.wid[l] / 2 + (R_.cls[l] <= 5 ? 2.5 : 0.8)) return true;   // (main roads: and the pavement)
    }
  }
  if (t.aero) {
    let inside = false;
    for (const r of t.aero) for (let i = 0, j = r.length - 2; i < r.length; j = i, i += 2) if ((r[i + 1] > Z) !== (r[j + 1] > Z) && X < (r[j] - r[i]) * (Z - r[i + 1]) / (r[j + 1] - r[i + 1]) + r[i]) inside = !inside;
    if (inside) return true;
  }
  const col = t.col; if (!col) return false;
  const G = 32, cx = clamp(Math.floor(X / t.W * G), 0, G - 1), cz = clamp(Math.floor(Z / t.Hm * G), 0, G - 1);
  for (let k = col.cells[cz * G + cx]; k < col.cells[cz * G + cx + 1]; k++) {
    const b = col.items[k], B = col.boxes;
    if (X < B[b * 5] || X > B[b * 5 + 2] || Z < B[b * 5 + 1] || Z > B[b * 5 + 3]) continue;
    let inside = false;
    for (let i = col.ringOff[b], j = col.ringOff[b + 1] - 2; i < col.ringOff[b + 1]; j = i, i += 2) {
      const zi = col.ringXZ[i + 1], zj = col.ringXZ[j + 1];
      if ((zi > Z) !== (zj > Z) && X < (col.ringXZ[j] - col.ringXZ[i]) * (Z - zi) / (zj - zi) + col.ringXZ[i]) inside = !inside;
    }
    if (inside) return true;
  }
  return false;
}
// the deck of a bridge or an elevated road under a world point, if there is one near the height yRef (you're on
// it, or just above it): its height, else null. Under an elevated road you're on the street below. (any: a deck
// here at whatever height)
function cityDeckAt(wx, wz, yRef, any = false) {
  if (!CITY.ready || !JW.online) return null;
  const [la, lo] = worldToGeo(wx, wz), n = 2 ** 14, x = Math.floor((lo + 180) / 360 * n), y = Math.floor(mercY(la) * n);
  const t = CITY.tiles.get(x + '/' + y); if (!t || t.state !== 2 || !t.rgrid || !t.roads.br) return null;
  const X = (lo - t.lonW) * t.kxN, Z = -(la - t.latN) * KZ;
  const g = t.rgrid, c = g.cells.get((Math.floor(Z / g.G) + 1) * g.nx + Math.floor(X / g.G) + 1), R_ = t.roads, P = R_.pts;
  if (!c) return null;
  let best = null;
  for (let i = 0; i < c.length; i += 2) {
    const l = c[i], k = c[i + 1]; if (!R_.br[l]) continue;
    const ax = P[k * 3], az = P[k * 3 + 1], dx = P[k * 3 + 3] - ax, dz = P[k * 3 + 4] - az, L2 = dx * dx + dz * dz || 1;
    const s = clamp(((X - ax) * dx + (Z - az) * dz) / L2, 0, 1);
    if (Math.hypot(X - ax - dx * s, Z - az - dz * s) > R_.wid[l] / 2 - 0.2) continue;
    const h = P[k * 3 + 2] + (P[k * 3 + 5] - P[k * 3 + 2]) * s - 0.05;
    if (!any && (h > yRef + 0.4 || h < yRef - 5)) continue;
    if (best == null || h > best) best = h;
  }
  return best;
}
// nearest point on a road to a world position (within maxD m): position, the road's direction there, width, class
function roadNearest(wx, wz, maxD = 400, minCls = 0) {
  const [la, lo] = worldToGeo(wx, wz);
  let best = null;
  for (const t of CITY.tiles.values()) {
    const R_ = t.roads; if (!R_ || !R_.pts.length) continue;
    const X = (lo - t.lonW) * t.kxN, Z = -(la - t.latN) * KZ;
    if (X < -maxD || Z < -maxD || X > (t.W || 2500) + maxD || Z > (t.Hm || 2500) + maxD) continue;
    const P = R_.pts;
    for (let l = 0; l < R_.cls.length; l++) {
      if (R_.cls[l] >= 8 || (minCls && R_.cls[l] > minCls)) continue;           // no tracks
      for (let k = R_.off[l]; k < R_.off[l + 1] - 1; k++) {
        const ax = P[k * 3], az = P[k * 3 + 1], bx = P[k * 3 + 3], bz = P[k * 3 + 4], dx = bx - ax, dz = bz - az, L2 = dx * dx + dz * dz;
        if (L2 < 1e-4) continue;
        const s = clamp(((X - ax) * dx + (Z - az) * dz) / L2, 0, 1), px = ax + dx * s, pz = az + dz * s, d = Math.hypot(X - px, Z - pz);
        if (d < maxD && (!best || d < best.d)) best = { d, t, X: px, Z: pz, y: P[k * 3 + 2] + (P[k * 3 + 5] - P[k * 3 + 2]) * s, dx, dz, w: R_.wid[l], cls: R_.cls[l], line: l, seg: k };
      }
    }
  }
  if (!best) return null;
  const w = geoToWorld(best.t.latN - best.Z / KZ, best.t.lonW + best.X / best.t.kxN), L = Math.hypot(best.dx, best.dz);
  return { d: best.d, x: w[0], z: w[1], y: best.y, dir: [best.dx / L, best.dz / L], w: best.w, cls: best.cls, tile: best.t, line: best.line, seg: best.seg };
}
let cityRoadVis = null;
function drawRoads(planes) {
  if (!CITY.ready || !JW.online) return;
  if (!CITY.roadProg) CITY.roadProg = makeProgram(ROAD_VS, ROAD_FS);
  const pr = CITY.roadProg, u = pr.u;
  setCommon(pr);
  gl.uniform1i(u.uNoise, 2);
  gl.disable(gl.CULL_FACE);
  gl.enable(gl.POLYGON_OFFSET_FILL); gl.polygonOffset(-2, -6);
  gl.enable(gl.BLEND); gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);   // (for the contact shadows, drawn last)
  for (const t of CITY.tiles.values()) {
    if (!t.rvao) continue;
    const b = cityBox(t);
    if (!boxInFrustum(planes, b[0], t.minG - 60, b[1], b[2], (t.maxTop > -1e8 ? t.maxTop : t.minG + 500) + 5, b[3])) continue;

    if (!t.rCell || !t.rCellB) { gl.drawElements(gl.TRIANGLES, t.rni, gl.UNSIGNED_INT, 0); continue; }
    // only the cells in view (the ground's pieces are at most some 45 m across: that much margin)
    // (each cell by the reach of its own triangles: some of the ground's pieces, an apron's, are big)
    const G = 16, B = t.rCellB, sx = (b[2] - b[0]) / (t.W || 2500), sz = (b[3] - b[1]) / (t.Hm || 2500), M = 2;
    const vis = cityRoadVis || (cityRoadVis = new Uint8Array(G * G));
    for (let i = 0; i < G * G; i++) {
      const e = i * 6; if (B[e] > B[e + 2]) { vis[i] = 0; continue; }
      vis[i] = boxInFrustum(planes, b[0] + B[e] * sx - M, B[e + 4] - 1, b[1] + B[e + 1] * sz - M, b[0] + B[e + 2] * sx + M, B[e + 5] + 2, b[1] + B[e + 3] * sz + M) ? 1 : 0;
    }
    for (let o = 0; o < t.rCell.length; o += G * G + 1) {
      let a = -1, e = -1;
      for (let i = 0; i < G * G; i++) {
        const s0 = t.rCell[o + i], s1 = t.rCell[o + i + 1];
        if (s1 === s0 || !vis[i]) continue;
        if (s0 === e) { e = s1; continue; }
        if (e > a && a >= 0) gl.drawElements(gl.TRIANGLES, e - a, gl.UNSIGNED_INT, a * 4);
        a = s0; e = s1;
      }
      if (e > a && a >= 0) gl.drawElements(gl.TRIANGLES, e - a, gl.UNSIGNED_INT, a * 4);
    }
  }
  gl.disable(gl.BLEND);
  if (typeof drawMarks === 'function') drawMarks(planes);           // (止まれ and the stop lines, p_marks.js)
  gl.disable(gl.POLYGON_OFFSET_FILL);
  gl.enable(gl.CULL_FACE);
}
// hide OSM buildings that a loaded PLATEAU model already shows (rects: [south, west, north, east] in degrees)
function cityApplyHidden(t, rects) {
  if (!t.vao || !t.bRange) return;
  const hit = rects.filter(r => r[0] < t.latN && r[2] > t.latS && r[1] < t.lonE && r[3] > t.lonW);
  let K = typeof PLAT !== 'undefined' && hit.length ? PLAT.keep : null;
  if (K && (K.lat - K.rLat > t.latN || K.lat + K.rLat < t.latS || K.lon - K.rLon > t.lonE || K.lon + K.rLon < t.lonW)) K = null;
  const A = hit.length && AUTH.area && geoDist((t.latN + t.latS) / 2, (t.lonW + t.lonE) / 2, AUTH.area.lat, AUTH.area.lon) < AUTH.area.r1 + 3000;
  const cmp = typeof platCmpOver === 'function' ? platCmpOver(t) : null;   // (PLATEAU's buildings round the car instead, for comparing them: p7_plateau.js)
  const key = hit.map(r => r[4]).join(',') + (K ? '|' + K.key : '') + (A ? '|A' + Math.round(AUTH.area.lat * 250) + ',' + Math.round(AUTH.area.lon * 250) : '') + (cmp ? '|C' + cmp.map(n => n.key).join(',') : '');
  if (key === t.hideKey) return false;
  t.hideKey = key;
  t.cmpHid = cmp ? new Uint8Array(t.bRange.length / 2) : null;
  // (the comparison's leaves' extent in this tile's metres: the buildings outside it aren't looked at closely)
  let cb = null;
  if (cmp) { cb = [1e9, 1e9, -1e9, -1e9]; for (const n of cmp) { cb[0] = Math.min(cb[0], (n.w - t.lonW) * t.kxN); cb[2] = Math.max(cb[2], (n.e - t.lonW) * t.kxN); cb[1] = Math.min(cb[1], -(n.n - t.latN) * KZ); cb[3] = Math.max(cb[3], -(n.s - t.latN) * KZ); } }
  const CB = t.col && t.col.boxes && t.col.boxes.length === t.bRange.length / 2 * 5 ? t.col.boxes : null;
  cityOrderIndex(t, !hit.length && !cmp ? null : b => {
    if (cmp) {
      const x0 = CB ? CB[b * 5] : t.bCen[b * 2], z0 = CB ? CB[b * 5 + 1] : t.bCen[b * 2 + 1], x1 = CB ? CB[b * 5 + 2] : x0, z1 = CB ? CB[b * 5 + 3] : z0;
      if (x1 > cb[0] && x0 < cb[2] && z1 > cb[1] && z0 < cb[3] && platCmpHides(t, b, cmp)) { t.cmpHid[b] = 1; return false; }
    }
    const lat = t.latN - t.bCen[b * 2 + 1] / KZ, lon = t.lonW + t.bCen[b * 2] / t.kxN;
    let hidden = false, byPhoto = false, best = 1e9;
    // (PLATEAU's blocks nest, coarse over fine: whether it has a photo is the finest block's)
    for (const r of hit) if (lat > r[0] && lat < r[2] && lon > r[1] && lon < r[3]) { hidden = true; const a = (r[2] - r[0]) * (r[3] - r[1]); if (a < best) { best = a; byPhoto = !!r[5]; } }
    // near the car the map's own building comes back where PLATEAU has only a plain box (not a photo-textured one)
    if (hidden && K && !byPhoto) { const a = (lat - K.lat) / K.rLat, o = (lon - K.lon) / K.rLon; if (a * a + o * o < 1) hidden = false; }
    if (hidden && A && authHas(lat, lon)) hidden = false;          // (a built-up area is the sim's own)
    return !hidden;
  });
  return true;
}
// a tile's building triangles, ordered by a 16 x 16 grid of cells (so the sun's shadow pass can draw just
// the cells near the car), leaving out the buildings keep() says no to; uploaded, with where each cell starts
const CITY_SHG = 16;
function cityOrderIndex(t, keep) {
  const G = CITY_SHG, nb = t.bRange.length / 2;
  if (!t.bOrder) {
    const cell = new Uint16Array(nb), cnt = new Uint32Array(G * G + 1), W = t.W || 2500, Hm = t.Hm || 2500;
    for (let b = 0; b < nb; b++) {
      const cx = clamp(Math.floor(t.bCen[b * 2] / W * G), 0, G - 1), cz = clamp(Math.floor(t.bCen[b * 2 + 1] / Hm * G), 0, G - 1);
      cell[b] = cz * G + cx; cnt[cell[b] + 1]++;
    }
    for (let c = 0; c < G * G; c++) cnt[c + 1] += cnt[c];
    const order = new Uint32Array(nb), at = cnt.slice();
    for (let b = 0; b < nb; b++) order[at[cell[b]]++] = b;
    t.bOrder = order; t.bCell = cell;
    // each cell's reach: how far its buildings stand out over its edges (tile metres, 8 m more for the block
    // walls and canopies outside the footprints) and its highest roof
    const B = t.col && t.col.boxes, E = new Float32Array(G * G * 5);
    for (let c = 0; c < G * G; c++) { const cx = c % G, cz = (c - cx) / G; E.set([cx * W / G, cz * Hm / G, (cx + 1) * W / G, (cz + 1) * Hm / G, -1e9], c * 5); }
    if (B && B.length / 5 === nb) for (let b = 0; b < nb; b++) {
      const c = cell[b] * 5;
      E[c] = Math.min(E[c], B[b * 5] - 8); E[c + 1] = Math.min(E[c + 1], B[b * 5 + 1] - 8);
      E[c + 2] = Math.max(E[c + 2], B[b * 5 + 2] + 8); E[c + 3] = Math.max(E[c + 3], B[b * 5 + 3] + 8); E[c + 4] = Math.max(E[c + 4], B[b * 5 + 4] + 2);
    } else for (let c = 0; c < G * G; c++) { E[c * 5] -= 60; E[c * 5 + 1] -= 60; E[c * 5 + 2] += 60; E[c * 5 + 3] += 60; E[c * 5 + 4] = t.maxTop + 5; }
    t.cellB = E;
  }
  // the bodies of the buildings cell by cell, then their details cell by cell (cellOff, cellOffD: where each starts)
  const out = new Uint32Array(t.ibCpu.length), off = new Uint32Array(G * G + 1), offD = new Uint32Array(G * G + 1), bb = t.bBase;
  let n = 0, kept = null;
  if (keep) { kept = new Uint8Array(nb); for (let b = 0; b < nb; b++) kept[b] = keep(b) ? 1 : 0; }   // (asked once a building, not for each pass)
  for (const det of [false, true]) {
    const o = det ? offD : off; let cur = 0;
    for (const b of t.bOrder) {
      const c = t.bCell[b]; while (cur <= c) o[cur++] = n;
      if (kept && !kept[b]) continue;
      const s0 = t.bRange[b * 2], k = t.bRange[b * 2 + 1], kb = bb ? bb[b] : k;
      const a0 = det ? s0 + kb : s0, a1 = det ? s0 + k : s0 + kb;
      if (a1 > a0) { out.set(t.ibCpu.subarray(a0, a1), n); n += a1 - a0; }
    }
    while (cur <= G * G) o[cur++] = n;
  }
  gl.bindVertexArray(null);
  gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, t.ib); gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, out.subarray(0, n), gl.STATIC_DRAW);
  t.ni = n; t.cellOff = off; t.cellOffD = offD;
}
function cityDispose(t) {
  if (t.vao) { gl.deleteVertexArray(t.vao); gl.deleteBuffer(t.vb); gl.deleteBuffer(t.ib); CITY.verts -= t.nv; }
  if (t.tvao) { gl.deleteVertexArray(t.tvao); gl.deleteBuffer(t.tb); CITY.trees -= t.nTrees; }
  if (t.rvao) { gl.deleteVertexArray(t.rvao); gl.deleteBuffer(t.rvb); gl.deleteBuffer(t.rib); }
  if (typeof streetDispose === 'function') streetDispose(t);
  if (typeof fenceDispose === 'function') fenceDispose(t);
  if (typeof markDispose === 'function') markDispose(t);
  t.vao = t.tvao = t.rvao = null; t.col = null; t.roads = null;
  CITY.tiles.delete(t.key);
}
// world-space bounding box, refreshed when the floating origin moves
function cityBox(t) {
  if (t.ov === JW.ov) return t.box;
  const c = [geoToWorld(t.latN, t.lonW), geoToWorld(t.latN, t.lonE), geoToWorld(t.latS, t.lonW), geoToWorld(t.latS, t.lonE)];
  t.box = [Math.min(...c.map(p => p[0])), Math.min(...c.map(p => p[1])), Math.max(...c.map(p => p[0])), Math.max(...c.map(p => p[1]))];
  t.ov = JW.ov;
  return t.box;
}
function cityUpdate() {
  if (JW.active && JW.online === false && R.time > (JW.recheckAt || 0)) { JW.recheckAt = R.time + 25; jwRecheckOnline(); }
  if (!JW.active || !JW.online || !CITY.on) return;
  if (!CITY.ready) { cityInit(); return; }
  const eye = R.eye, [la, lo] = worldToGeo(eye[0], eye[2]);
  const agl = eye[1] - Math.max(rawHeightGeo(la, lo), 0);
  const radius = game.touch ? clamp(1800 + agl * 1.5, 1800, 3500) : clamp(2800 + agl * 2.2, 2800, 7000);
  CITY.treeR = game.touch ? 1500 : 2600;
  const n = 2 ** 14, fx = (lo + 180) / 360 * n, fy = mercY(la) * n;
  const tw = (360 / n) * kxAt(la), rr = Math.ceil(radius / tw) + 1;
  const want = [];
  const f = R.frame || 0;
  for (let dy = -rr; dy <= rr; dy++) for (let dx = -rr; dx <= rr; dx++) {
    const x = Math.floor(fx) + dx, y = Math.floor(fy) + dy;
    const cx = (x + 0.5 - fx) * tw, cy = (y + 0.5 - fy) * tw, d = Math.max(Math.hypot(cx, cy) - tw * 0.7, 0);
    if (d > radius) continue;
    const t = cityTile(x, y); t.used = f; t.dist = d;
    want.push(t);
  }
  // a built-up area (p_auth.js) is loaded whole while it's in use
  if (typeof AUTH !== 'undefined' && AUTH.on) {
    const A = AUTH.area, dLa = A.r1 / 111320, dLo = A.r1 / (111320 * Math.cos(A.lat * DEG));
    for (let x = Math.floor((A.lon - dLo + 180) / 360 * n); x <= Math.floor((A.lon + dLo + 180) / 360 * n); x++)
      for (let y = Math.floor(mercY(A.lat + dLa) * n); y <= Math.floor(mercY(A.lat - dLa) * n); y++) {
        const t = cityTile(x, y); if (t.used === f) continue;
        t.used = f; t.dist = Math.max(Math.hypot((x + 0.5 - fx) * tw, (y + 0.5 - fy) * tw) - tw * 0.7, 0);
        want.push(t);
      }
  }
  want.sort((a, b) => a.dist - b.dist);
  const vBudget = game.touch ? 1.5e6 : 5e6;
  let busy = 0; for (const w of CITY.workers) busy += w.busy;
  for (const t of want) {
    if (busy >= CITY.workers.length * 2) break;
    if (CITY.verts > vBudget && t.dist > 1500) break;
    if (t.state === 3 && R.time > t.retry && CITY.failed < 40) t.state = 0;
    if (t.state === 2 && t.demOk === false && getTile(14, t.x, t.y).demState === 2) { cityDispose(t); continue; }   // (its fine elevation came after all)
    if (t.state !== 0) continue;
    const dt = getTile(14, t.x, t.y);
    if (dt.demState === 0) { loadDem(dt); dt.demT = R.time; continue; }
    if (dt.demState === 1 && R.time - (dt.demT ?? R.time) < 8) { if (dt.demT == null) dt.demT = R.time; continue; }   // (a slow elevation tile: after 8 s the city goes ahead on the coarse one, rebuilt when it comes)
    cityDispatch(t); busy++;
  }
  // drop tiles that fell well outside the radius
  if (CITY.tiles.size > want.length + 6 || CITY.verts > vBudget) {
    for (const t of [...CITY.tiles.values()]) if (t.used < f - 30 && t.state !== 1) cityDispose(t);
  }
}
function citySetTile(u, t) {
  gl.uniform4f(u.uTileA, t.lonW - JW.lon0, t.latN - JW.lat0, t.latN, t.kxN);
}
let cityCellVis = null;
function drawCity(planes) {
  if (!CITY.ready || !JW.online) return;
  const pr = CITY.prog, u = pr.u;
  setCommon(pr);
  gl.uniform1i(u.uNoise, 2);
  gl.uniform1f(u.uNight, ENV.night ? 1 : clamp((0.3 - ENV.sunDir[1]) * 4, 0, 1));
  gl.enable(gl.CULL_FACE);
  // cell by cell (16 x 16 to a tile): only the cells in view, and a building's details (windows, doors,
  // fittings, balconies, block walls: most of a house's triangles) only in the cells near the camera
  const DR = game.touch ? 280 : 480, eye = R.eye;
  gl.uniform1f(u.uDetR, DR); gl.uniform1f(u.uRelief, !game.touch || (SUNSH.auto && SUNSH.auto.on) ? 1 : 0);   // (the facades' relief: on a phone with the sun's shadows, while there is the frame time)
  for (const t of CITY.tiles.values()) {
    if (!t.vao) continue;
    const b = cityBox(t), y0 = t.minG - 60, y1 = t.maxTop + 5;
    if (!boxInFrustum(planes, b[0], y0, b[1], b[2], y1, b[3])) continue;
    citySetTile(u, t);
    gl.bindVertexArray(t.vao);
    if (!t.cellOff) { gl.drawElements(gl.TRIANGLES, t.ni, gl.UNSIGNED_INT, 0); continue; }
    const G = CITY_SHG, E = t.cellB, sx = (b[2] - b[0]) / (t.W || 2500), sz = (b[3] - b[1]) / (t.Hm || 2500);   // (tile metres to the world box)
    const vis = cityCellVis || (cityCellVis = new Uint8Array(G * G));
    for (let i = 0; i < G * G; i++) {
      if (t.cellOff[i + 1] === t.cellOff[i] && t.cellOffD[i + 1] === t.cellOffD[i]) { vis[i] = 0; continue; }
      const x0 = b[0] + E[i * 5] * sx, z0 = b[1] + E[i * 5 + 1] * sz, x1 = b[0] + E[i * 5 + 2] * sx, z1 = b[1] + E[i * 5 + 3] * sz;
      if (!boxInFrustum(planes, x0, y0, z0, x1, Math.min(y1, E[i * 5 + 4]), z1)) { vis[i] = 0; continue; }
      const dx = Math.max(x0 - eye[0], eye[0] - x1, 0), dz = Math.max(z0 - eye[2], eye[2] - z1, 0);
      vis[i] = dx * dx + dz * dz < DR * DR ? 2 : 1;
    }
    // runs of consecutive visible cells, each one draw
    for (const [off, need] of [[t.cellOff, 1], [t.cellOffD, 2]]) {
      let a = -1, e = -1;
      for (let i = 0; i < G * G; i++) {
        if (vis[i] < need || off[i + 1] === off[i]) continue;
        if (off[i] === e) { e = off[i + 1]; continue; }
        if (e > a && a >= 0) gl.drawElements(gl.TRIANGLES, e - a, gl.UNSIGNED_INT, a * 4);
        a = off[i]; e = off[i + 1];
      }
      if (e > a && a >= 0) gl.drawElements(gl.TRIANGLES, e - a, gl.UNSIGNED_INT, a * 4);
    }
  }
  // trees
  const tp = CITY.treeProg, tu = tp.u;
  setCommon(tp);
  gl.activeTexture(gl.TEXTURE7); gl.bindTexture(gl.TEXTURE_2D, CITY.treeTex); gl.activeTexture(gl.TEXTURE0);
  gl.uniform1i(tu.uTreeTex, 7);
  gl.uniform1f(tu.uFade, CITY.treeR);
  gl.disable(gl.CULL_FACE);
  for (const t of CITY.tiles.values()) {
    if (!t.tvao || t.dist > CITY.treeR) continue;
    const b = cityBox(t);
    if (!boxInFrustum(planes, b[0], -50, b[1], b[2], 4000, b[3])) continue;
    citySetTile(tu, t);
    gl.bindVertexArray(t.tvao);
    gl.drawArraysInstanced(gl.TRIANGLES, 0, CITY.treeGeo.count, t.nTrees);
  }
  gl.enable(gl.CULL_FACE);
}
// collision: aircraft centre and wing tips against building footprints
function cityCollide() {
  if (!CITY.ready || !JW.online || sim.crashed || sim.onGround) return null;
  const r = qrot(sim.q, [1, 0, 0]), half = (AC.b || 10) * 0.45;
  for (const s of [0, -1, 1]) {
    const p = [sim.pos[0] + r[0] * half * s, sim.pos[1] + r[1] * half * s, sim.pos[2] + r[2] * half * s];
    const [la, lo] = worldToGeo(p[0], p[2]);
    const n = 2 ** 14, x = Math.floor((lo + 180) / 360 * n), y = Math.floor(mercY(la) * n);
    const t = CITY.tiles.get(x + '/' + y);
    if (!t || !t.col || p[1] > t.maxTop + 2) continue;
    const X = (lo - t.lonW) * t.kxN, Z = -(la - t.latN) * KZ;
    const G = 32, cx = clamp(Math.floor(X / t.W * G), 0, G - 1), cz = clamp(Math.floor(Z / t.Hm * G), 0, G - 1), c = t.col;
    for (let k = c.cells[cz * G + cx]; k < c.cells[cz * G + cx + 1]; k++) {
      const b = c.items[k], B = c.boxes;
      if (p[1] > B[b * 5 + 4] || X < B[b * 5] || X > B[b * 5 + 2] || Z < B[b * 5 + 1] || Z > B[b * 5 + 3]) continue;
      let inside = false;
      for (let i = c.ringOff[b], j = c.ringOff[b + 1] - 2; i < c.ringOff[b + 1]; j = i, i += 2) {
        const zi = c.ringXZ[i + 1], zj = c.ringXZ[j + 1];
        if ((zi > Z) !== (zj > Z) && X < (c.ringXZ[j] - c.ringXZ[i]) * (Z - zi) / (zj - zi) + c.ringXZ[i]) inside = !inside;
      }
      if (inside) return s === 0 ? '建物に衝突しました' : '翼が建物に接触しました';
    }
  }
  return null;
}

const CITY_POS_GLSL = `
uniform vec4 uTileA;   // lonW - lon0, latN - lat0, latN, metres per degree of longitude at latN
vec3 cityWorld(vec2 l, float y){

  float lat = uTileA.z + dlat;
  return vec3((uTileA.x + l.x / uTileA.w) * 111320.0 * cos(radians(lat)), y, -(uTileA.y + dlat) * 110574.0);
}`;
const BLDG_VS = GLSL_HEAD + `
layout(location=0) in vec2 aXZ;
layout(location=1) in float aY;
layout(location=2) in vec3 aN;
layout(location=3) in vec4 aM;       // u (dm), seed + 256*roof + 512*feature wall + 1024*kind, height (dm), height above ground (dm)
uniform mat4 uViewProj;
uniform vec3 uCamPos;
uniform float uCurv;
` + CITY_POS_GLSL + `
out vec3 vWorld; out vec3 vN; out vec3 vM; flat out float vSeed; flat out float vRoof; flat out float vFeat; flat out float vKind;
void main(){

  vec2 d = w.xz - uCamPos.xz;
  vWorld = w;
  w.y -= dot(d, d) * uCurv;
  vN = aN;

}`;
const BLDG_FS = GLSL_HEAD + GLSL_COMMON + GLSL_SHADOW + `
in vec3 vWorld; in vec3 vN; in vec3 vM; flat in float vSeed; flat in float vRoof; flat in float vFeat; flat in float vKind;
uniform sampler2D uNoise;
uniform float uNight;
uniform float uDetR, uRelief;
out vec4 outColor;
float h1(float n){ return fract(sin(n * 12.9898) * 43758.5453); }
float h2(vec2 p){ return fract(sin(dot(p, vec2(12.9898, 78.233))) * 43758.5453); }
void main(){
  if (zoneCut(vWorld) || zoneAbove(vWorld) || lmCut(vWorld)) discard;

  vec3 Vt = vec3(dot(V, T), V.y, max(dot(V, n), 0.06)), Lt = vec3(dot(uSunDir, T), uSunDir.y, dot(uSunDir, n));

    float joint = max(1.0 - smoothstep(fw.x, fw.x + 0.04, min(f.x, 1.0 - f.x)), 1.0 - smoothstep(fw.y, fw.y + 0.07, min(f.y, 1.0 - f.y)));
    vec3 c = mix(vec3(0.6, 0.6, 0.58), vec3(0.7, 0.69, 0.66), s1) * (0.94 + 0.06 * h2(floor(g) + vSeed));
    c *= 1.0 - 0.18 * joint;
    if (n.y > 0.5) c = vec3(0.55, 0.55, 0.53);
    c *= mix(0.75, 1.0, smoothstep(0.0, 0.6, y)) * (0.9 + 0.2 * texture(uNoise, vWorld.xz / 3.0 + vWorld.y / 2.0).b);
    col = c * (uSunCol * diff * 0.9 + amb);
  } else
  if (vKind > 9.5 && vRoof < 0.5) {
    // the airport's own buildings: a terminal's (or the tower cab's) glass curtain wall, a hangar's ribbed metal
    // cladding and its great door
    float u = vM.x, y = vM.z;
    vec3 lit0 = uSunCol * diff * 0.9 + amb;
    if (vKind < 10.5) {
      vec2 g = vec2(u / 1.5, y / 4.5), f = fract(g), fw = max(fwidth(g), vec2(1e-4));
      float mul = 1.0 - smoothstep(fw.x * 1.2, fw.x * 1.2 + 0.03, min(f.x, 1.0 - f.x));
      float slab = 1.0 - smoothstep(0.1, 0.1 + fw.y * 1.5, f.y);
      float top = step(H - 1.6, y), bot = step(y, 0.35);

      col = mix(glass, vec3(0.74, 0.76, 0.78) * lit0, solid);

    } else {

      if (y < 1.2) c = vec3(0.36, 0.37, 0.38);
      if (vFeat > 0.5 && y < H * 0.78 && y > 0.05) {

        float leaf = fract(u / 7.0), seam = 1.0 - smoothstep(0.0, 0.01 + fwidth(u / 7.0), min(leaf, 1.0 - leaf));
        c = vec3(0.7, 0.72, 0.74) * (0.9 + 0.1 * step(0.5, fract(y / 0.6))) * (1.0 - 0.6 * seam);
      } else if (vFeat > 0.5 && y < H * 0.9) c = s1 < 0.5 ? vec3(0.75, 0.08, 0.1) : vec3(0.05, 0.2, 0.55);
      col = c * lit0;
      bEm = vec3(1.0, 0.95, 0.85) * 0.25 * uNight * step(y, H * 0.78) * vFeat;
    }
    col *= 0.93 + 0.1 * texture(uNoise, vWorld.xz / 23.0 + vWorld.y / 31.0).b;
  } else
  if (vKind > 3.5 && vKind < 5.5 || vKind > 7.5) {
    // a house's openings and fittings (real geometry): windows and sliding doors in aluminium frames, the
    // front door, canopies, the air-conditioner unit, downpipes
    float u = vM.x, y = vM.z, w = max(H, 0.3);
    vec3 alu = (s2 > 0.5 ? vec3(0.28, 0.25, 0.22) : vec3(0.68, 0.69, 0.68));
    vec3 lit0 = uSunCol * diff * 0.9 + amb;
    if (vKind < 4.5) {                                                   // fittings
      vec3 c = mix(vec3(0.72, 0.72, 0.7), vec3(0.85, 0.84, 0.8), s1);
      if (vFeat > 0.5) {                                                 // the AC unit's front: a round fan grille
        float r = length(vec2(u - w * 0.38, y - 0.3)) / 0.21;

        c *= 1.0 - 0.15 * step(0.5, fract((u - w * 0.72) / 0.03)) * step(w * 0.7, u) * step(u, w * 0.95) * step(0.1, y) * step(y, 0.5);

      if (y > 1.6 && edge > 0.05) c = mix(c, vec3(0.1, 0.12, 0.15), 0.7);    // a pane above

    } else {                                                             // windows and sliding doors
      float h = vKind > 7.5 ? 1.85 : 1.1;
      float edge = min(min(u, w - u), min(y, h - y)), mid = abs(u - w * 0.5);
      // the glass set 5 cm back in its aluminium frame: the frame's inner sides seen at an angle (lit as they
      // face), the glass and the middle bar shifted, the glass in the frame's shadow when the sun is behind it
      float rvW = 0.0, shW = 1.0; vec3 nW = n;
      if (edge > 0.045 && dist < 140.0) {
        vec2 pm = vec2(u, y), lo = vec2(0.045), hi = vec2(w - 0.045, h - 0.045), dq = -Vt.xy / Vt.z;
        float tx = dq.x > 1e-4 ? (hi.x - pm.x) / dq.x : dq.x < -1e-4 ? (lo.x - pm.x) / dq.x : 1e9;
        float ty = dq.y > 1e-4 ? (hi.y - pm.y) / dq.y : dq.y < -1e-4 ? (lo.y - pm.y) / dq.y : 1e9;
        if (min(tx, ty) < 0.05) { rvW = 1.0; nW = tx < ty ? (dq.x > 0.0 ? -T : T) : vec3(0.0, dq.y > 0.0 ? -1.0 : 1.0, 0.0); }
        else {
          vec2 q = pm + dq * 0.05, so = q + Lt.xy / max(Lt.z, 0.05) * 0.05;
          mid = abs(q.x - w * 0.5);
          shW = Lt.z > 0.0 && so.x > lo.x && so.x < hi.x && so.y > lo.y && so.y < hi.y ? 1.0 : 0.0;
        }
      }
      float frame = step(edge, 0.045) + step(mid, 0.022) + rvW;
      vec3 r = reflect(-V, n);
      float fres = 0.15 + 0.7 * pow(1.0 - max(dot(n, V), 0.0), 4.0);
      vec3 glz = mix(vec3(0.05, 0.06, 0.07), envRefl(r, 0.02) * 0.85, fres) + uSunCol * pow(max(dot(r, uSunDir), 0.0), 150.0) * 2.0 * shW;
      glz *= mix(0.75, 1.0, shW * diff + (1.0 - diff));
      if (rvW > 0.5) lit0 = uSunCol * max(dot(nW, uSunDir), 0.0) * 0.9 * step(0.001, diff) + skyIrr(nW) * 0.7;
      float cur = step(0.55, h2(vec2(floor(vWorld.x * 0.3), vSeed)));
      glz = mix(glz, vec3(0.7, 0.66, 0.58) * lit0 * 0.5, cur * 0.45 * step(mid, w * 0.5 - 0.1));      // curtains
      float on = step(0.72, h2(vec2(floor(vWorld.x * 0.4 + vWorld.z * 0.4), vSeed + 3.0)));
      col = mix(glz * mix(1.0, 0.35, uNight), alu * lit0, min(frame, 1.0));
      bEm = vec3(1.0, 0.78, 0.5) * on * uNight * (1.0 - min(frame, 1.0)) * 1.2;
    }
  } else if (vKind > 5.5) {
    // balconies: a concrete slab, and a parapet of frosted glass, a painted panel or open metal bars
    vec3 cc = mix(vec3(0.74, 0.73, 0.70), vec3(0.86, 0.85, 0.81), s1);
    vec3 c = cc; float y = vM.z;
    if (vKind > 6.5 && y > 0.04) {
      float aaB = clamp(1.0 - fwidth(vM.x / 0.11) * 1.5, 0.0, 1.0);
      if (y > 0.98) c = vec3(0.55, 0.56, 0.56);                                  // the hand rail
      else if (s3 < 0.35) {                                                      // open bars: see the rooms through them
        float bar = step(0.72, fract(vM.x / 0.11));
        if (y < 0.9 && y > 0.1 && bar < 0.5 && aaB > 0.3) discard;
        c = vec3(0.5, 0.51, 0.5);
      } else if (s3 < 0.7) {                                                     // frosted glass
        vec3 r = reflect(-V, n);
        c = mix(vec3(0.46, 0.52, 0.55), skyColor(r) * 0.7, 0.25 + 0.5 * pow(1.0 - max(dot(n, V), 0.0), 3.0));
        c = mix(c, vec3(0.5), step(0.96, fract(vM.x / 1.8)) * aaB);               // posts
      } else c = cc * (0.96 + 0.04 * step(0.5, fract(vM.x / 0.9)));             // painted panel
    }
    if (vKind < 6.5 && n.y < -0.5) c *= 0.8;                                      // the slab's underside
    col = c * (uSunCol * diff * 0.9 + amb);
  } else if (vRoof > 0.5) {
    vec3 rc;
    if (house) {
      int k = int(s1 * 7.0);
      rc = k == 0 ? vec3(0.22, 0.24, 0.28) : k == 1 ? vec3(0.34, 0.25, 0.20) : k == 2 ? vec3(0.42, 0.43, 0.44) :
           k == 3 ? vec3(0.19, 0.28, 0.32) : k == 4 ? vec3(0.44, 0.23, 0.17) : k == 5 ? vec3(0.60, 0.61, 0.62) : vec3(0.30, 0.31, 0.30);
    } else {
      rc = mix(vec3(0.50, 0.50, 0.49), vec3(0.66, 0.65, 0.62), s1);
      if (s2 > 0.85) rc = vec3(0.30, 0.42, 0.26);                 // rooftop garden
    }
    float nz = texture(uNoise, vWorld.xz / 7.0).b * 0.5 + texture(uNoise, vWorld.xz / 1.3).b * 0.5;
    rc *= 0.86 + nz * 0.28;
    // pitched roofs: courses of tiles (a shadow line under each course, the joints staggered)
    if (n.y < 0.97 && n.y > 0.2) {
      float ty = vWorld.y / 0.095, aaR = clamp(1.0 - fwidth(ty) * 1.5, 0.0, 1.0);

           k == 3 ? vec3(0.64, 0.55, 0.45) : k == 4 ? vec3(0.9, 0.89, 0.86) : vec3(0.68, 0.7, 0.72);
    } else if (!tower) {

           k == 3 ? vec3(0.84, 0.82, 0.76) : k == 4 ? vec3(0.52, 0.47, 0.42) : vec3(0.72, 0.70, 0.66);

    float z0 = shop ? 4.3 : 0.0;                                                 // storeys over a shop start above its sign

    // the wall's own finish: lap siding on houses, small tiles on some mansions, rain streaks under windows
    float nz = texture(uNoise, vWorld.xz / 5.0 + vWorld.y / 9.0).b;

    float aaF = clamp(1.0 - fwidth(vM.z / 0.24) * 2.0, 0.0, 1.0);

      fin = (1.0 - 0.1 * smoothstep(0.38, 0.5, max(tf.x * 0.25, tf.y)) * aaT) * (0.95 + 0.1 * nHash(floor(tp)) * aaT);
    } else fin = 0.93 + 0.14 * nz;
    if (tower) {
      float band = smoothstep(0.10, 0.16, fp.y) * (1.0 - smoothstep(0.86, 0.92, fp.y));
      float mull = smoothstep(0.03, 0.08, fp.x) * (1.0 - smoothstep(0.92, 0.97, fp.x));
      win = band * mull; cov = 0.66;
      if (s3 > 0.5) { win = band; cov = 0.76; }                        // ribbon glazing
    } else if (veranda) {
      // behind the balconies: sliding doors, two panes in an aluminium frame
      float x0 = 0.1, x1 = 0.9, y0 = 0.03, y1 = 0.8;
      float wx = smoothstep(x0 - 0.01, x0 + 0.01, fp.x) * (1.0 - smoothstep(x1 - 0.01, x1 + 0.01, fp.x));
      float wy = smoothstep(y0 - 0.01, y0 + 0.01, fp.y) * (1.0 - smoothstep(y1 - 0.01, y1 + 0.01, fp.y));

      frame = win * max(step(min(fx, fy), 0.05), step(fm, 0.035));
    } else {
      float x0 = 0.2, x1 = 0.8, y0 = 0.32, y1 = house ? 0.8 : 0.86;

      float wy = smoothstep(y0 - 0.02, y0 + 0.02, fp.y) * (1.0 - smoothstep(y1 - 0.02, y1 + 0.02, fp.y));
      win = wx * wy;
      float skipW = house ? 0.45 : 0.12;
      if (h2(cell + vSeed) < skipW) win = 0.0;
      cov = 0.36 * (1.0 - skipW);
      float fx = min(fp.x - x0, x1 - fp.x) * ws, fy = min(fp.y - y0, y1 - fp.y) * fh;
      frame = win * max(step(min(fx, fy), 0.055), house ? step(abs(fp.x - 0.5) * ws, 0.03) : 0.0);
      // the window set back in the wall: the view ray, going in, meets a side of the opening (a reveal: the
      // jambs, the sill, the head) before the glass, and that side is what is seen, lit by its own facing;
      // otherwise the glass and its frame, shifted, and in the head's or a jamb's shadow when the sun is behind it
      float RD = house ? 0.07 : mansion ? 0.13 : 0.2;
      if (win > 0.5 && aa > 0.5 && dist < 260.0) {
        vec2 pm = fp * vec2(ws, fh), lo = vec2(x0 * ws, y0 * fh), hi = vec2(x1 * ws, y1 * fh), dq = -Vt.xy / Vt.z;
        float tx = dq.x > 1e-4 ? (hi.x - pm.x) / dq.x : dq.x < -1e-4 ? (lo.x - pm.x) / dq.x : 1e9;
        float ty = dq.y > 1e-4 ? (hi.y - pm.y) / dq.y : dq.y < -1e-4 ? (lo.y - pm.y) / dq.y : 1e9;
        if (min(tx, ty) < RD) { rv = 1.0; rvN = tx < ty ? (dq.x > 0.0 ? -T : T) : vec3(0.0, dq.y > 0.0 ? -1.0 : 1.0, 0.0); }
        else {
          vec2 q = pm + dq * RD, so = q + Lt.xy / max(Lt.z, 0.05) * RD;
          float fx2 = min(q.x - lo.x, hi.x - q.x), fy2 = min(q.y - lo.y, hi.y - q.y);
          frame = max(step(min(fx2, fy2), 0.05), house ? step(abs(q.x - (lo.x + hi.x) * 0.5), 0.03) : 0.0);
          glassSh = Lt.z > 0.0 && so.x > lo.x && so.x < hi.x && so.y > lo.y && so.y < hi.y ? 1.0 : 0.0;
        }
      }
      if (!house && s3 > 0.6) { win = wy * smoothstep(0.02, 0.06, fp.x) * (1.0 - smoothstep(0.94, 0.98, fp.x)); cov = 0.5; frame = win * step(fract(g.x * 2.0) * ws * 0.5, 0.05); }   // strip windows
      // rain streaks run down from the sills
      if (!house) fin *= 1.0 - 0.13 * wx * (1.0 - smoothstep(0.0, 0.3, fp.y)) * texture(uNoise, vec2(vM.x * 1.7, vM.z * 0.08)).a * aa;
    }
    if (vM.z < 0.3 || vM.z > H - 0.9 || (shop && vM.z < z0) || (house && dist < uDetR)) { win = 0.0; rv = 0.0; frame = 0.0; }   // (houses near by have real windows; far off, where those aren't drawn, painted ones)  // plinth, parapet, the shop storey
    frame *= step(0.001, win) * aa;
    win = mix(cov, win, aa);
    // glass: sky reflection with Fresnel, dark interior
    vec3 r = reflect(-V, n);
    float fres = 0.12 + 0.75 * pow(1.0 - max(dot(n, V), 0.0), 4.0);
    vec3 glass = mix(vec3(0.07, 0.08, 0.1), envRefl(r, 0.02) * 0.9, fres + (tower ? 0.25 : 0.0));
    glass += uSunCol * pow(max(dot(r, uSunDir), 0.0), 120.0) * 2.0 * glassSh;
    glass *= mix(0.72, 1.0, glassSh * diff + (1.0 - diff));               // (the head's shadow across the glass)
    if (house || veranda) glass = mix(glass, vec3(0.62, 0.6, 0.55) * (uSunCol * 0.3 + amb), 0.35 * step(0.5, h2(cell * 5.1 + vSeed)));   // curtains drawn
    // only some rooms are lit (rooms span 2-3 windows), some buildings are nearly dark, brightness varies
    float bmood = h2(vec2(vSeed * 3.1, 1.7));
    float share = (tower ? 0.28 : house ? 0.2 : 0.22) * (bmood < 0.2 ? 0.15 : 0.6 + 0.8 * bmood);
    vec2 room = vec2(floor(cell.x / (house ? 2.0 : 3.0)), cell.y);
    float on = step(1.0 - share, h2(room * 1.7 + vSeed)) * step(0.15, h2(cell * 3.3 + vSeed));
    on *= 0.45 + 0.75 * h2(room * 2.9 + vSeed + 4.0);
    // far away the windows blur into their average: keep that dim and warm, so walls stay dark with a glow
    float lit = mix(share * 0.07, on, aa * aa) * uNight;
    vec3 wcol = mix(vec3(1.0, 0.72, 0.42), mix(vec3(1.0, 0.78, 0.5), vec3(0.84, 0.9, 1.0), step(tower ? 0.45 : 0.75, h2(cell * 2.3 + vSeed))), aa);
    glass = mix(glass, glass * 0.3, uNight);
    bEm = wcol * lit * win * (1.0 - frame) * 1.5;
    // the photographed wall finish (tiles, concrete, plaster, siding) in the building's own colour
    if (uMatOn > 0.5 && !tower) {
      // (a Japanese house is clad in clean factory-made siding or plaster: the photographed plaster's grain, and
      // the siding's own lines drawn over it; not the mossy boards of a barn)
      float L = house ? 11.0 : mansion ? (s3 > 0.35 ? 9.0 : 10.0) : (s3 > 0.5 ? 10.0 : 11.0);
      float S = L == 9.0 ? 1.2 : 3.0;

      wt = house ? mix(vec3(1.0), wt, 0.6) : wt;
      fin = mix(1.0, fin, house ? 0.85 : 0.3);
      wc *= clamp(wt, 0.4, 1.8);

      // laps) and its own gloss (glazed tiles shine, plaster doesn't)

      // (not merely what is dark in the photo: a dark stone panel no longer mirrors the sky), mullions and panel

      // no two buildings alike: a tint, rain streaks down the wall, grime near the street
      day *= mix(vec3(0.86), vec3(1.12), vec3(h1(vSeed + 1.1), h1(vSeed + 1.1) * 0.6 + h1(vSeed + 2.7) * 0.4, h1(vSeed + 3.9) * 0.5 + h1(vSeed + 1.1) * 0.5));

      day *= 1.0 - 0.16 * smoothstep(0.45, 0.8, stk) * (1.0 - gm) - 0.12 * (1.0 - smoothstep(0.0, 6.0, vM.z));

        // (granite slabs in 1.2 x 0.6 m courses, a light grey or a warm beige; the lobby behind its glass lit by its
        // ceiling lights, a warm glow even by day, with the street mirrored over it)
        vec2 sg = vec2(vM.x / 1.2 + step(0.5, fract(vM.z / 1.2)) * 0.5, vM.z / 0.6), sf = abs(fract(sg) - 0.5), sw = max(fwidth(sg), vec2(1e-4));

        vec3 stone = mix(vec3(0.58, 0.57, 0.55), vec3(0.62, 0.57, 0.5), step(0.5, h1(vSeed + 3.3))) * (0.9 + 0.2 * texture(uNoise, vec2(vM.x, vM.z) * 0.3).b) * (1.0 - 0.25 * sj) * (uSunCol * (0.35 + 0.5 * diff) + amb * 0.85);

        gm = lob;
      }
      col = mix(day, day * 0.18, uNight);
      vec3 nt = matC(20.0 + step(0.5, s3), vec2(vM.x + vSeed * 11.0, vM.z) / 30.0);
      bEm = nt * nt * 2.4 * uNight * (0.4 + 0.6 * gm) * (bmood < 0.2 ? 0.3 : 1.0);
    }
    // ground-floor shade and contact darkening (the street's own light keeps it from going black)
    col *= mix(0.74, 1.0, smoothstep(0.0, 4.5, vM.z));
    if (shop && vM.z < z0) {
      // a shop front: glazing or a roller shutter, pillars between the shops, and a sign band that lights up
      float unit = floor(vM.x / 6.5), ux = fract(vM.x / 6.5), sh = h2(vec2(unit, vSeed));
      float aaS = clamp(1.0 - fwidth(vM.x / 6.5) * 8.0, 0.0, 1.0);
      vec3 hue = 0.5 + 0.5 * cos(6.2832 * (sh * 3.7 + vec3(0.0, 0.33, 0.67)));
      vec3 lit0 = uSunCol * diff * 0.9 + amb;
      if (vM.z > 3.35) {
        vec3 sign = mix(vec3(0.92), hue, 0.75);
        float letters = step(0.5, h2(floor(vec2(vM.x / 0.55, (vM.z - 3.35) / 0.3)) + sh)) * step(0.15, abs(ux - 0.5)) * step(abs(ux - 0.5), 0.42) * step(3.5, vM.z) * step(vM.z, 4.1);
        sign = mix(sign, mix(vec3(0.1), vec3(0.95), step(0.5, fract(sh * 7.0))), letters * 0.8 * aaS);
        col = sign * lit0 * 0.9;
        bEm += sign * (0.25 + 1.3 * uNight) * aaS;
        if (vM.z > 4.2) col = wc * lit0 * 0.85;
      } else {
        float pil = 1.0 - step(0.035, ux) * step(ux, 0.965);
        if (sh < 0.22 || (uNight > 0.5 && sh < 0.55)) {                         // closed: a ribbed steel shutter
          float rib = 0.85 + 0.15 * smoothstep(0.3, 0.5, abs(fract(vM.z / 0.09) - 0.5)) * aa;
          col = vec3(0.6, 0.61, 0.6) * rib * lit0 * (vM.z > 3.0 ? 0.75 : 1.0);
        } else {
          vec3 shopIn = mix(vec3(0.95, 0.92, 0.82), hue, 0.25);
          float door = step(0.4, ux) * step(ux, 0.6) * step(vM.z, 2.2);
          float mul = step(0.01, abs(fract(vM.x / 1.6) - 0.5) - 0.48);
          col = mix(glass * 0.7 + shopIn * 0.32, shopIn * 0.55 * lit0, 0.35);
          col = mix(col, alu, max(mul, step(vM.z, 0.35)) * aaS);
          bEm += shopIn * (0.35 + 1.1 * uNight) * (1.0 - mul) * step(0.35, vM.z) * (1.0 - door * 0.5) * aaS;
        }

function platBox(n) {
  if (n.ov === JW.ov) return n.box;
  const c = [geoToWorld(n.s, n.w), geoToWorld(n.s, n.e), geoToWorld(n.n, n.w), geoToWorld(n.n, n.e)];
  n.box = [Math.min(c[0][0], c[1][0], c[2][0], c[3][0]), Math.min(c[0][1], c[1][1], c[2][1], c[3][1]), Math.max(c[0][0], c[1][0], c[2][0], c[3][0]), Math.max(c[0][1], c[1][1], c[2][1], c[3][1])];
  n.ov = JW.ov;
  return n.box;
}
function platUpdate(planes) {
  if (!JW.active || !JW.online || !PLAT.on) return;
  if (!PLAT.ready) { platInit(); if (!PLAT.ready) return; }
  const f = ++PLAT.frame, eye = R.eye, [la, lo] = worldToGeo(eye[0], eye[2]);
  const agl = eye[1] - Math.max(rawHeightGeo(la, lo), 0);
  const radius = game.touch ? 1800 : 3200;
  if (agl > 3500) { PLAT.active = []; platEvict(f); return; }
  const dLat = radius / KZ, dLon = radius / kxAt(la);
  const out = [], req = [];
  const K = R.H / (2 * R.tanY), SSE = game.touch ? 24 : 14;
  const dist = n => { const b = platBox(n), off = n.set.off || 37; if (authHides(b[0], b[1], b[2], b[3])) return 1e9; return Math.hypot(Math.max(b[0] - eye[0], 0, eye[0] - b[2]), Math.max(n.h0 - off - eye[1], 0, eye[1] - (n.h1 - off)), Math.max(b[1] - eye[2], 0, eye[2] - b[3])); };
  const ready = n => n.uri ? !!n.mesh : n.ext ? false : n.children.length > 0 && n.children.every(c => dist(c) > radius || ready(c));
  const visit = n => {
    const d = dist(n);
    if (d > radius) return;
    n.used = f;
    if (n.ext) { if (!n.extState) platLoadExt(n); return; }
    const b = platBox(n), off = n.set.off || 37;
    const vis = boxInFrustum(planes, b[0], n.h0 - off - 5, b[1], b[2], n.h1 - off + 5, b[3]);
    const big = n.size > 1400;                                        // very large nodes are only containers
    const wantKids = n.children.length && (big || !n.uri || n.ge * K / Math.max(d, 1) > SSE);
    if (!wantKids) { if (n.uri) { if (n.mesh) { if (vis) out.push(n); } else req.push([n, d]); } return; }
    const kidsReady = n.children.every(c => dist(c) > radius || ready(c));
    if (kidsReady || big || !n.uri) { for (const c of n.children) visit(c); if (!kidsReady && n.uri && !big && !n.mesh) req.push([n, d]); else if (!kidsReady && n.mesh && vis && !big) out.push(n); }
    else { if (n.mesh) { if (vis) out.push(n); } else req.push([n, d]); for (const c of n.children) { const dc = dist(c); if (dc <= radius) { c.used = f; if (c.uri && !c.mesh) req.push([c, dc + 50]); } } }
  };
  // (while Google's 3D is shown PLATEAU's city isn't drawn: none of it is fetched or decoded then, and what was
  // loaded goes after a while; only the developer panel's comparison round the car still loads its own)
  const googOn = typeof GOOG !== 'undefined' && GOOG.covering;
  for (const S of PLAT.list) {
    if (googOn) break;
    if (S.n < la - dLat || S.s > la + dLat || S.e < lo - dLon || S.w > lo + dLon) continue;
    if (S.state === 0 || (S.state === 3 && R.time > S.retry)) { platLoadSet(S); continue; }
    if (S.state !== 2) continue;
    S.used = f;
    visit(S.root);
  }
  PLAT.active = out;
  // requests: the comparison round the car first (p_dev.js), then nearest first
  const creq = platCmpUpdate(planes, f);
  req.sort((a, b) => a[1] - b[1]);
  let busy = 0; for (const w of PLAT.workers) busy += w.busy;
  for (const [n, d, kind] of creq.concat(req)) {
    if (busy >= PLAT.workers.length * 2) break;
    if (n.state !== 0) continue;
    platDispatch(n, d, kind ? kind === 'bldg' : PLAT.cmp > 0 && !n.children.length && !n.ext, kind && kind !== 'bldg' ? kind : undefined); busy++;
  }
  platEvict(f);
  // hide OSM buildings under loaded PLATEAU nodes (throttled). Not while Google's 3D is on: PLATEAU isn't
  // drawn then, and the OSM buildings are the ones shown round the car (and where Google is flat)
  const goog = typeof GOOG !== 'undefined' && GOOG.covering;
  if (goog !== PLAT.rectsGoog) { PLAT.rectsGoog = goog; PLAT.rectsDirty = true; PLAT.rectsT = 0; }
  if (PLAT.rectsDirty && R.time > (PLAT.rectsT || 0)) {
    PLAT.rectsDirty = false; PLAT.rectsT = R.time + 0.7;
    const rects = [];
    if (!goog) for (const S of PLAT.list) if (S.root) (function walk(n) { if (n.mesh && !n.mesh.empty && n.size <= 1400) rects.push([n.s, n.w, n.n, n.e, n.key, n.mesh.tex ? 1 : 0]); for (const c of n.children) walk(c); })(S.root);
    PLAT.rects = rects;
    PLAT.hideQueue = [...CITY.tiles.values()];
  }
  // driving: the map's buildings under PLATEAU come back within the near zone round the car (kept by their
  // centres, a little past the zone's outer edge; moved on every 60 m)
  const zp = typeof ZONE !== 'undefined' && ZONE.plat;
  if (!zp ? PLAT.keep : !PLAT.keep || Math.hypot(sim.pos[0] - PLAT.keep.x, sim.pos[2] - PLAT.keep.z) > 60) {
    if (zp) { const [la, lo] = worldToGeo(sim.pos[0], sim.pos[2]), r = ZONE.r1 + 70;
      PLAT.keep = { x: sim.pos[0], z: sim.pos[2], lat: la, lon: lo, r, rLat: r / 111320, rLon: r / (111320 * Math.cos(la * Math.PI / 180)), key: 'k' + Math.round(sim.pos[0]) + ',' + Math.round(sim.pos[2]) }; }
    else PLAT.keep = null;
    PLAT.hideQueue = [...CITY.tiles.values()];
  }
  // (a built-up area coming into use or going: its buildings are the sim's own, p_auth.js)
  const aid = typeof AUTH !== 'undefined' && AUTH.area ? AUTH.area.id + ':' + Math.round(AUTH.area.lat * 250) + ',' + Math.round(AUTH.area.lon * 250) : '';
  if (aid !== (PLAT.authId || '')) { PLAT.authId = aid; PLAT.hideQueue = [...CITY.tiles.values()]; }
  if (PLAT.hideQueue && PLAT.hideQueue.length) {
    let n = 0;
    while (PLAT.hideQueue.length && n < 2) { const t = PLAT.hideQueue.shift(); if (cityApplyHidden(t, PLAT.rects)) n++; }
  }
}
function platLoadExt(n) {
  n.extState = 1;
  fetch(n.ext, { mode: 'cors' }).then(r => r.json()).then(j => {
    const sub = n.set.conv(j.root, n, n.ext);
    n.children.push(sub); n.ext = null; n.extState = 2;
  }).catch(() => { n.extState = 3; });
}
function platDispatch(n, d, occ = false, kind = undefined) {
  const w = PLAT.workers.reduce((a, b) => (b.busy < a.busy ? b : a));
  const id = PLAT.nextId++;
  n.state = 1; n.job = id; w.busy++;
  n.refLat = (n.s + n.n) / 2; n.refLon = (n.w + n.e) / 2; n.kx = kxAt(n.refLat);
  PLAT.jobs.set(id, { n, w });
  w.postMessage({ id, url: n.uri, refLat: n.refLat, refLon: n.refLon, maxTex: game.touch ? 1024 : d < 700 ? 2048 : 1024, occ, kind });
}
function platReceive(r) {
  const job = PLAT.jobs.get(r.id); if (!job) return;
  PLAT.jobs.delete(r.id); job.w.busy--;
  if (job.goog) { googReceive(job, r); return; }
  const n = job.n;
  if (n.job !== r.id) { if (r.bitmap) r.bitmap.close(); return; }
  if (r.error) { n.state = 3; if (r.error !== '404') setTimeout(() => { if (n.state === 3) n.state = 0; }, 20000); return; }
  n.state = 2;
  if (!r.nv && !r.trees) { n.mesh = { empty: true, count: 0 }; PLAT.rectsDirty = true; return; }
  const tr0 = performance.now();
  // height datum: PLATEAU heights are ellipsoidal; compare building bases with the terrain
  const sm = new Float32Array(r.samples), diffs = [];
  for (let k = 0; k < sm.length; k += 3) {
    const zt = 2 ** 14, dt = JW.tiles.get('14/' + Math.floor((sm[k + 1] + 180) / 360 * zt) + '/' + Math.floor(mercY(sm[k]) * zt));
    if (!dt || !dt.dem) continue;                                     // only trust the 10 m elevation data
    const gh = sampleDem(dt, sm[k], sm[k + 1]); if (gh > -5) diffs.push(sm[k + 2] - gh);
  }
  if (diffs.length >= 3) {
    diffs.sort((a, b) => a - b);
    const med = diffs[diffs.length >> 1];
    const S = n.set; S.offs = S.offs || []; S.offs.push(clamp(med, 20, 55)); if (S.offs.length > 25) S.offs.shift();
    const so = [...S.offs].sort((a, b) => a - b); S.off = so[so.length >> 1];
  }
  if (r.trees) { n.trees = new Float32Array(r.trees); n.mesh = { empty: true, count: 0 }; return; }   // (PLATEAU's trees: only where they stand)
  const vao = gl.createVertexArray(); gl.bindVertexArray(vao);
  const mk = (data, loc, size, type, norm) => { const b = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, b); gl.bufferData(gl.ARRAY_BUFFER, data, gl.STATIC_DRAW); gl.enableVertexAttribArray(loc); gl.vertexAttribPointer(loc, size, type, norm, 0, 0); return b; };
  const bufs = [mk(r.pos, 0, 3, gl.FLOAT, false), mk(r.nrm, 1, 3, gl.BYTE, true), mk(r.uv, 2, 2, gl.FLOAT, false), mk(r.col, 3, 4, gl.UNSIGNED_BYTE, true)];
  const ib = gl.createBuffer(); gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, ib); gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, r.idx, gl.STATIC_DRAW); bufs.push(ib);
  gl.bindVertexArray(null);
  let tex = null;
  if (r.bitmap) {
    tex = gl.createTexture(); gl.bindTexture(gl.TEXTURE_2D, tex);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, r.bitmap);
    gl.generateMipmap(gl.TEXTURE_2D);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    if (anisoExt) gl.texParameterf(gl.TEXTURE_2D, anisoExt.TEXTURE_MAX_ANISOTROPY_EXT, game.touch ? 4 : Math.min(16, gl.getParameter(anisoExt.MAX_TEXTURE_MAX_ANISOTROPY_EXT)));
    PLAT.bytes += r.bitmap.width * r.bitmap.height * 4 * 1.33; n.tw = r.bitmap.width + "x" + r.bitmap.height;
    n.texBytes = r.bitmap.width * r.bitmap.height * 4 * 1.33;
    r.bitmap.close();
  }
  n.mesh = { vao, bufs, tex, count: r.ni };
  n.vbytes = r.nv * 32 + r.ni * 4; PLAT.bytes += n.vbytes; n.nv = r.nv; n.uri2 = n.uri.split("/").pop();
  n.hgrid = r.hgrid ? { x0: r.hgrid.x0, z0: r.hgrid.z0, gw: r.hgrid.gw, gh: r.hgrid.gh, cell: r.hgrid.cell, h: new Float32Array(r.hgrid.h) } : null;
  n.occ = r.occ ? { x0: r.occ.x0, z0: r.occ.z0, gw: r.occ.gw, gh: r.occ.gh, cell: r.occ.cell, grid: new Uint16Array(r.occ.grid), base: new Float32Array(r.occ.base), top: new Float32Array(r.occ.top), ctr: new Float32Array(r.occ.ctr) } : null;
  PLAT.loaded++;
  PLAT.rectsDirty = true;
  devTile('plat', r.msBuild, r.msNet, performance.now() - tr0, r.nv);
}
function platFree(n) {
  if (n.mesh && !n.mesh.empty) {
    gl.deleteVertexArray(n.mesh.vao); for (const b of n.mesh.bufs) gl.deleteBuffer(b);
    if (n.mesh.tex) gl.deleteTexture(n.mesh.tex);
    PLAT.bytes -= (n.vbytes || 0) + (n.texBytes || 0); PLAT.loaded--;
  }
  n.mesh = null; n.state = 0; n.job = 0; n.occ = null; n.hgrid = null; n.trees = null;
  PLAT.rectsDirty = true;
}
function platEvict(f) {
  if (f % 30) return;
  const budget = game.touch ? 250e6 : 700e6;
  const all = [];
  for (const S of PLAT.list.concat(PLAT.l3list, PLAT.nearList)) if (S.root) (function walk(n) { if (n.mesh) all.push(n); for (const c of n.children) walk(c); })(S.root);
  for (const n of all) if (n.used < f - 240) platFree(n);
  if (PLAT.bytes > budget) {
    all.filter(n => n.mesh).sort((a, b) => a.used - b.used).slice(0, 20).forEach(platFree);
  }
}
function drawPlateau() {
  if (!PLAT.ready || !PLAT.active.length) return;
  const pr = PLAT.prog, u = pr.u;
  setCommon(pr);
  gl.uniform1i(u.uTex, 8);
  gl.disable(gl.CULL_FACE);
  gl.activeTexture(gl.TEXTURE8);
  for (const n of PLAT.active) {
    if (!n.mesh || n.mesh.empty || (PLAT.cmp && platCmpNear(n))) continue;
    gl.uniform4f(u.uTileA, n.refLon - JW.lon0, n.refLat - JW.lat0, n.refLat, n.kx);
    gl.uniform1f(u.uYOff, n.set.off || 37);
    if (n.mesh.tex) { gl.bindTexture(gl.TEXTURE_2D, n.mesh.tex); gl.uniform1i(u.uHasTex, 1); } else gl.uniform1i(u.uHasTex, 0);
    // driving: plain boxes (no photo) give way near the car to the sim's own buildings; photo-textured ones stay
    if (ZONE.plat) gl.uniform1f(u.uZoneSide, n.mesh.tex && !AUTH.on ? 0 : 1);
    gl.bindVertexArray(n.mesh.vao);
    gl.drawElements(gl.TRIANGLES, n.mesh.count, gl.UNSIGNED_INT, 0);
  }
  gl.activeTexture(gl.TEXTURE0);
  gl.enable(gl.CULL_FACE);
}

function platCmpSet(m, force) {
  m = m | 0;
  if (m === (PLAT.cmp || 0) && !force) return;

function platCmpUpdate(planes, f) {
  if (!PLAT.cmp || !CAR.on || !JW.active) {
