// ---------- geometry helpers ----------
function fuseAt(F, z) {
  for (let i = 0; i < F.length - 1; i++) {
    const a = F[i], b = F[i + 1];
    if (z <= b[0]) { const t = clamp((z - a[0]) / (b[0] - a[0]), 0, 1); return a.map((v, k) => lerp(v, b[k], t)); }
  }
  return F[F.length - 1];
}
function superRing(z, hw, hh, yc, n = 22, e = 2.5) {
  return Array.from({ length: n }, (_, k) => {
    const t = k / n * Math.PI * 2, c = Math.cos(t), s = Math.sin(t);
    return [hw * Math.sign(c) * Math.pow(Math.abs(c), 2 / e), yc + hh * Math.sign(s) * Math.pow(Math.abs(s), 2 / e), z];
  });
}
const NACA = xc => 5 * (0.2969 * Math.sqrt(xc) - 0.1260 * xc - 0.3516 * xc * xc + 0.2843 * xc ** 3 - 0.1036 * xc ** 4);
// airfoil ring. horizontal: points (span, base±t, z) ; vertical: (±t, span, z)
function foil(span, base, zLE, chord, thick, xc0, xc1, vertical = false) {
  // fixed station count so sections with different extents can be lofted together
  const ss = xc0 <= 1e-3 ? [0, 0.015, 0.05, 0.12, 0.22, 0.35, 0.5, 0.65, 0.8, 0.92, 1] : [0, 0.35, 0.7, 1];
  const xs = ss.map(v => xc0 + (xc1 - xc0) * v);
  const P = (xc, sgn) => {
    const t = NACA(Math.max(xc, 0)) * thick * chord * sgn, z = zLE + xc * chord;
    return vertical ? [t, span, z] : [span, base + t, z];
  };
  const up = xs.slice().reverse().map(x => P(x, 1));
  const lo = xs.map(x => P(x, -1));
  if (xc0 <= 1e-3) lo.shift();
  if (xc1 >= 0.999) lo.pop();
  return [...up, ...lo];
}
function wingFns(w) {
  const span = w.tipX - w.rootX, f = ax => clamp((ax - w.rootX) / span, 0, 1);
  const W = {
    y: ax => w.y + (ax - w.rootX) * Math.tan(w.dih * DEG),
    c: ax => lerp(w.rootC, w.tipC, f(ax)),
    le: ax => lerp(w.rootLE, w.tipLE, f(ax)),
    t: ax => lerp(w.t0, w.t1, f(ax)),
  };
  W.hinge = x => { const ax = Math.abs(x); return [x, W.y(ax), W.le(ax) + w.hinge * W.c(ax)]; };
  return W;
}

// forces (body frame) and moments about the centre of gravity from every piece
// o: flapF, elev, ail, rud (control positions), dq (slipstream dynamic pressure on the tail),
//    hgt(p) (height of a piece above the ground, for ground effect), gust(p) (extra wind at a piece, body frame)
const _v = [0, 0, 0];
function aeroElements(E, P, vb, w, rho, o) {
  const F = [0, 0, 0], M = [0, 0, 0];
  const V = Math.hypot(vb[0], vb[1], vb[2]) || 1, sb = vb[0] / V;
  let clw = 0, stw = 0, eps = 0;
  const aS0 = P.aStall - P.flapStall * o.flapF;
  for (let pass = 0; pass < 2; pass++) for (const p of E.pcs) {
    if ((p.kind === 0) !== (pass === 0)) continue;              // wing first: its lift sets the downwash on the tail
    const r = p.r;
    let vx = vb[0] + w[1] * r[2] - w[2] * r[1], vy = vb[1] + w[2] * r[0] - w[0] * r[2], vz = vb[2] + w[0] * r[1] - w[1] * r[0];
    if (o.gust) { const g = o.gust(p); vx -= g[0]; vy -= g[1]; vz -= g[2]; }
    const s = p.s, n = p.n, vs = vx * s[0] + vy * s[1] + vz * s[2];
    const px = vx - vs * s[0], py = vy - vs * s[1], pz = vz - vs * s[2], Vp = Math.hypot(px, py, pz);
    if (Vp < 0.5) continue;
    let a = Math.atan2(-(px * n[0] + py * n[1] + pz * n[2]), -pz);
    let qd = 0.5 * rho * Vp * Vp, slope, cl0 = 0, dcl = 0, aS, cdi, ge = 1;
    if (p.kind === 0) {
      a += p.side * E.gdi * sb;                                   // effective dihedral (wing position, fuselage flow)
      if (o.hgt) { const h = Math.max(o.hgt(p), 0.3) / P.b; ge = (16 * h * h) / (1 + 16 * h * h); }
      slope = P.CLa * (1 + 0.06 * (1 - ge));
      cl0 = P.CL0 + (p.flap ? P.flapCL * o.flapF * E.flapK : 0);
      dcl = p.ail ? E.ka * o.ail * p.ail : 0;
      aS = aS0 + 0.03 * p.span01;                                 // washout: the root stalls first, the ailerons keep working
      cdi = P.K * ge;
    } else if (p.kind === 1) {
      a += E.it - eps; slope = E.at; dcl = E.ke * o.elev; aS = 0.32; cdi = 0.09;
    } else {
      slope = E.af; dcl = E.kr * o.rud; aS = 0.36; cdi = 0.09;
    }
    const over = Math.max(a - aS, -aS - 0.06 - a), st = smoothstep(0, 0.07, over);
    const lin = cl0 + slope * a + dcl * (1 - 0.5 * st);
    const CL = lerp(lin, 1.15 * Math.sin(2 * a), st), sa = Math.sin(a);
    const CD = cdi * lin * lin * (1 - st) + st * 1.25 * sa * sa;
    // lift is perpendicular to the flow and the span, drag along the flow
    let lx = s[1] * pz - s[2] * py, ly = s[2] * px - s[0] * pz, lz = s[0] * py - s[1] * px;
    const ll = Math.hypot(lx, ly, lz) || 1, k = qd * p.S;
    // the propeller slipstream over the tail makes the elevator and rudder bite harder (taxiing, low speed)
    const cx = p.kind ? o.dq * p.wash * p.S * dcl * (1 - 0.5 * st) : 0;
    const fx = k * (CL * lx / ll - CD * px / Vp) + cx * lx / ll, fy = k * (CL * ly / ll - CD * py / Vp) + cx * ly / ll, fz = k * (CL * lz / ll - CD * pz / Vp) + cx * lz / ll;
    // the aircraft's reference lift curve already includes the tailplane: its lift only turns the aircraft
    if (p.kind === 1) { F[0] -= k * CD * px / Vp; F[1] -= k * CD * py / Vp; F[2] -= k * CD * pz / Vp; }
    else { F[0] += fx; F[1] += fy; F[2] += fz; }
    // pitch: the wing's force is taken at the height of the CG (the reference pitch data already holds the
    // effect of a high or low wing); roll and yaw use the true position
    M[0] += (p.kind === 0 ? 0 : r[1]) * fz - r[2] * fy; M[1] += r[2] * fx - r[0] * fz; M[2] += r[0] * fy - r[1] * fx;
    p.cl = CL; p.st = st; p.a = a;                               // kept for the physics view
    if (p.kind === 0) { clw += CL * p.S; stw += st * p.S; }
    if (pass === 0) eps = E.deps * (clw / P.S) / P.CLa;
  }
  return { F, M, CLw: clw / P.S, stallW: stw / P.S };
}

function deriveAircraft(A) {
  const P = { ...A.phys, mass: A.massNow || A.phys.mass }, m = P.mass, g = 9.81;
  const I = [m * (0.18 * P.L) ** 2, m * (0.19 * (P.L + P.b) / 2) ** 2, m * (0.12 * P.b) ** 2];
  const wr = A.gear.wheelR, bottom = A.gear.nose[1] - wr;
  const zn = A.gear.nose[2], zm = A.gear.mains[0][2];
  const Fn = m * g * zm / (zm - zn), Fm = m * g * -zn / (zm - zn) / (A.gear.mains.length * 2);
  const defl = clamp(wr * 0.38, 0.05, 0.22);
  const spring = (load, share) => { const k = load / defl; return { k, c: 2 * 0.65 * Math.sqrt(k * load / g) }; };
  const wheels = [{ off: [0, bottom, zn], ...spring(Math.max(Fn, m * g * 0.35)), nose: true }];   // nose leg sized for braking loads
  for (const mm of A.gear.mains) for (const s of [-1, 1]) wheels.push({ off: [s * mm[0], bottom, mm[2]], ...spring(Fm) });
  const F = A.fuse, zTail = F[F.length - 1][0], tailSt = fuseAt(F, zTail - (zTail - F[0][0]) * 0.1);
  const W = wingFns(A.wing), hs = A.hstab, vs = A.vstab;
  const hard = [
    { off: [0, F[0][3], F[0][0] - (A.engines[0].type === 'nose' ? 0.4 : 0)], name: '機首' },
    { off: [-A.wing.tipX, W.y(A.wing.tipX), W.le(A.wing.tipX) + W.c(A.wing.tipX) * 0.4], name: '左の翼端' },
    { off: [A.wing.tipX, W.y(A.wing.tipX), W.le(A.wing.tipX) + W.c(A.wing.tipX) * 0.4], name: '右の翼端' },
    { off: [0, vs.y1, vs.topLE + 0.2], name: '垂直尾翼' },
    { off: [-hs.span, hs.y, (hs.tipLE + hs.te) / 2], name: '水平尾翼' },
    { off: [hs.span, hs.y, (hs.tipLE + hs.te) / 2], name: '水平尾翼' },
  ];
  for (const s of [-1, 1]) { const x = A.wing.tipX * 0.55; hard.push({ off: [s * x, W.y(x) - W.t(x) * W.c(x) * 0.3, W.le(x) + W.c(x) * 0.4], name: s < 0 ? '左翼' : '右翼' }); }
  for (const z of [F[0][0] * 0.45, F[F.length - 1][0] * 0.3]) { const f = fuseAt(F, z); hard.push({ off: [0, f[3] - f[2], z], name: '胴体下面' }); }
  { const zc = A.eye[2], f = fuseAt(F, zc); hard.push({ off: [0, f[3] + f[2] + (A.canopy ? 0.35 : 0), zc], name: '機体の上面' }); }
  for (const e of A.engines) {
    if (e.type === 'nose') hard.push({ off: [0, e.y - e.r, e.z], name: 'プロペラ' });
    if (e.type === 'wing') for (const s of [-1, 1]) hard.push({ off: [s * e.x, e.y - e.r, e.z], name: 'プロペラ' });
    if (e.type === 'pod') for (const s of [-1, 1]) { const fl = e.flat || 1; hard.push({ off: [s * e.x, e.y + e.r * (1 - fl) * 0.5 - e.r * fl, (e.z0 + e.z1) / 2], name: 'エンジン' }); }
  }
  // propeller: design speed of the fixed-pitch prop, and the fin offset that cancels the
  // left-turning tendencies at cruise (as real single-engine aircraft are rigged)
  const E = P.engine, prop = E.type !== 'jet';
  const pr = {};
  if (prop) {
    const Vc = A.v.cruise / 1.94384, qc = 0.5 * 1.225 * Vc * Vc, CLc = m * g / (qc * P.S), ac = (CLc - P.CL0) / P.CLa;
    const Tc = (P.CD0 + (P.CDgear || 0) * (A.gear.fixed ? 1 : 0) + P.K * CLc * CLc) * qc * P.S;
    pr.Vd = Vc * 1.2;                                   // prop efficiency peaks a little above cruise
    pr.a0 = 0.012 * P.b; pr.a1 = 0.15 * P.b;            // slipstream / P-factor lever arms (m, m per rad)
    pr.fin = Tc * (pr.a0 + pr.a1 * ac) / qc;            // fin offset yaw per unit dynamic pressure (N m / Pa)
    pr.Ip = 2.5 * E.P / 92000;                          // propeller polar moment of inertia (kg m^2)
    pr.tq = 0.12 * Math.min(1, 92000 / E.P);                                       // share of engine torque felt as a rolling moment
    // wing/aileron rigging cancels the torque at cruise power; being aerodynamic it scales with dynamic pressure
    const thrC = TRIM[A.id] ? TRIM[A.id].c_level_0[1] / 100 : 0.65, omC = rpmFor(thrC, Vc) * Math.PI / 30;
    pr.rig = (0.04 + 0.96 * thrC) * E.P / 0.85 / omC * pr.tq / qc;
  }
  const ac = { ...P, I, prop: prop ? pr : null, // jets: yaw damper / fly-by-wire hides these couplings
    Cnda: P.Cnda ?? (prop ? -0.1 * P.Clda : 0), Cnp: P.Cnp ?? (prop ? -0.04 : 0), Clr: P.Clr ?? (prop ? 0.06 : 0) };
  ac.el = buildElements(A, ac);
  return {
    AC: ac, WHEELS: wheels, WHEEL_BOTTOM: bottom, HARDPTS: hard,
    TAILSKID: { off: [0, tailSt[3] - tailSt[2], zTail - (zTail - F[0][0]) * 0.1], k: 30 * m, c: 4 * m },
  };
}
function setAircraft(id, custom = true) {
  ACFT = AIRCRAFT.find(a => a.id === id) || AIRCRAFT[0];
  const A = ACFT;
  // weight from the load (fuel, people, cargo); the speeds that depend on it scale with the square root
  A.custom = custom; A.massNow = typeof acMass === 'function' ? acMass(A, custom) : A.phys.mass;
  if (!A.v0) A.v0 = { ...A.v };
  const k = Math.sqrt(A.massNow / A.phys.mass);
  A.v = { ...A.v0 }; for (const key of ['s0', 's1', 'rotate', 'appr']) A.v[key] = Math.round(A.v0[key] * k);
  const d = deriveAircraft(ACFT);
  AC = d.AC; WHEELS = d.WHEELS; WHEEL_BOTTOM = d.WHEEL_BOTTOM; HARDPTS = d.HARDPTS; TAILSKID = d.TAILSKID;
  sim.contacts = WHEELS.map(() => false);
}

const sim = {
  pos: [0, 0, 0], vel: [0, 0, 0], q: [0, 0, 0, 1], w: [0, 0, 0],
  thr: 0, elev: 0, ail: 0, rud: 0, trim: 0, brake: false, pIn: 0, rIn: 0,
  flapIdx: 0, flapDeg: 0, gearDown: true, gearPos: 1, smoke: false,
  rpm: 0, propAng: 0, n1: 0,
  onGround: false, mainContact: false, airTime: 0, crashed: false, crashReason: '',
  alpha: 0, beta: 0, ias: 0, tas: 0, gLoad: 1, stallW: 0, stallWarn: false, agl: 0, groundH: 0, vs: 0,
  contacts: [], assistOn: false,
};

// fixed-pitch propeller rpm: grows with power and with airspeed (the prop is driven by the air);
// about 2,300 static at full throttle, 2,500 in a Vy climb, 2,400 at cruise, redline near 2,700 in a dive
function rpmFor(thr, u, pw = 1, dead = false) {
  if (dead) return clamp(14 * u, 0, 1300);
  return Math.min(650 + 1650 * Math.sqrt(Math.max(thr, 0)) * Math.pow(Math.max(pw, 0.1), 0.3) + 6 * Math.max(u, 0), 2850);
}
// ICAO standard atmosphere (troposphere + lower stratosphere); ISA.dev shifts the temperature
const ISA = { dev: 0 };
function atmos(h) {
  h = Math.max(h, -500);
  let Ts, p;
  if (h < 11000) { Ts = 288.15 - 0.0065 * h; p = 101325 * Math.pow(Ts / 288.15, 5.25588); }
  else { Ts = 216.65; p = 22632.1 * Math.exp(-(h - 11000) / 6341.62); }
  const T = Ts + ISA.dev, rho = p / (287.05 * T);
  return { T, p, rho, a: Math.sqrt(1.4 * 287.05 * T), sigma: rho / 1.225 };
}
// wind near the ground is slowed by friction: log profile referenced to the 10 m reported wind
function windGradient(agl) { const h = clamp(agl, 1, 400); return clamp(Math.log(h / 0.03) / Math.log(10 / 0.03), 0.55, 1.35); }

// wind blows from the active runway's heading (a headwind for takeoff and landing)
function windAt(p, t, agl = p[1]) {
  const gust = Math.min(1, Math.max(agl, 0) / 300 + 0.4), wg = windGradient(agl);
  const sw = typeof SCHOOL !== 'undefined' && SCHOOL.active && SCHOOL.wind;
  const w = sw || RWY.windDir || [-3, 0, 0];
  if (sw) { const k = SCHOOL.gust; return [w[0] * wg + perlin(t * 0.35, 1.7) * 1.6 * gust * k, perlin(t * 0.5, 4.4) * 0.5 * gust * k, w[2] * wg + perlin(t * 0.31, 8.2) * 1.3 * gust * k]; }
  return [w[0] * wg + perlin(t * 0.35, 1.7) * 1.2 * gust, perlin(t * 0.5, 4.4) * 0.45 * gust, w[2] * wg + perlin(t * 0.31, 8.2) * 0.9 * gust];
}
// the trim range (fraction of elevator travel): enough to fly hands-off from the approach to cruise
const TRIM_MAX = 0.45;
// T key: the trim that holds 1 g at the present speed with the stick let go. It starts from the elevator in
// use now and corrects it for the angle of attack being off the 1 g value (a pull-up is not trimmed in)
function trimLevel() {
  const s = sim, flapF = ACFT.wing.flap ? s.flapDeg / 30 : 0, rho = s.rho || atmos(s.pos[1]).rho;
  const a1 = trimFor(Math.max(s.tas || 0, 20), flapF, 1, rho).alpha;
  return clamp(s.elev + s.trim + AC.Cma / AC.Cmde * (s.alpha - a1), -TRIM_MAX, TRIM_MAX);
}
// lift coefficient needed for a load factor, and the elevator that holds that angle of attack
function trimFor(V, flapF, n = 1, rho = 1.225) {
  const qd = 0.5 * rho * V * V, cl0 = AC.CL0 + AC.flapCL * flapF;
  const alpha = (n * AC.mass * 9.81 / (qd * AC.S) - cl0) / AC.CLa;
  return { alpha, elev: clamp((-AC.Cma * Math.sin(alpha) - AC.Cm0) / AC.Cmde, -1, 1) };
}

// Flight assist (fly-by-wire): stick left/right = bank angle, stick fore/aft = climb angle.
// Letting go of the stick levels the wings and returns to level flight.
// direct (no assist): pitch/roll = control deflection per unit of stick. Pitch is scaled up at low
// airspeed so that the same stick input gives a similar pitch response at every speed.
const SENS = {
  low: { direct: 0.45, pitch: 0.16, roll: 0.36, pMax: 0.4, rMax: 0.6, bank: 30, gamma: 8 },
  mid: { direct: 0.7, pitch: 0.26, roll: 0.55, pMax: 0.55, rMax: 0.8, bank: 45, gamma: 12 },
  high: { direct: 1.0, pitch: 0.42, roll: 0.85, pMax: 0.75, rMax: 1, bank: 60, gamma: 18 },
};
function applyControls(s, V, alpha, qd, qdc, flapF, aS, cl0) {
  const sens = SENS[game.sens] || SENS.low;
  const airborne = !s.onGround && s.airTime > 0.6 && V > 12;
  const leveling = game.levelT > 0;
  const auto = game.ap && airborne && !s.crashed;
  s.assistOn = (game.assist || leveling || auto) && airborne && !s.crashed;
  if (!s.assistOn) {
    s.gI = 0; s.bI = 0;
    const vr = ACFT.v.cruise / 1.94384, spd = Math.max(V, 12);
    // the stick maps to deflection along one smooth, progressive curve (no sudden jump near full travel).
    // Full stick gives sens.pMax of the elevator at cruise; slower flight allows more, so a flare or a
    // stall is still possible at low speed.
    const pMax = Math.min(1, sens.pMax * clamp(vr / spd, 1, 2.6)), rMax = Math.min(1, sens.rMax * clamp(vr / spd, 1, 2));
    // full elevator for rotation, blended out over the first seconds after lift-off
    const kg = s.onGround ? 1 : s.airTime < 3 ? 1 - s.airTime / 3 : 0;
    // the curve is gentle around the centre at cruise, and closer to linear when slow (rotation, flare)
    const lin = Math.max(clamp(0.45 + (vr / spd - 1) * 0.35, 0.45, 0.8), 0.45 + 0.35 * kg);
    const shape = (x, a) => a * x + (1 - a) * x * x * x;
    const kp = Math.max(pMax, kg), kr = rMax;
    let el = kp * shape(clamp(s.pIn, -1, 1), lin);
    // just after trimming (T) the stick is still held: the part of it now in the trim is not applied twice
    if (s.trimHold) { if (Math.abs(s.pIn) < 0.04 || Math.sign(el) !== Math.sign(s.trimHold)) s.trimHold = 0; else el -= Math.sign(el) * Math.min(Math.abs(el), Math.abs(s.trimHold)); }
    s.elev = clamp(el, -1, 1); s.ail = clamp(kr * shape(clamp(s.rIn, -1, 1), 0.45), -1, 1);
    // with the assist on, the takeoff rotation is guided: back stick asks for a pitch attitude (up to about
    // 10 degrees) instead of full elevator, so holding the key can't over-rotate and stall at lift-off
    if (game.assist && !s.crashed && V > 12 && s.pIn > 0.05 && (s.onGround || s.airTime < 3)) {
      const f = qrot(s.q, [0, 0, -1]), th = Math.asin(clamp(f[1], -1, 1));
      const thMax = (AC.maxG > 6 ? 13 : ACFT.v.cruise > 200 ? 11 : 10) * DEG;
      let thCmd = Math.min(1, s.pIn * 1.8) * thMax;               // half stick already asks for most of the rotation
      if (alpha > aS - 0.06) thCmd = Math.min(thCmd, th - (alpha - (aS - 0.06)) * 2);   // never into the stall
      const e = clamp((thCmd - th) * 10 - s.w[0] * 2.5, -1, 1);
      s.elev = clamp(Math.min(e, 1) - s.trim, -1, 1);
    }
    return;
  }
  const r = qrot(s.q, [1, 0, 0]), up = qrot(s.q, [0, 1, 0]);
  const bank = Math.atan2(-r[1], up[1]);
  const spd = vlen(s.vel) || 1, gamma = Math.asin(clamp(s.vel[1] / spd, -1, 1));
  const p = -s.w[2];
  // --- roll: bank-angle command ---
  const bankMax = Math.min(sens.bank * (AC.rollRate > 150 ? 1.4 : 1), 80) * DEG;
  let rIn = leveling ? 0 : s.rIn, pIn = leveling ? 0 : s.pIn;
  const gMax0 = sens.gamma * (AC.maxG > 6 ? 1.6 : 1) * DEG;
  if (auto) {
    // autopilot: fly toward the destination (or hold heading) and hold the selected altitude
    const f = qrot(s.q, [0, 0, -1]), hdg = (Math.atan2(f[0], -f[2]) / DEG + 360) % 360;
    // lateral: NAV (to the destination, when there is one) or HDG (the selected heading), as the panel's keys set
    if (game.apLat !== 'hdg' && typeof JW !== 'undefined' && JW.active && JW.dest) { const g = planeGeo(); game.apHdg = geoBearing(g[0], g[1], JW.dest.lat, JW.dest.lon); }
    const err = ((game.apHdg - hdg + 540) % 360) - 180;
    rIn = clamp(clamp(err * 1.2, -25, 25) * DEG / bankMax, -1, 1);
    let alt = game.apAlt;
    const look = vmadd(s.pos, s.vel, 45), terr = Math.max(terrainHeight(look[0], look[2]), terrainHeight(s.pos[0], s.pos[2]), 0);
    if (alt < terr + 300) alt = terr + 300;                              // terrain safety
    // vertical: ALT hold, or VS / FLC (a set climb or descent rate) toward the selected altitude, captured there
    const vm = game.apVert, dAlt = alt - s.pos[1];
    if ((vm === 'vs' || vm === 'flc') && Math.abs(dAlt) > 25) {
      const want = (vm === 'flc' ? Math.sign(dAlt) * 700 : (game.apVs || 0)) * 0.00508;          // ft/min -> m/s
      const toward = Math.sign(want) === Math.sign(dAlt) || want === 0 ? want : 0;
      pIn = clamp(clamp(Math.asin(clamp(toward / spd, -0.5, 0.5)) / DEG + (toward - s.vel[1]) * 0.6, -6, 8) * DEG / gMax0, -1, 1);
    } else {
      if (vm === 'vs' || vm === 'flc') { game.apVert = 'alt'; }
      pIn = clamp(clamp(dAlt * 0.02, -4, 5) * DEG / gMax0, -1, 1);
    }
    // autothrottle (turbine aircraft, as on real airliners): holds the speed the autopilot was engaged at
    if (ACFT.v.cruise > 150 && game.apSpd) {
      const kt = V * Math.sqrt(s.rho ? s.rho / 1.225 : 1) * 1.94384;
      s.thr = clamp(s.thr + clamp((game.apSpd - kt) * 0.004 - (s.vs || 0) * 0.0006, -0.004, 0.004), 0.02, 0.95);
    }
  }
  const bankCmd = rIn * bankMax;
  const pMax = AC.rollRate * DEG * (game.sens === 'high' ? 1 : game.sens === 'mid' ? 0.75 : 0.55);
  // integral term: a steady rolling moment (engine torque at climb power, fuel imbalance) can't leave the wings a little low
  const bErr = bankCmd - bank;
  const hold = Math.abs(bankCmd) < 2 * DEG && Math.abs(bErr) < 3 * DEG;            // only while holding the wings level
  s.bI = hold ? clamp((s.bI || 0) + bErr * (1 / 120) * 1.5 * Math.min(1, AC.rollRate / 50), -0.08, 0.08) : (s.bI || 0) * 0.98;
  const pCmd = clamp(bErr * 1.6 + s.bI, -pMax, pMax);
  const kAil = Math.abs(AC.Clp) * AC.b / (2 * Math.max(V, 10) * AC.Clda);
  s.ail = clamp((pCmd + 0.9 * (pCmd - p)) * kAil, -1, 1);
  // --- pitch: flight-path-angle command, turns are flown level automatically ---
  const gMax = sens.gamma * (AC.maxG > 6 ? 1.6 : 1) * DEG;
  const gammaCmd = pIn * gMax;
  // integral term: removes the steady sink that a pure proportional command leaves in turns
  s.gI = clamp((s.gI || 0) * (1 - 0.1 / 120) + clamp(gammaCmd - gamma, -1.5 * DEG, 1.5 * DEG) * (1 / 120) * 0.35, -0.08, 0.08);
  const gdot = clamp((gammaCmd - gamma) * 0.7 + s.gI, -AC.pitchRate * DEG, AC.pitchRate * DEG);
  const nCmd = clamp(Math.cos(gamma) / Math.max(Math.cos(bank), 0.35) + spd * gdot / 9.81, -1, AC.maxG);
  const aFree = (nCmd * AC.mass * 9.81 / Math.max(qd * AC.S, 1) - cl0) / AC.CLa;
  const aReq = clamp(aFree, -aS * 0.7, aS - 0.04);          // stall protection
  if (aReq !== aFree) s.gI *= 0.97;                         // no wind-up while the wing is at its limit
  const ratio = qd / Math.max(qdc, 1);
  let e = (-AC.Cma * Math.sin(aReq) - AC.Cm0) / AC.Cmde * ratio;
  // pitch-rate loop sized by the aircraft's own inertia and elevator power, so a heavy airliner is damped as
  // well as a light trainer (a fixed gain left big aircraft porpoising on the approach): the rate asked for is
  // the steady turn/pull-up rate plus a correction that brings the angle of attack to its target
  const Mde = Math.max(qdc * AC.S * AC.c * AC.Cmde / AC.I[0], 0.02);
  const qCmd = 9.81 / spd * (nCmd - Math.cos(gamma) * Math.cos(bank)) + 2.0 * (aReq - alpha);
  e += 1.2 * (aReq - alpha) + clamp(2.5 / Mde, 0.1, 30) * (qCmd - s.w[0]);
  s.elev = clamp(e - s.trim, -1, 1);
}

function physicsStep(dt, t) {
  const s = sim;
  if (sysFlapsOK()) s.flapDeg += clamp(FLAP_DEG[s.flapIdx] - s.flapDeg, -8 * dt, 8 * dt);
  if (ACFT.gear.fixed) { s.gearDown = true; s.gearPos = 1; }
  else if (!(s.onGround && !s.gearDown) && sysGearOK()) s.gearPos = clamp(s.gearPos + (s.gearDown ? dt : -dt) / 4.5, 0, 1);
  const flapF = ACFT.wing.flap ? s.flapDeg / 30 : 0;
  const atm = atmos(s.pos[1]), rho = atm.rho;
  const ground = groundAt(s.pos[0], s.pos[2]);
  const agl = s.pos[1] + WHEEL_BOTTOM - ground.h;
  const qi = qconj(s.q);
  const vb = qrot(qi, vsub(s.vel, windAt(s.pos, t, agl + 2)));
  const V = vlen(vb);
  const u = -vb[2];
  const alpha = Math.atan2(-vb[1], u);
  const beta = V > 0.5 ? Math.asin(clamp(vb[0] / V, -1, 1)) : 0;
  const qd = 0.5 * rho * V * V;
  const jet = AC.engine.type === 'jet';
  const qdc = qd + (jet ? 0 : 240 * s.thr * Math.sqrt(AC.mass / 1000));
  const aS = AC.aStall - AC.flapStall * flapF;
  const cl0 = AC.CL0 + AC.flapCL * flapF;
  applyControls(s, V, alpha, qd, qdc, flapF, aS, cl0);

  const F = [0, 0, 0];
  const mach = V / atm.a, EL = AC.el;
  // --- engines ---
  let T;
  if (jet) {
    const E = AC.engine, dry = SYS.run ? Math.min(s.thr / 0.95, 1) : 0;
    s.n1 += (dry - s.n1) * Math.min(1, dt / E.spool * (s.n1 < dry ? 1 : 1.6));
    const ab = E.ab && s.thr > 0.95 && s.n1 > 0.9 ? E.ab : 0;
    T = SYS.run ? (E.T * (0.06 + 0.94 * s.n1 * s.n1) + ab) * Math.pow(atm.sigma, 0.7) * (1 - 0.25 * Math.min(mach, 1.2)) : 0;
    s.ab = ab > 0;
  } else {
    // normally aspirated piston power falls faster than density (Gagg-Ferrar); a fixed-pitch prop
    // is least efficient at low speed, and its static thrust falls off as the aircraft accelerates
    const E = AC.engine, pw = Math.max(0, 1.132 * atm.sigma - 0.132) * SYS.pf, ua = Math.max(u, 0), eDead = s.dead || !SYS.run;
    s.shaftP = eDead ? 0 : (0.04 + 0.96 * s.thr) * pw * E.P / 0.85;
    if (eDead) T = -0.026 * qd * AC.S * (AC.mass / 1050) ** 0.3;                 // windmilling prop drag
    else {
      const eta = Math.pow(Math.min(1, Math.max(ua, 1) / AC.prop.Vd), 0.35);
      const thrEff = 0.04 + 0.96 * s.thr;
      T = thrEff * pw * Math.min(E.T0 * (1 - 0.25 * Math.min(ua / AC.prop.Vd, 1)), eta * E.P / Math.max(ua, 1)) - (1 - s.thr) * 2.5 * ua * AC.mass / 1000;
    }
    s.ab = false;
  }
  if (jet) s.shaftP = 0;
  s.oat = atm.T - 273.15; s.thrust = T; s.rho = rho; s.nProp = 0;
  // density altitude: the ISA altitude that has this air density
  s.densAlt = (1 - Math.pow(Math.min(atm.sigma, 1.2), 0.234969)) * 44330.8;
  F[2] -= T;
  // --- moments that are not from the lifting pieces (pilot convention: roll right, pitch up, yaw right) ---
  const p = -s.w[2], qq = s.w[0], r = -s.w[1];
  const hv = 0.25 * rho * V;
  const elevEff = clamp(s.elev + s.trim, -1, 1);
  // yaw: the propeller's left-turning tendencies (slipstream swirl + P-factor, minus the fin offset that
  // cancels them at cruise) and what the pieces don't produce of adverse yaw
  let Nx = qd * AC.S * AC.b * EL.dCnda * s.ail + hv * AC.S * AC.b * AC.b * EL.dCnp * p, Rl = 0, Ng = 0;
  if (AC.prop) {
    const Pr = AC.prop, Tp = Math.max(T, 0);
    s.nProp = -(Tp * (Pr.a0 + Pr.a1 * clamp(alpha, -0.1, 0.35))) + Pr.fin * qd;
    Nx += s.nProp;
    const om = Math.max(s.rpm, 300) * Math.PI / 30;
    Rl = -(s.shaftP || 0) / om * Pr.tq + Pr.rig * qd;                                          // engine torque reaction (partly rigged out): rolls left
    Ng = Pr.Ip * om * qq;                                                    // gyroscopic precession: pitch up -> yaw right
  }
  // automatic rudder (ball kept centred) for the assist and for keyboard pilots who want it
  const autoR = s.assistOn || (typeof game !== 'undefined' && game.autoRud);
  const rAuth = Math.max(qdc * AC.S * AC.b * AC.Cndr, 1);
  s.rudNeed = clamp(-(Nx + Ng) / rAuth, -1, 1);
  let yd = 0;
  if (autoR && !s.onGround && V > 10) {                                        // yaw damper: only the yaw rate a coordinated turn doesn't need
    const rgt = qrot(s.q, [1, 0, 0]), upv = qrot(s.q, [0, 1, 0]), sb = -rgt[1] / Math.hypot(rgt[1], upv[1]);
    yd = -clamp(1.5 * AC.I[2] / rAuth, 0.8, 20) * (r - 9.81 * sb / V);          // gain sized by yaw inertia: big jets need a strong damper
  }
  s.rudAuto = autoR ? clamp(-(Nx + Ng) / rAuth + 1.5 * beta + yd, -1, 1) * (s.onGround && V < 8 ? V / 8 : 1) : 0;
  const rudEff = clamp(s.rud + s.rudAuto, -1, 1);
  // --- aerodynamics: the lifting pieces ---
  const upB = qrot(qi, [0, 1, 0]), hCG = s.pos[1] - ground.h;
  // turbulence varies across the span and along the fuselage: bumps that roll and pitch the aircraft
  const gA = (typeof SCHOOL !== 'undefined' && SCHOOL.active && SCHOOL.wind ? SCHOOL.gust : 1) * Math.min(1, Math.max(agl, 0) / 300 + 0.4) * 0.9 * smoothstep(3, 40, agl);
  const gW = perlin(t * 0.9, 23.1) * gA / (AC.b / 2), gT = perlin(t * 0.7, 41.9) * gA * 0.2;
  const ae = aeroElements(EL, AC, vb, s.w, rho, {
    flapF, elev: elevEff, ail: s.ail, rud: rudEff, dq: jet ? 0 : 240 * s.thr * Math.sqrt(AC.mass / 1000),
    hgt: pc => hCG + pc.r[0] * upB[0] + pc.r[1] * upB[1] + pc.r[2] * upB[2],
    gust: pc => { const g = pc.kind === 0 ? gW * pc.r[0] : pc.kind === 1 ? gT : 0; return [upB[0] * g, upB[1] * g, upB[2] * g]; },
  });
  const stallW = ae.stallW;
  const hx = Math.max(agl + 1.3, 0.3) / AC.b, ge = (16 * hx * hx) / (1 + 16 * hx * hx);
  // drag of everything that doesn't lift (fuselage, gear, flaps' profile drag, sideslip, compressibility)
  const CDp = AC.CD0 + (0.02 * flapF + 0.05 * flapF * flapF) * (AC.flapCD ?? 1) + AC.CDgear * s.gearPos + 0.6 * beta * beta
    + (AC.wave ? 0.035 : 0.08) * smoothstep(AC.wave ? 0.88 : 0.78, AC.wave ? 1.12 : 0.95, mach);
  F[0] += ae.F[0] + qd * AC.S * EL.dCY * beta; F[1] += ae.F[1]; F[2] += ae.F[2];
  if (V > 0.3) { const k = qd * AC.S * CDp / V; F[0] -= vb[0] * k; F[1] -= vb[1] * k; F[2] -= vb[2] * k; }
  { const ln = Math.hypot(vb[1], vb[2]) || 1, qs = Math.max(qd * AC.S, 1e-6);
    s.liftN = (ae.F[1] * -vb[2] + ae.F[2] * vb[1]) / ln; s.dragN = -(F[0] * vb[0] + F[1] * vb[1] + (F[2] + T) * vb[2]) / Math.max(V, 1e-3);
    s.CL = ae.CLw; s.CD = s.dragN / qs; s.ge = ge; }
  const Mp = ae.M[0] + qd * AC.S * AC.c * (AC.Cm0 + EL.dCma * Math.sin(alpha) + EL.dCmF * flapF)
    + hv * AC.S * AC.c * AC.c * EL.dCmq * qq;
  let Lr = -ae.M[2] + hv * AC.S * AC.b * AC.b * (EL.dClp * p + EL.dClr * r) + Rl;
  if (stallW > 0.15) Lr += qd * AC.S * AC.b * 0.012 * stallW * perlin(t * 1.3, 12.5);          // buffet
  const Nr = -ae.M[1] + hv * AC.S * AC.b * AC.b * EL.dCnr * r + Nx + Ng;
  const Tb = [Mp, -Nr, -Lr];

  // --- ground contact (world frame) ---
  const Fw = qrot(s.q, F);
  const Tw = [0, 0, 0];
  const wW = qrot(s.q, s.w);
  const gs = Math.hypot(s.vel[0], s.vel[2]);
  const steer = clamp(s.rud + s.rIn * 0.5, -1, 1) * lerp(32, 4, clamp(gs / 25, 0, 1)) * DEG;
  const noseOff = WHEELS[0].off;
  const crashPen = Math.max(0.5, ACFT.gear.wheelR * 1.4);
  let yLoad = F[1];
  const contact = (off, k, c, kind) => {
    const rw = qrot(s.q, off), cp = vadd(s.pos, rw);
    const g = groundAt(cp[0], cp[2]);
    const pen = g.h - cp[1];
    if (pen <= 0) return false;
    if (g.water) { crash(kind === 'hard' ? '海に墜落しました' : '海に着水してしまいました'); return true; }
    const vc = vadd(s.vel, vcross(wW, rw));
    const vn = vdot(vc, g.n);
    if (kind === 'hard') { crash(off.name + 'が地面に接触しました'); return true; }
    if (kind === 'wheel' && (pen > crashPen || vn < -6)) { crash('接地の衝撃で脚が壊れました（降下率 ' + Math.round(-vn * 196.85) + ' fpm）'); return true; }
    if (kind === 'skid' && vn < -4) { crash('尾部を強く打ちつけました'); return true; }
    const Fn = Math.max(0, k * pen - c * vn);
    let fb = [0, 0, -1];
    if (off === noseOff) fb = [Math.sin(steer), 0, -Math.cos(steer)];
    let fwd = qrot(s.q, fb);
    fwd = vnorm(vsub(fwd, vscale(g.n, vdot(fwd, g.n))));
    const lat = vnorm(vcross(g.n, fwd));
    const vl = vdot(vc, fwd), vt = vdot(vc, lat);
    const mu = kind === 'skid' ? 0.5 : g.paved ? 0.85 : 0.6;
    const rr = kind === 'skid' ? 0.5 : (g.paved ? 0.02 : 0.05) + (s.brake && off !== noseOff ? (g.paved ? 0.42 : 0.3) : 0);
    const Ft = clamp(-vt * 9 * AC.mass, -mu * Fn, mu * Fn);
    const Fl = clamp(-vl * 6 * AC.mass, -rr * Fn, rr * Fn);
    const Fc = vadd(vadd(vscale(g.n, Fn), vscale(fwd, Fl)), vscale(lat, Ft));
    Fw[0] += Fc[0]; Fw[1] += Fc[1]; Fw[2] += Fc[2];
    const tq = vcross(rw, Fc); Tw[0] += tq[0]; Tw[1] += tq[1]; Tw[2] += tq[2];
    yLoad += vdot(qrot(qi, Fc), [0, 1, 0]);
    return true;
  };
  const hard = HARDPTS.map(h => Object.assign(h.off.slice(), { name: h.name }));
  const reach = Math.max(8, AC.b * 0.6);
  if (agl < reach) {
    for (let i = 0; i < WHEELS.length; i++) s.contacts[i] = s.gearPos > 0.98 ? contact(WHEELS[i].off, WHEELS[i].k, WHEELS[i].c, 'wheel') : false;
    contact(TAILSKID.off, TAILSKID.k, TAILSKID.c, 'skid');
    for (const h of hard) { if (s.crashed) break; contact(h, 0, 0, 'hard'); }
  } else {
    if (agl < reach + 30) for (const h of hard) { if (s.crashed) break; contact(h, 0, 0, 'hard'); }
    s.contacts.fill(false);
  }
  if (s.crashed) return;

  // --- integrate ---
  Fw[1] -= AC.mass * 9.81;
  const saved = { pos: s.pos, vel: s.vel, q: s.q, w: s.w.slice() };
  s.vel = vmadd(s.vel, Fw, dt / AC.mass);
  s.pos = vmadd(s.pos, s.vel, dt);
  const Tt = vadd(Tb, qrot(qi, Tw));
  const Iw = [AC.I[0] * s.w[0], AC.I[1] * s.w[1], AC.I[2] * s.w[2]];
  const gyro = vcross(s.w, Iw);
  for (let k = 0; k < 3; k++) s.w[k] += (Tt[k] - gyro[k] - AC.I[k] * s.w[k] * 0.05) / AC.I[k] * dt;
  const dq = qmul(s.q, [s.w[0], s.w[1], s.w[2], 0]);
  s.q = qnorm([s.q[0] + 0.5 * dt * dq[0], s.q[1] + 0.5 * dt * dq[1], s.q[2] + 0.5 * dt * dq[2], s.q[3] + 0.5 * dt * dq[3]]);

  if (![...s.pos, ...s.vel, ...s.q, ...s.w].every(Number.isFinite)) {
    Object.assign(s, saved);
    crash('機体が制御不能になりました');
    return;
  }
  // --- derived state ---
  const wasMain = s.mainContact;
  s.mainContact = s.contacts.some((c, i) => c && i > 0);
  s.onGround = s.contacts.some(c => c);
  if (s.mainContact && !wasMain && s.airTime > 1.5) onTouchdown();
  s.airTime = s.onGround ? 0 : s.airTime + dt;
  s.alpha = alpha; s.beta = beta; s.tas = V; s.ias = V * Math.sqrt(atm.sigma); s.mach = mach; s.rudEff = rudEff;
  s.stallW = stallW;
  s.stallWarn = !s.onGround && V > 8 && alpha > aS - 0.045;
  s.gLoad = lerp(s.gLoad, yLoad / (AC.mass * 9.81), 0.1);
  s.agl = agl; s.groundH = ground.h; s.vs = s.vel[1];
  if (jet) s.rpm = SYS.run ? s.n1 * 100 : 0;
  else {
    const rpmT = SYS.run || s.dead ? rpmFor(s.thr, u, Math.max(0, 1.132 * atm.sigma - 0.132) * SYS.pf, s.dead) * (s.dead ? 1 : SYS.rpmK || 1) : Math.max(sysCrankRpm(), rpmFor(0, u, 1, true));
    s.rpm += (rpmT - s.rpm) * Math.min(1, dt * 1.8);
  }
  s.propAng = (s.propAng + (jet ? 0 : s.rpm) / 60 * Math.PI * 2 * dt) % (Math.PI * 2);
}

// ---------- drawing ----------
function planeMatrix(pos, q) { return M4.fromQP(q, pos); }
function setLivery(A, paint) {
  const u = progMesh.u, cp = !paint && A === ACFT && typeof acPaint === 'function' ? acPaint(A) : null, L = cp ? { ...A.livery, base: cp[0], a: cp[1], b: cp[2] } : A.livery, Wn = A.windows || { mode: 0 };
  gl.uniform1i(u.uLivery, 1);
  gl.uniform3fv(u.uLivBase, L.base);
  gl.uniform3fv(u.uLivA, paint || L.a);
  gl.uniform3fv(u.uLivB, paint ? vscale(paint, 0.35) : L.b);
  gl.uniform4f(u.uLivP, L.fw, L.y0, L.slope, L.tip);
  gl.uniform4fv(u.uLivQ, L.bands);
  gl.uniform4fv(u.uLivF, L.fin);
  gl.uniform4f(u.uLivN, L.nose, L.glare, A.hstab.span * 0.8, A.hstab.rootLE);
  gl.uniform4f(u.uWin0, Wn.mode || 0, Wn.z0 || 0, Wn.z1 || 0, Wn.pitch || 1);
  gl.uniform4f(u.uWin1, Wn.y0 || 0, Wn.y1 || 0, Wn.cz0 || 0, Wn.cz1 || 0);
  gl.uniform1f(u.uWinCy, Wn.cy || 0);
}
function drawPlane(base, st, opt = {}) {
  const A = opt.acft || ACFT, u = progMesh.u;
  setLivery(A, opt.paint);
  gl.uniform3f(u.uEmissive, 0, 0, 0);
  gl.uniform1f(u.uAlpha, 1);
  const fastProp = st.rpm > 900 && A.phys.engine.type === 'prop';
  for (const part of A.parts) {
    if (part.disc || part.flame) continue;
    if (opt.cockpit && part.cockpitHide) continue;
    if (part.propBlade && fastProp) continue;
    if (part.gear && st.gearPos < 0.02) continue;
    let m = base;
    if (part.pivot) m = M4.mul(base, M4.pivotRot(part.pivot, part.axis, part.angle(st)));
    gl.uniformMatrix4fv(u.uModel, false, m);
    gl.uniform1f(u.uSpec, part.spec);
    gl.bindVertexArray(part.vao.vao);
    gl.drawArrays(gl.TRIANGLES, 0, part.vao.count);
  }
}
function drawPropDisc(base, st) {
  const u = progMesh.u, A = ACFT;
  gl.uniform1i(u.uLivery, 0);
  gl.uniform1f(u.uSpec, 0);
  gl.uniformMatrix4fv(u.uModel, false, base);
  if (A.phys.engine.type === 'prop' && st.rpm > 900) {
    gl.uniform3f(u.uEmissive, 0, 0, 0);
    gl.uniform1f(u.uAlpha, clamp((st.rpm - 900) / 2000, 0.12, 0.3));
    for (const part of A.parts) if (part.disc) { gl.bindVertexArray(part.vao.vao); gl.drawArrays(gl.TRIANGLES, 0, part.vao.count); }
  }
  if (st.ab) {
    const flicker = 0.8 + 0.2 * Math.sin(R.time * 40);
    gl.uniform3f(u.uEmissive, 1.6 * flicker, 0.75 * flicker, 0.35);
    gl.uniform1f(u.uAlpha, 0.55);
    gl.blendFunc(gl.SRC_ALPHA, gl.ONE);
    for (const part of A.parts) if (part.flame) { gl.bindVertexArray(part.vao.vao); gl.drawArrays(gl.TRIANGLES, 0, part.vao.count); }
    gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);
  }
}

const R_EARTH = 6371000, KZ = 110574;
const GSI = 'https://cyberjapandata.gsi.go.jp/xyz/';

const kxAt = lat => 111320 * Math.cos(lat * DEG);
const wrapLon = d => { d = (d + 180) % 360; return (d < 0 ? d + 360 : d) - 180; };   // longitude (difference) into [-180, 180)
function geoToWorld(lat, lon) { return [wrapLon(lon - JW.lon0) * kxAt(lat), -(lat - JW.lat0) * KZ]; }
function worldToGeo(x, z) { const lat = JW.lat0 - z / KZ; return [lat, wrapLon(JW.lon0 + x / Math.max(kxAt(lat), 50))]; }
function geoDist(a0, o0, a1, o1) {
  const p = (a1 - a0) * DEG, l = (o1 - o0) * DEG;
  const h = Math.sin(p / 2) ** 2 + Math.cos(a0 * DEG) * Math.cos(a1 * DEG) * Math.sin(l / 2) ** 2;
  return 2 * R_EARTH * Math.asin(Math.min(1, Math.sqrt(h)));
}
function geoBearing(a0, o0, a1, o1) {
  const y = Math.sin((o1 - o0) * DEG) * Math.cos(a1 * DEG);
  const x = Math.cos(a0 * DEG) * Math.sin(a1 * DEG) - Math.sin(a0 * DEG) * Math.cos(a1 * DEG) * Math.cos((o1 - o0) * DEG);
  return (Math.atan2(y, x) / DEG + 360) % 360;
}
const mercY = lat => (1 - Math.asinh(Math.tan(lat * DEG)) / Math.PI) / 2;
const latFromMerc = my => Math.atan(Math.sinh(Math.PI * (1 - 2 * my))) / DEG;

async function inflateB64(b64) {
  const bin = Uint8Array.from(atob(b64), c => c.charCodeAt(0));
  const stream = new Blob([bin]).stream().pipeThrough(new DecompressionStream('deflate'));
  return new Int16Array(await new Response(stream).arrayBuffer());
}
function undelta(d, W, H) {
  for (let j = 0; j < H; j++) { let acc = 0; for (let i = 0; i < W; i++) { const k = j * W + i; acc = ((acc + d[k]) << 16) >> 16; d[k] = acc; } }
  return d;
}
async function jwInitData() {
  if (JW.coarse) return;
  JW.coarse = undelta(await inflateB64(JP_COARSE.data), JP_COARSE.w, JP_COARSE.h);
  JW.world = undelta(await inflateB64(WORLD_COARSE.data), WORLD_COARSE.w, WORLD_COARSE.h);
  // 16-pixel blocks of the Japan mosaic that contain land: near them the coarse world data is not used
  { const C = JP_COARSE, B = 16, bw = Math.ceil(C.w / B), bh = Math.ceil(C.h / B), m = new Uint8Array(bw * bh);
    for (let j = 0; j < C.h; j++) for (let i = 0; i < C.w; i++) if (JW.coarse[j * C.w + i] !== -32768) m[(j / B | 0) * bw + (i / B | 0)] = 1;
    const g = new Uint8Array(bw * bh);
    for (let j = 0; j < bh; j++) for (let i = 0; i < bw; i++) {
      let v = 0; for (let b = -1; b <= 1; b++) for (let a = -1; a <= 1; a++) { const jj = j + b, ii = i + a; if (jj >= 0 && ii >= 0 && jj < bh && ii < bw && m[jj * bw + ii]) v = 1; }
      g[j * bw + i] = v;
    }
    JW.jpNear = { B, bw, bh, g }; }
  for (const a of JP_AIRPORTS) { a.cc = 'JP'; a.big = a.rw[0].len >= 2400 ? 1 : 0; }
  JW.airports = JP_AIRPORTS.concat(WORLD_AIRPORTS);
  JW.places = JP_PLACES.concat(WORLD_PLACES);
  for (const [id, p] of Object.entries(JP_PATCHES)) {
    const N = 1024, raw = undelta(await inflateB64(p.data), N, N), h = new Float32Array(N * N);
    for (let k = 0; k < N * N; k++) h[k] = raw[k] === -32768 ? NaN : raw[k] / 10;
    JW.patches.push({ id, lat: p.lat, lon: p.lon, cell: p.cell, half: N * p.cell / 2, kx: kxAt(p.lat), h });
  }
  for (const ap of JW.airports) for (const r of ap.rw) {
    const lat = (r.le[0] + r.he[0]) / 2, lon = (r.le[1] + r.he[1]) / 2, kx = kxAt(lat);
    const dx = (r.he[1] - r.le[1]) * kx, dz = -(r.he[0] - r.le[0]) * KZ;
    const hdg = (Math.atan2(dx, -dz) / DEG + 360) % 360;
    JW.rws.push(Object.assign(r, { ap, lat, lon, kx, hdg, sin: Math.sin(hdg * DEG), cos: Math.cos(hdg * DEG) }));
  }
  // spatial index: 0.5-degree cells -> runways within reach (flattening extends ~ len/2 + 300 m)
  JW.rwGrid = new Map();
  for (const r of JW.rws) {
    const ext = (r.len / 2 + 400) / KZ, extLon = (r.len / 2 + 400) / Math.max(r.kx, 1000);
    for (let a = Math.floor((r.lat - ext) * 2); a <= Math.floor((r.lat + ext) * 2); a++)
      for (let b = Math.floor((r.lon - extLon) * 2); b <= Math.floor((r.lon + extLon) * 2); b++) {
        const k = a * 1000 + b; let l = JW.rwGrid.get(k); if (!l) JW.rwGrid.set(k, l = []); l.push(r);
      }
  }
}
const NO_RW = [];
const rwNear = (lat, lon) => JW.rwGrid ? JW.rwGrid.get(Math.floor(lat * 2) * 1000 + Math.floor(lon * 2)) || NO_RW : JW.rws;
function worldCoarseH(lat, lon) {
  const C = WORLD_COARSE, n = 256 * 2 ** C.z, d = JW.world;
  if (!d || lat > 85 || lat < -85) return -30;
  const fx = ((wrapLon(lon) + 180) / 360 * n - 0.5 + n) % n, fy = clamp(mercY(lat) * n - 0.5, 0, C.h - 1.001);
  const ix = Math.floor(fx), iy = Math.floor(fy), tx = fx - ix, ty = fy - iy, ix1 = (ix + 1) % C.w;
  const k0 = iy * C.w, k1 = Math.min(iy + 1, C.h - 1) * C.w;
  return lerp(lerp(d[k0 + ix], d[k0 + ix1], tx), lerp(d[k1 + ix], d[k1 + ix1], tx), ty);
}
function jpNearLand(lat, lon) {
  const C = JP_COARSE, n = 256 * 2 ** C.z, N = JW.jpNear;
  if (!N) return false;
  const fx = (lon + 180) / 360 * n - C.x0 * 256, fy = mercY(lat) * n - C.y0 * 256;
  if (fx < 0 || fy < 0 || fx >= C.w || fy >= C.h) return false;
  return !!N.g[(fy / N.B | 0) * N.bw + (fx / N.B | 0)];
}
function coarseH(lat, lon) {
  const h = jpCoarseH(lat, lon);
  if (!Number.isNaN(h)) return h;
  if (jpNearLand(lat, lon)) return NaN;
  const w = worldCoarseH(lat, lon);
  return w > 0.5 ? w : NaN;
}
function jpCoarseH(lat, lon) {
  const C = JP_COARSE, n = 256 * 2 ** C.z;
  const fx = (lon + 180) / 360 * n - C.x0 * 256 - 0.5, fy = mercY(lat) * n - C.y0 * 256 - 0.5;
  if (fx < 0 || fy < 0 || fx >= C.w - 1 || fy >= C.h - 1) return NaN;
  const ix = Math.floor(fx), iy = Math.floor(fy), tx = fx - ix, ty = fy - iy, k = iy * C.w + ix, d = JW.coarse;
  const v = [d[k], d[k + 1], d[k + C.w], d[k + C.w + 1]];
  if (v.every(q => q === -32768)) return NaN;
  const f = q => q === -32768 ? -8 : q;
  return lerp(lerp(f(v[0]), f(v[1]), tx), lerp(f(v[2]), f(v[3]), tx), ty);
}
function patchH(lat, lon) {
  for (const p of JW.patches) {
    const x = (lon - p.lon) * p.kx, z = -(lat - p.lat) * KZ;
    const fx = (x + p.half) / p.cell, fz = (z + p.half) / p.cell;
    if (fx < 6 || fz < 6 || fx > 1017 || fz > 1017) continue;
    const ix = Math.floor(fx), iz = Math.floor(fz), tx = fx - ix, tz = fz - iz, k = iz * 1024 + ix;
    const g = q => Number.isNaN(q) ? -6 : q;
    const h = lerp(lerp(g(p.h[k]), g(p.h[k + 1]), tx), lerp(g(p.h[k + 1024]), g(p.h[k + 1025]), tx), tz);
    const w = smoothstep(6, 60, Math.min(fx, fz, 1023 - fx, 1023 - fz));
    return [h, w];
  }
  return null;
}
// height from the best available source (metres, NaN = open sea), then runway flattening
function sampleDem(t, lat, lon) {
  const n = 2 ** t.z, px = ((lon + 180) / 360 * n - t.x) * 256 - 0.5, py = (mercY(lat) * n - t.y) * 256 - 0.5;
  const ix = clamp(Math.floor(px), 0, 254), iy = clamp(Math.floor(py), 0, 254), tx = clamp(px - ix, 0, 1), ty = clamp(py - iy, 0, 1);
  const d = t.dem, k = iy * 256 + ix, v = [d[k], d[k + 1], d[k + 256], d[k + 257]];
  const valid = v.filter(q => !Number.isNaN(q));
  if (valid.length < 4) { if (valid.length < 2) return NaN; const m = valid.reduce((a, b) => a + b) / valid.length; for (let i = 0; i < 4; i++) if (Number.isNaN(v[i])) v[i] = m; }
  return lerp(lerp(v[0], v[1], tx), lerp(v[2], v[3], tx), ty);
}
function rawHeightGeo(lat, lon) {
  if (JW.online) {
    for (let z = 14; z >= 6; z--) {
      const n = 2 ** z, t = JW.tiles.get(z + '/' + Math.floor((lon + 180) / 360 * n) + '/' + Math.floor(mercY(lat) * n));
      if (t && t.dem) { const h = sampleDem(t, lat, lon); return Number.isNaN(h) ? -12 : h; }
    }
  }
  let h = coarseH(lat, lon);
  if (Number.isNaN(h)) h = -30;
  const p = patchH(lat, lon);
  if (p) h = lerp(h, p[0], p[1]);
  return h;
}
function runwayFlatten(lat, lon, h) {
  for (const r of rwNear(lat, lon)) {
    if (Math.abs(lat - r.lat) > 0.04 || Math.abs(lon - r.lon) > 0.05) continue;
    const dx = (lon - r.lon) * r.kx, dz = -(lat - r.lat) * KZ;
    const u = dx * r.sin - dz * r.cos, v = dx * r.cos + dz * r.sin;
    const e = Math.hypot(Math.max(Math.abs(u) - r.len / 2 - 90, 0), Math.max(Math.abs(v) - r.wid / 2 - 60, 0));
    if (e < 220) h = lerp(r.elev, h, smoothstep(0, 220, e));
  }
  return h;
}
function jwHeightGeo(lat, lon) { return runwayFlatten(lat, lon, rawHeightGeo(lat, lon)); }
function jwHeight(x, z) { const [lat, lon] = worldToGeo(x, z); return jwHeightGeo(lat, lon); }
function jwNormal(x, z) {
  const e = 15;
  return vnorm([jwHeight(x - e, z) - jwHeight(x + e, z), 2 * e, jwHeight(x, z - e) - jwHeight(x, z + e)]);
}
function jwRunwayAt(x, z) {
  const [lat, lon] = worldToGeo(x, z);
  for (const r of rwNear(lat, lon)) {
    if (Math.abs(lat - r.lat) > 0.03 || Math.abs(lon - r.lon) > 0.04) continue;
    const dx = (lon - r.lon) * r.kx, dz = -(lat - r.lat) * KZ;
    const u = dx * r.sin - dz * r.cos, v = dx * r.cos + dz * r.sin;
    if (Math.abs(u) <= r.len / 2 && Math.abs(v) <= r.wid / 2) return r;
  }
  return null;
}
function jwOnPaved(x, z) { return jwRunwayAt(x, z) ? 'rwy' : null; }
function jwGroundAt(x, z) {
  const r = jwRunwayAt(x, z);
  if (r) return { h: r.elev + 0.15, n: [0, 1, 0], paved: true, water: false };
  const h = jwHeight(x, z);
  if (!(h > -0.5)) return { h: 0, n: [0, 1, 0], paved: false, water: true };
  return { h: Math.max(h, 0), n: jwNormal(x, z), paved: false, water: false };
}

// ---------- airports and the active runway ----------
function nearestAirport(lat, lon, maxM = 1e9) {
  let best = null, bd = maxM;
  for (const ap of JW.airports) { const d = geoDist(lat, lon, ap.lat, ap.lon); if (d < bd) { bd = d; best = ap; } }
  return best;
}
// make runway `r` (in direction `rev`) the active runway used for spawning, PAPI and the landing HUD
function jwSetActive(r, rev) {
  const [cx, cz] = geoToWorld(r.lat, r.lon);
  setRunway({ cx, cz, hdg: rev ? (r.hdg + 180) % 360 : r.hdg, len: r.len, wid: r.wid, elev: r.elev,
    idents: rev ? [r.ids[1], r.ids[0]] : [r.ids[0], r.ids[1]], side: 1, rec: r, rev, ap: r.ap });
}
function jwChooseRunway(ap) {
  // longest runway; the end whose 3° approach is clearest of terrain
  const r = ap.rw[0];
  const clear = rev => {
    const hdg = (rev ? r.hdg + 180 : r.hdg) * DEG, kx = r.kx;
    let mn = 1e9;
    for (let d = 300; d <= 7000; d += 150) {
      const back = r.len / 2 + d, dx = -Math.sin(hdg) * back, dz = Math.cos(hdg) * back;
      const lat = r.lat - dz / KZ, lon = r.lon + dx / kx;
      mn = Math.min(mn, r.elev + Math.tan(3 * DEG) * d - Math.max(rawHeightGeo(lat, lon), 0));
    }
    return mn;
  };
  const a = clear(false), b = clear(true);
  jwSetActive(r, b > a + 30 && a < 80);
}
function jwUpdateActive() {
  const [lat, lon] = planeGeo();
  const ap = nearestAirport(lat, lon, 40000);
  if (!ap) return;
  const hdg = attitude().hdg;
  let best = null, bs = 1e9;
  for (const r of ap.rw) for (const rev of [false, true]) {
    const dh = (rev ? r.hdg + 180 : r.hdg) % 360;
    const onIt = jwRunwayAt(sim.pos[0], sim.pos[2]) === r;
    const thrLat = r.lat + (rev ? 1 : -1) * r.cos * r.len / 2 / KZ, thrLon = r.lon - (rev ? 1 : -1) * r.sin * r.len / 2 / r.kx;
    const toThr = geoBearing(lat, lon, thrLat, thrLon);
    const angd = (a, b) => Math.abs(((a - b + 540) % 360) - 180);
    const sc = angd(hdg, dh) + (onIt || sim.onGround ? 0 : angd(toThr, dh) * 0.7) - (onIt ? 100 : 0);
    if (sc < bs) { bs = sc; best = [r, rev]; }
  }
  if (best && (RWY.rec !== best[0] || RWY.rev !== best[1])) jwSetActive(best[0], best[1]);
}

// ---------- floating origin ----------
function jwRecenter(force) {
  if (!force && Math.hypot(sim.pos[0], sim.pos[2]) < 60000) return;
  const conv = p => { const [lat, lon] = worldToGeo(p[0], p[2]); return lat + ',' + lon; };
  const oldGeo = p => worldToGeo(p[0], p[2]);
  const moved = [sim.pos, ...(typeof CAR !== 'undefined' && CAR.on && CAR.plane ? [CAR.plane.pos] : []), ...particles.map(q => q.p), ...(course.rings || []).map(r => r.p)].map(p => [p, oldGeo(p)]);
  const camGeo = oldGeo(cam.eye);
  const [lat, lon] = worldToGeo(sim.pos[0], sim.pos[2]);
  JW.lat0 = lat; JW.lon0 = lon; JW.ov++;
  for (const [p, g] of moved) { const w = geoToWorld(g[0], g[1]); p[0] = w[0]; p[2] = w[1]; }
  if (course.centre) { const g = course.centreGeo; if (g) { const w = geoToWorld(g[0], g[1]); course.centre = [w[0], 0, w[1]]; } }
  const ce = geoToWorld(camGeo[0], camGeo[1]); cam.eye = [ce[0], cam.eye[1], ce[1]]; cam.flyPos = null;
  if (RWY.rec) jwSetActive(RWY.rec, RWY.rev);
  JW.cloudKey = null;
  void conv;
}

// ---------- terrain tiles ----------
const TILE_G = 32, TILE_NV = (TILE_G + 1) ** 2 + 4 * (TILE_G + 1);
let tileIndex = null, tileUv = null;
function tileTopology() {
  const G = TILE_G, idx = [];
  for (let j = 0; j < G; j++) for (let i = 0; i < G; i++) {
    const a = j * (G + 1) + i, b = a + 1, c = a + G + 1, d = c + 1;
    idx.push(a, c, b, b, c, d);
  }
  const base = (G + 1) ** 2, edges = [];
  for (let i = 0; i <= G; i++) edges.push(i);                          // north
  for (let i = 0; i <= G; i++) edges.push(G * (G + 1) + i);            // south
  for (let j = 0; j <= G; j++) edges.push(j * (G + 1));                // west
  for (let j = 0; j <= G; j++) edges.push(j * (G + 1) + G);            // east
  for (let s = 0; s < 4; s++) for (let k = 0; k < G; k++) {
    const e0 = edges[s * (G + 1) + k], e1 = edges[s * (G + 1) + k + 1], s0 = base + s * (G + 1) + k, s1 = s0 + 1;
    idx.push(e0, s0, e1, e1, s0, s1);
  }
  tileIndex = new Uint16Array(idx);
  tileUv = new Float32Array(TILE_NV * 2);
  for (let j = 0; j <= G; j++) for (let i = 0; i <= G; i++) { const k = j * (G + 1) + i; tileUv[k * 2] = i / G; tileUv[k * 2 + 1] = j / G; }
  for (let k = 0; k < edges.length; k++) { tileUv[(base + k) * 2] = tileUv[edges[k] * 2]; tileUv[(base + k) * 2 + 1] = tileUv[edges[k] * 2 + 1]; }
  JW.edges = edges;
  JW.uvBuf = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, JW.uvBuf); gl.bufferData(gl.ARRAY_BUFFER, tileUv, gl.STATIC_DRAW);
  JW.idxBuf = gl.createBuffer(); gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, JW.idxBuf); gl.bufferData(gl.ELEMENT_ARRAY_BUFFER, tileIndex, gl.STATIC_DRAW);
}
function getTile(z, x, y) {
  const key = z + '/' + x + '/' + y;
  let t = JW.tiles.get(key);
  if (t) return t;
  const n = 2 ** z;
  t = { z, x, y, key, lonW: x / n * 360 - 180, lonE: (x + 1) / n * 360 - 180, latN: latFromMerc(y / n), latS: latFromMerc((y + 1) / n),
    demState: 0, photoState: 0, mesh: null, tex: null, used: 0 };
  t.size = (t.lonE - t.lonW) * kxAt((t.latN + t.latS) / 2);
  // rough height range from the coarse data until the mesh exists
  let mn = 1e9, mx = -1e9;
  for (let a = 0; a <= 4; a++) for (let b = 0; b <= 4; b++) {
    const h = coarseH(lerp(t.latN, t.latS, a / 4), lerp(t.lonW, t.lonE, b / 4));
    const hh = Number.isNaN(h) ? -30 : h; mn = Math.min(mn, hh); mx = Math.max(mx, hh);
  }
  const pad = z < 8 ? 2500 : z < 11 ? 900 : 250;
  t.hmin = mn - pad; t.hmax = mx + pad;
  t.seaGuess = mx < -1 && z >= 6;
  JW.tiles.set(key, t);
  return t;
}
// does this tile show Japanese land? (then GSI data is used, otherwise world-wide sources)
function tileIsJapan(t) {
  if (t.jp !== undefined) return t.jp;
  t.jp = false;
  if (t.latS > 46.2 || t.latN < 23.8 || t.lonE < 122.3 || t.lonW > 146.7 || t.z < 5) return t.jp;
  for (let a = 0; a <= 6 && !t.jp; a++) for (let b = 0; b <= 6; b++) if (!Number.isNaN(jpCoarseH(lerp(t.latN, t.latS, a / 6), lerp(t.lonW, t.lonE, b / 6)))) { t.jp = true; break; }
  return t.jp;
}
function buildTileMesh(t) {
  const G = TILE_G, n = 2 ** t.z, nv = TILE_NV;
  const lat = t.vLat || (t.vLat = new Float64Array(nv)), lon = t.vLon || (t.vLon = new Float64Array(nv));
  const hv = t.vH || (t.vH = new Float32Array(nv));
  const own = t.dem && t.z <= 14;
  let mn = 1e9, mx = -1e9;
  for (let j = 0; j <= G; j++) for (let i = 0; i <= G; i++) {
    const k = j * (G + 1) + i;
    const la = latFromMerc((t.y + j / G) / n), lo = (t.x + i / G) / n * 360 - 180;
    lat[k] = la; lon[k] = lo;
    let h = own ? sampleDem(t, la, lo) : rawHeightGeo(la, lo);
    if (Number.isNaN(h)) h = -12;
    h = runwayFlatten(la, lo, h);
    if (h < 0.3) h = Math.min(h, -1.5) - 1.5;                           // sea: sink below the water surface
    hv[k] = h; mn = Math.min(mn, h); mx = Math.max(mx, h);
  }
  const skirt = Math.max(15, t.size * 0.03), base = (G + 1) ** 2;
  for (let k = 0; k < JW.edges.length; k++) { const e = JW.edges[k]; lat[base + k] = lat[e]; lon[base + k] = lon[e]; hv[base + k] = hv[e] - skirt; }
  t.hmin = mn - skirt; t.hmax = mx;
  t.sea = mx < -1;
  t.demUsed = !!(t.dem || (t.z > 14 && demAncestor(t) && demAncestor(t).dem));
  projectTileMesh(t);
}
function demAncestor(t) {
  if (t.z <= 14) return t;
  const s = t.z - 14; return JW.tiles.get('14/' + (t.x >> s) + '/' + (t.y >> s)) || getTile(14, t.x >> s, t.y >> s);
}
function projectTileMesh(t) {
  const G = TILE_G, nv = TILE_NV, pos = t.pos || (t.pos = new Float32Array(nv * 3)), nrm = t.nrm || (t.nrm = new Float32Array(nv * 3));
  const [cx, cz] = geoToWorld((t.latN + t.latS) / 2, (t.lonW + t.lonE) / 2);
  t.cx = cx; t.cz = cz;
  let x0 = 1e18, x1 = -1e18, z0 = 1e18, z1 = -1e18;
  for (let k = 0; k < nv; k++) {
    const la = t.vLat[k];
    const x = wrapLon(t.vLon[k] - JW.lon0) * kxAt(la) - cx, z = -(la - JW.lat0) * KZ - cz;
    pos[k * 3] = x; pos[k * 3 + 1] = t.vH[k]; pos[k * 3 + 2] = z;
    if (x < x0) x0 = x; if (x > x1) x1 = x; if (z < z0) z0 = z; if (z > z1) z1 = z;
  }
  t.box = [cx + x0, cz + z0, cx + x1, cz + z1];
  for (let j = 0; j <= G; j++) for (let i = 0; i <= G; i++) {
    const k = j * (G + 1) + i, kl = j * (G + 1) + Math.max(i - 1, 0), kr = j * (G + 1) + Math.min(i + 1, G);
    const ku = Math.max(j - 1, 0) * (G + 1) + i, kd = Math.min(j + 1, G) * (G + 1) + i;
    const tx = [pos[kr * 3] - pos[kl * 3], pos[kr * 3 + 1] - pos[kl * 3 + 1], pos[kr * 3 + 2] - pos[kl * 3 + 2]];
    const tz = [pos[kd * 3] - pos[ku * 3], pos[kd * 3 + 1] - pos[ku * 3 + 1], pos[kd * 3 + 2] - pos[ku * 3 + 2]];
    const nn = vnorm(vcross(tz, tx));
    nrm[k * 3] = nn[0]; nrm[k * 3 + 1] = nn[1]; nrm[k * 3 + 2] = nn[2];
  }
  const base = (G + 1) ** 2;
  for (let k = 0; k < JW.edges.length; k++) { const e = JW.edges[k]; for (let c = 0; c < 3; c++) nrm[(base + k) * 3 + c] = nrm[e * 3 + c]; }
  if (!t.mesh) {
    const vao = gl.createVertexArray(); gl.bindVertexArray(vao);
    const pb = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, pb); gl.bufferData(gl.ARRAY_BUFFER, pos, gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(0); gl.vertexAttribPointer(0, 3, gl.FLOAT, false, 0, 0);
    const nb = gl.createBuffer(); gl.bindBuffer(gl.ARRAY_BUFFER, nb); gl.bufferData(gl.ARRAY_BUFFER, nrm, gl.DYNAMIC_DRAW);
    gl.enableVertexAttribArray(1); gl.vertexAttribPointer(1, 3, gl.FLOAT, false, 0, 0);
    gl.bindBuffer(gl.ARRAY_BUFFER, JW.uvBuf); gl.enableVertexAttribArray(2); gl.vertexAttribPointer(2, 2, gl.FLOAT, false, 0, 0);
    gl.bindBuffer(gl.ELEMENT_ARRAY_BUFFER, JW.idxBuf);
    gl.bindVertexArray(null);
    t.mesh = { vao, pb, nb };
  } else {
    gl.bindBuffer(gl.ARRAY_BUFFER, t.mesh.pb); gl.bufferSubData(gl.ARRAY_BUFFER, 0, pos);
    gl.bindBuffer(gl.ARRAY_BUFFER, t.mesh.nb); gl.bufferSubData(gl.ARRAY_BUFFER, 0, nrm);
  }
  t.ov = JW.ov;
}
function disposeTile(t) {
  if (t.mesh) { gl.deleteVertexArray(t.mesh.vao); gl.deleteBuffer(t.mesh.pb); gl.deleteBuffer(t.mesh.nb); t.mesh = null; }
  if (t.tex) { gl.deleteTexture(t.tex); t.tex = null; JW.texCount--; if (t.photoState === 2) t.photoState = 0; }
  t.pos = t.nrm = t.vLat = t.vLon = t.vH = null;
}

// ---------- streaming ----------
let demCanvas = null;
async function fetchBitmap(url) {
  const ctl = new AbortController(), timer = setTimeout(() => ctl.abort(), 15000);
  try {
    const r = await fetch(url, { signal: ctl.signal, mode: 'cors' });
    if (r.status === 404) return 404;
    if (!r.ok) throw new Error('HTTP ' + r.status);
    return await createImageBitmap(await r.blob(), { premultiplyAlpha: 'none', colorSpaceConversion: 'none' });
  } finally { clearTimeout(timer); }
}
async function loadDem(t) {
  t.demState = 1; JW.inflight++;
  try {
    const jp = tileIsJapan(t);
    const bmp = await fetchBitmap(jp ? `${GSI}dem_png/${t.z}/${t.x}/${t.y}.png` : `${TERRARIUM}${Math.min(t.z, 15)}/${t.x}/${t.y}.png`);
    if (bmp === 404) { t.demState = 3; return; }
    if (!demCanvas) { demCanvas = document.createElement('canvas'); demCanvas.width = demCanvas.height = 256; }
    const g = demCanvas.getContext('2d', { willReadFrequently: true });
    g.clearRect(0, 0, 256, 256); g.drawImage(bmp, 0, 0); bmp.close && bmp.close();
    const px = g.getImageData(0, 0, 256, 256).data, dem = new Float32Array(65536);
    if (jp) for (let k = 0; k < 65536; k++) {
      const v = (px[k * 4] << 16) | (px[k * 4 + 1] << 8) | px[k * 4 + 2];
      dem[k] = v === 0x800000 ? NaN : (v >= 0x800000 ? v - 0x1000000 : v) * 0.01;
    }
    else for (let k = 0; k < 65536; k++) {                            // terrarium: (R*256 + G + B/256) - 32768
      const h = px[k * 4] * 256 + px[k * 4 + 1] + px[k * 4 + 2] / 256 - 32768;
      dem[k] = h;
    }
    t.dem = dem; t.demState = 2; JW.loadedDems++;
    JW.dirty.push(t);
    if (t.z === 14) for (const d of JW.tiles.values()) if (d.z > 14 && d.mesh && (d.x >> (d.z - 14)) === t.x && (d.y >> (d.z - 14)) === t.y) JW.dirty.push(d);
  } catch (e) { t.demState = 3; if (JW.loadedDems === 0) jwNetFail(); }
  finally { JW.inflight--; }
}
async function loadPhoto(t) {
  t.photoState = 1; JW.inflight++;
  try {
    const bmp = await fetchBitmap(tileIsJapan(t) && t.z >= 2 ? `${GSI}seamlessphoto/${t.z}/${t.x}/${t.y}.jpg` : `${ESRI}World_Imagery/MapServer/tile/${t.z}/${t.y}/${t.x}?blankTile=false`);
    if (bmp === 404) { t.photoState = 3; return; }
    if (!JW.tiles.has(t.key)) { bmp.close && bmp.close(); return; }
    const tex = gl.createTexture();
    gl.bindTexture(gl.TEXTURE_2D, tex);
    gl.pixelStorei(gl.UNPACK_ALIGNMENT, 1);
    gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGB, gl.RGB, gl.UNSIGNED_BYTE, bmp);
    gl.generateMipmap(gl.TEXTURE_2D);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
    gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
    if (anisoExt) gl.texParameterf(gl.TEXTURE_2D, anisoExt.TEXTURE_MAX_ANISOTROPY_EXT, game.touch ? 4 : Math.min(16, gl.getParameter(anisoExt.MAX_TEXTURE_MAX_ANISOTROPY_EXT)));
    bmp.close && bmp.close();
    t.tex = tex; t.photoState = 2; JW.texCount++; JW.loadedPhotos++;
  } catch (e) { t.photoState = 3; if (JW.loadedPhotos === 0 && JW.loadedDems === 0) jwNetFail(); }
  finally { JW.inflight--; }
}
function jwNetFail() {
  JW.netFails = (JW.netFails || 0) + 1;
  if (JW.netFails > 6 && !JW.loadedPhotos && !JW.loadedDems) { JW.online = false; toast('詳細地図を読み込めないため、内蔵の簡易地形で表示します', 3500); }
}
// if the first check failed (slow start, flaky connection), keep trying quietly in the background
async function jwRecheckOnline() {
  if (JW.rechecking) return;
  JW.rechecking = true;
  try {
    const b = await Promise.race([fetchBitmap(`${GSI}dem_png/5/28/12.png`), new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), 12000))]);
    if (b && b !== 404 && JW.online === false) {
      JW.online = true; JW.netFails = 0;
      for (const t of JW.tiles.values()) if (t.mesh) JW.dirty.push(t);
      toast('詳細地図（航空写真・建物）に接続しました', 2500);
      if (game.state === 'menu') updateMenuTexts();
    }
  } catch (e) { /* still offline */ }
  finally { JW.rechecking = false; }
}
async function jwCheckOnline() {
  if (JW.online !== null) return JW.online;
  try { const b = await Promise.race([fetchBitmap(`${GSI}dem_png/5/28/12.png`), new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), 12000))]); JW.online = !!b && b !== 404; }
  catch (e) { JW.online = false; }
  return JW.online;
}

// ---------- tile selection ----------
function jwSelect(planes, eye) {
  const f = ++JW.frame, out = [], wantMesh = [], want = [];
  // driving: one zoom level finer (about 0.6 m per pixel) and refined sooner, since the ground is right there
  const drive = typeof CAR !== 'undefined' && CAR.on;
  const zmax = JW.online ? (game.touch ? 16 : drive ? 18 : 17) : 12, K = game.touch ? 0.9 : drive ? 1.8 : 1.25;
  const [eLat, eLon] = worldToGeo(eye[0], eye[2]);
  const reach = 700000 + Math.max(eye[1], 0) * 30;
  const geoFar = t => {
    const la = clamp(eLat, t.latS, t.latN);
    const dl = wrapLon(eLon - (t.lonW + t.lonE) / 2), half = (t.lonE - t.lonW) / 2;
    const lo = Math.abs(dl) <= half ? eLon : (t.lonW + t.lonE) / 2 + Math.sign(dl) * half;
    return geoDist(eLat, eLon, la, lo) > reach;
  };
  const visit = t => {
    t.used = f;
    if (t.z <= 7 && geoFar(t)) return;
    if (t.seaGuess && !t.mesh) return;
    if (t.mesh && t.ov !== JW.ov) projectTileMesh(t);
    if (t.mesh && t.sea && t.z >= 7) return;                       // open sea: the water surface covers it
    const b = t.box || (() => { const c1 = geoToWorld(t.latN, t.lonW), c2 = geoToWorld(t.latS, t.lonE), c3 = geoToWorld(t.latN, t.lonE), c4 = geoToWorld(t.latS, t.lonW);
      return [Math.min(c1[0], c4[0]), Math.min(c1[1], c3[1]), Math.max(c2[0], c3[0]), Math.max(c2[1], c4[1])]; })();
    const dxy = Math.hypot(Math.max(b[0] - eye[0], 0, eye[0] - b[2]), Math.max(b[1] - eye[2], 0, eye[2] - b[3]));
    const drop = dxy * dxy / (2 * R_EARTH);
    if (!boxInFrustum(planes, b[0], t.hmin - drop - 50, b[1], b[2], t.hmax + 50, b[3])) return;
    const dy = Math.max(t.hmin - drop - eye[1], 0, eye[1] - t.hmax);
    const d = Math.hypot(dxy, dy);
    if (t.z < zmax && t.size / Math.max(d, 1) > K) {
      const kids = [0, 1, 2, 3].map(i => getTile(t.z + 1, t.x * 2 + (i & 1), t.y * 2 + (i >> 1)));
      if (kids.every(k => k.mesh || k.seaGuess)) { for (const k of kids) visit(k); return; }
      for (const k of kids) if (!k.mesh && !k.seaGuess) wantMesh.push([k, d]);
    }
    if (!t.mesh) { wantMesh.push([t, d]); return; }
    out.push(t);
    if (JW.online) {
      const pr = d / t.size;
      if (t.photoState === 0) want.push(['p', t, pr]);
      if (t.z >= 6) { const a = demAncestor(t); if (a.demState === 0) want.push(['d', a, pr * 0.7]); }
    }
  };
  for (const r of JW.roots) visit(r);
  // build meshes: nearest first, limited per frame
  wantMesh.sort((a, b) => a[1] - b[1]);
  let built = 0;
  for (const [t] of wantMesh) { if (built >= 24) break; if (!t.mesh) { buildTileMesh(t); built++; } }
  while (JW.dirty.length && built < 40) { const t = JW.dirty.shift(); if (t.mesh || t.used > f - 3) { buildTileMesh(t); built++; } }
  if (JW.online) {
    want.sort((a, b) => a[2] - b[2]);
    for (const [k, t] of want) {
      if (JW.inflight >= 10) break;
      if (k === 'p' && t.photoState === 0) loadPhoto(t);
      else if (k === 'd' && t.demState === 0) loadDem(t);
    }
  }
  // evict tiles that have not been used for a while
  if (JW.tiles.size > 1500 || JW.texCount > 380) {
    const old = [...JW.tiles.values()].filter(t => t.used < f - 90 && t.z > 5).sort((a, b) => a.used - b.used);
    for (const t of old.slice(0, Math.max(JW.tiles.size - 1200, JW.texCount - 300, 60))) { disposeTile(t); JW.tiles.delete(t.key); }
  }
  JW.drawn = out.length;
  return out;
}
function photoFor(t) {
  // own photo, else the nearest ancestor's photo with a sub-rectangle
  let a = t, s = 1, ox = 0, oy = 0;
  while (a && !a.tex) {
    if (a.z <= 5) return null;
    ox = (ox + (a.x & 1)) / 2; oy = (oy + (a.y & 1)) / 2; s /= 2;
    a = JW.tiles.get((a.z - 1) + '/' + (a.x >> 1) + '/' + (a.y >> 1));
  }
  return a ? [a.tex, s, ox, oy] : null;
}

const TILE_FS = GLSL_HEAD + GLSL_COMMON + GLSL_SHADOW + `

uniform sampler2D uPhoto;
uniform sampler2D uNoise;
// close-up detail: an aerial photo is about a metre per pixel, so near the camera (driving) a fine grain is
// laid over it - tarmac gets a speckled aggregate, grass and fields a tufted, patchy texture
vec3 groundDetail(vec3 ph, vec3 w, sampler2D nz){
  float dist = length(w - uCamPos), near = 1.0 - smoothstep(25.0, 260.0, dist);
  if (near <= 0.0) return ph;

  float tar = (1.0 - green) * smoothstep(0.14, 0.05, sat);
  float f1 = texture(nz, w.xz / 0.37).b, f2 = texture(nz, w.xz / 1.3).a, f3 = texture(nz, w.xz / 5.3).b, f4 = texture(nz, w.xz / 0.11).a;

  float tuft = mix(1.0, 0.72 + 0.34 * f2 + 0.18 * f1, fine) * (0.9 + 0.2 * f3);
  vec3 g = ph * mix(tuft, grain, tar);
  // near the ground: photographed grass and asphalt grain (p_citymat.js), coloured by the aerial photo
  if (uMatOn > 0.5) {

}

  const dir = [r.sin, -r.cos], perp = [r.cos, r.sin], out = [];
  const P = (u, v, yy, c, s) => out.push(cx + dir[0] * u + perp[0] * v, yy, cz + dir[1] * u + perp[1] * v, c[0], c[1], c[2], s);
  for (let u = -L; u <= L + 1; u += 60) { const c = Math.abs(u) > L - 600 ? [1, 0.85, 0.45] : [1, 0.96, 0.85]; P(u, W, y, c, 1); P(u, -W, y, c, 1); }
  for (let v = -W; v <= W + 0.1; v += 4) { P(-L - 2, v, y, [0.3, 1, 0.45], 1.1); P(L + 2, v, y, [0.3, 1, 0.45], 1.1); }
  for (let v = -W; v <= W + 0.1; v += 6) { P(-L + 3, v, y, [1, 0.2, 0.15], 0.8); P(L - 3, v, y, [1, 0.2, 0.15], 0.8); }
  for (const sgn of [-1, 1]) for (let k = 1; k <= 30; k++) {                 // approach light bars, 900 m

  setCommon(progLight);
  gl.uniform1f(progLight.u.uPx, (R.H / 900) * night);
  gl.bindVertexArray(JW.lightVAO);
  gl.bindBuffer(gl.ARRAY_BUFFER, lightBuf); gl.bufferData(gl.ARRAY_BUFFER, all, gl.STREAM_DRAW);
  gl.enable(gl.BLEND); gl.blendFunc(gl.SRC_ALPHA, gl.ONE); gl.depthMask(false);
  gl.drawArrays(gl.POINTS, 0, n / 7);
  gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);
}
// clouds live on a lat/lon grid around the aircraft
function jwUpdateClouds() {
  const [lat, lon] = worldToGeo(R.eye[0] || sim.pos[0], R.eye[2] || sim.pos[2]);
  const cs = 0.18, ci = Math.floor(lat / cs), cj = Math.floor(lon / cs), key = ci + ',' + cj + ',' + JW.ov;
  if (JW.cloudKey === key) return;
  JW.cloudKey = key;
  const puffs = [];
  for (let a = -3; a <= 3; a++) for (let b = -3; b <= 3; b++) {
    const i = ci + a, j = cj + b;
    const nC = Math.floor(hash2(i, j, 71) * 3);
    for (let c = 0; c < nC; c++) {
      const la = (i + hash2(i, j, 72 + c)) * cs, lo = (j + hash2(i, j, 80 + c)) * cs;
      const [cx, cz] = geoToWorld(la, lo);
      const base = 1300 + hash2(i, j, 90 + c) * 900, w = 450 + hash2(i, j, 95 + c) * 900, k = 8 + Math.floor(hash2(i, j, 99 + c) * 10);
      for (let q = 0; q < k; q++) {
        const r1 = hash2(i * 31 + q, j, c), r2 = hash2(i, j * 17 + q, c + 5), r3 = hash2(i + q, j - q, c + 9);
        const ang = r1 * Math.PI * 2, rr = Math.sqrt(r2) * w, up = r3 * w * 0.38 * (1 - rr / w * 0.6), size = 190 + r3 * 240;
        puffs.push({ p: [cx + Math.cos(ang) * rr * 1.3, base + up + size * 0.35, cz + Math.sin(ang) * rr * 0.8], size, shade: clamp(up / (w * 0.3), 0, 1), rot: r1 * 6.28 });
      }
    }
  }
  clouds.puffs = puffs; clouds.order = null; clouds.d2 = null;
  clouds.data = new Float32Array(Math.max(puffs.length, 1) * 9);
  if (clouds.vao) for (const bf of clouds.vao.bufs) gl.deleteBuffer(bf);
  clouds.vao = makeVAO([{ loc: 0, size: 2, data: new Float32Array([-1, -1, 1, -1, -1, 1, 1, 1]) },
  { data: clouds.data, stride: 9, layout: [[4, 4, 0], [5, 4, 4], [6, 1, 8]], divisor: 1, dynamic: true }]);
}

function jwBuildRings(ap) {
  const [cx, cz] = geoToWorld(ap.lat, ap.lon);
  course.centre = [cx, 0, cz]; course.centreGeo = [ap.lat, ap.lon]; course.landR = 4000;
  const pts = [], n = 10, rng = makeRng(ap.icao.charCodeAt(2) * 97 + ap.icao.charCodeAt(3));
  for (let i = 0; i < n; i++) {
    const a = Math.PI * 0.55 - i / n * Math.PI * 2 * 0.92, r = 5200 * (1 + Math.sin(i * 2.1) * 0.22) + rng() * 400;
    pts.push([cx + Math.cos(a) * r, 0, cz + Math.sin(a) * r]);
  }
  for (const p of pts) {
    let top = 0;
    for (let k = 0; k < 9; k++) { const a = k / 9 * Math.PI * 2, rr = k ? 350 : 0; top = Math.max(top, jwHeight(p[0] + Math.cos(a) * rr, p[2] + Math.sin(a) * rr)); }
    p[1] = Math.max(top + 200 + rng() * 80, 220);
  }
  for (let pass = 0; pass < 6; pass++) for (let i = 0; i < n - 1; i++) {
    const a = pts[i], b = pts[i + 1];
    for (let t = 0.05; t < 1; t += 0.05) {
      const y = lerp(a[1], b[1], t), need = Math.max(jwHeight(lerp(a[0], b[0], t), lerp(a[2], b[2], t)), 0) + 180;
      if (y < need) { a[1] += (need - y) * 0.6; b[1] += (need - y) * 0.6; }
    }
  }
  course.rings = pts.map((p, i) => {
    const prev = pts[Math.max(0, i - 1)], next = pts[Math.min(n - 1, i + 1)];
    const dir = i === 0 ? vnorm(vsub(next, p)) : i === n - 1 ? vnorm(vsub(p, prev)) : vnorm(vsub(next, prev));
    return { p, n: vnorm([dir[0], dir[1] * 0.5, dir[2]]) };
  });
}
function nearestPlace(lat, lon, maxM) {
  let best = null, bd = maxM;
  for (const p of JW.places) {
    if (p.k !== 'c' || Math.abs(p.lat - lat) > 0.4 || Math.abs(p.lon - lon) > 0.5) continue;
    const d = geoDist(lat, lon, p.lat, p.lon) / (1 + Math.log10(Math.max(p.p, 10000) / 10000));
    if (d < bd) { bd = d; best = p; }
  }
  return best;
}

// offline relief of the whole world (drawn under the Japan relief)
function worldReliefCanvas() {
  if (mapv.wrelief) return mapv.wrelief;
  const C = WORLD_COARSE, s = 2, W = C.w / s, H = C.h / s, c = document.createElement('canvas');
  c.width = W; c.height = H;
  const g = c.getContext('2d'), img = g.createImageData(W, H), d = JW.world;
  for (let j = 0; j < H; j++) for (let i = 0; i < W; i++) {
    const k = (j * s) * C.w + i * s, h = d[k], o = (j * W + i) * 4;
    if (h <= 0) { img.data[o] = 158; img.data[o + 1] = 196; img.data[o + 2] = 222; img.data[o + 3] = 255; continue; }
    const hx = (d[k + s] || 0) - (d[k - s] || 0), hy = (d[k + s * C.w] || 0) - (d[k - s * C.w] || 0);
    const shade = clamp(1 - (hx + hy) * 0.0012, 0.6, 1.25);
    const lat = latFromMerc((j * s + 0.5) / C.h);
    const col = Math.abs(lat) > 66 || h > 4200 ? [236, 238, 242] : h < 200 ? [190, 210, 160] : h < 800 ? [172, 194, 140] : h < 1800 ? [184, 176, 140] : [200, 190, 170];
    img.data[o] = col[0] * shade; img.data[o + 1] = col[1] * shade; img.data[o + 2] = col[2] * shade; img.data[o + 3] = 255;
  }
  g.putImageData(img, 0, 0);
  return (mapv.wrelief = c);
}
function drawRelief(g, ox, oy, zoom) {
  const W = worldReliefCanvas(), wz = WORLD_COARSE.z, ws = 2 ** (zoom - wz) * 2;
  const full = 256 * 2 ** zoom;
  for (const shift of [-full, 0, full]) g.drawImage(W, ox + shift, oy, W.width * ws, W.height * ws);
  if (zoom >= 4) { const rel = reliefCanvas(), C = JP_COARSE, rs = 2 ** (zoom - C.z) * 2;
    g.drawImage(rel, ox + C.x0 * 256 * 2 ** (zoom - C.z), oy + C.y0 * 256 * 2 ** (zoom - C.z), rel.width * rs, rel.height * rs); }
}
function reliefCanvas() {
  if (mapv.relief) return mapv.relief;
  const C = JP_COARSE, s = 2, W = C.w / s, H = C.h / s, c = document.createElement('canvas');
  c.width = W; c.height = H;
  const g = c.getContext('2d'), img = g.createImageData(W, H), d = JW.coarse;
  for (let j = 0; j < H; j++) for (let i = 0; i < W; i++) {
    const k = (j * s) * C.w + i * s, h = d[k], o = (j * W + i) * 4;
    if (h === -32768) {
      const N = JW.jpNear, near = N && N.g[((j * s) / N.B | 0) * N.bw + ((i * s) / N.B | 0)];
      if (near) { img.data[o] = 158; img.data[o + 1] = 196; img.data[o + 2] = 222; img.data[o + 3] = 255; } else img.data[o + 3] = 0;
      continue;
    }
    const hx = (d[k + s] === -32768 ? 0 : d[k + s]) - (d[k - s] === -32768 ? 0 : d[k - s] || 0);
    const hy = (d[k + s * C.w] === -32768 ? 0 : d[k + s * C.w] || 0) - (d[k - s * C.w] === -32768 ? 0 : d[k - s * C.w] || 0);
    const shade = clamp(1 - (hx + hy) * 0.004, 0.6, 1.25);
    const col = h < 50 ? [196, 214, 168] : h < 400 ? [170, 196, 140] : h < 1200 ? [150, 170, 120] : h < 2200 ? [176, 164, 136] : [232, 232, 236];
    img.data[o] = col[0] * shade; img.data[o + 1] = col[1] * shade; img.data[o + 2] = col[2] * shade; img.data[o + 3] = 255;
  }
  g.putImageData(img, 0, 0);
  mapv.relief = c;
  return c;
}
function mapTile(z, x, y) {
  const key = z + '/' + x + '/' + y;
  let im = mapv.imgs.get(key);
  if (!im) {
    if (mapv.imgs.size > 400) { const first = mapv.imgs.keys().next().value; mapv.imgs.delete(first); }
    im = new Image(); im.decoding = 'async';
    im.onerror = () => { im.failed = true; mapv.fails = (mapv.fails || 0) + 1; if (mapv.fails > 8 && !mapv.oks) JW.mapBlocked = true; };
    im.onload = () => { mapv.oks = (mapv.oks || 0) + 1; };
    const n = 2 ** z, lat = latFromMerc((y + 0.5) / n), lon = (x + 0.5) / n * 360 - 180;
    const jp = z >= 5 && !Number.isNaN(jpCoarseH(lat, lon)) || (z >= 7 && lat > 24 && lat < 46 && lon > 123 && lon < 146 && !Number.isNaN(jpCoarseH(latFromMerc(y / n), x / n * 360 - 180)));
    im.src = jp ? `${GSI}std/${z}/${x}/${y}.png` : `${ESRI}World_Street_Map/MapServer/tile/${z}/${y}/${x}`;
    mapv.imgs.set(key, im);
  }
  return im;
}
// draw a map into ctx rect (x,y,w,h) centred on (lat,lon) at fractional zoom
function drawMapTo(g, x, y, w, h, lat, lon, zoom, opt = {}) {
  lon = wrapLon(lon);
  g.save(); g.beginPath(); g.rect(x, y, w, h); g.clip();
  g.fillStyle = '#9ec4de'; g.fillRect(x, y, w, h);
  const [cx, cy] = mapPx(lat, lon, zoom), ox = x + w / 2 - cx, oy = y + h / 2 - cy;
  const useTiles = JW.online && !JW.mapBlocked;
  if (useTiles) {
    const z = clamp(Math.round(zoom), 2, 17), sc = 2 ** (zoom - z), ts = 256 * sc;
    const [tcx, tcy] = mapPx(lat, lon, z);
    const x0 = Math.floor((tcx - w / 2 / sc) / 256), x1 = Math.floor((tcx + w / 2 / sc) / 256);
    const y0 = Math.floor((tcy - h / 2 / sc) / 256), y1 = Math.floor((tcy + h / 2 / sc) / 256);
    // relief underneath while tiles load
    drawRelief(g, ox, oy, zoom);
    for (let ty = y0; ty <= y1; ty++) for (let tx = x0; tx <= x1; tx++) {
      if (ty < 0 || ty >= 2 ** z) continue;
      const im = mapTile(z, ((tx % 2 ** z) + 2 ** z) % 2 ** z, ty);
      if (im.complete && !im.failed && im.naturalWidth) g.drawImage(im, x + w / 2 + (tx * 256 - tcx) * sc, y + h / 2 + (ty * 256 - tcy) * sc, ts + 0.5, ts + 0.5);
    }
  } else if (JW.coarse) {
    g.imageSmoothingEnabled = true;
    drawRelief(g, ox, oy, zoom);
  }
  const FULLW = 256 * 2 ** zoom;
  const P = (la, lo) => { const p = mapPx(la, lo, zoom); let dx = p[0] - cx; dx -= Math.round(dx / FULLW) * FULLW; return [x + w / 2 + dx, p[1] + oy]; };
  const inView = p => p[0] > x - 40 && p[0] < x + w + 40 && p[1] > y - 20 && p[1] < y + h + 20;
  // labels are placed in priority order and skipped when they would overlap
  const boxes = [];
  const fits = (bx, by, bw, bh) => {
    for (const q of boxes) if (bx < q[0] + q[2] && bx + bw > q[0] && by < q[1] + q[3] && by + bh > q[1]) return false;
    boxes.push([bx, by, bw, bh]); return true;
  };
  const aps = [];
  for (const ap of JW.airports) {
    const s = P(ap.lat, ap.lon); if (!inView(s)) continue;
    const major = ap.rw[0].len >= 2400;
    const keep = JW.dest === ap || JW.dep === ap || mapv.sel === ap;
    if (!keep && ap.cc !== 'JP' && (zoom < 4.6 ? !(ap.big && ap.rw[0].len >= 3200) : zoom < 6.2 ? !ap.big : false)) continue;
    if (!keep && ap.cc === 'JP' && zoom < 4.6 && !['RJTT', 'RJAA', 'RJBB', 'RJCC', 'ROAH', 'RJFF', 'RJGG'].includes(ap.icao)) continue;
    if (zoom < 6 && !major && !keep) continue;
    const pri = (mapv.sel === ap ? 4 : 0) + (JW.dest === ap ? 3 : 0) + (JW.dep === ap ? 2 : 0) + (major ? 1 : 0) + ap.rw[0].len / 1e5;
    aps.push({ ap, s, major, pri });
  }
  aps.sort((p, q) => q.pri - p.pri);
  const apLabels = [];
  if (opt.labels) {
    g.font = `700 12px ${UI}`;
    for (const o of aps) {
      o.lw = g.measureText(o.ap.name).width;
      if ((zoom >= 6.3 || o.pri >= 1) && fits(o.s[0] - 7, o.s[1] - 8, o.lw + 17, 16)) apLabels.push(o);
      else fits(o.s[0] - 6, o.s[1] - 6, 12, 12);
    }
  }
  const placeLabels = [];
  if (opt.labels && !useTiles) {
    if (!JW.placesByPop) JW.placesByPop = JW.places.filter(p => p.k === 'c').sort((p, q) => q.p - p.p).concat(JW.places.filter(p => p.k === 'm'));
    const minPop = zoom < 6 ? 900000 : zoom < 7 ? 250000 : zoom < 8 ? 100000 : 0;
    g.font = `12px ${UI}`;
    for (const p of JW.placesByPop) {
      if (p.k === 'c' && p.p < minPop) continue;
      if (p.k === 'm' && zoom < 8) continue;
      const s = P(p.lat, p.lon); if (!inView(s)) continue;
      const lw = g.measureText(p.n).width;
      if (fits(s[0] - 3, s[1] - 7, lw + 8, 14)) placeLabels.push([p, s]);
      if (placeLabels.length > 160) break;
    }
  }
  g.textAlign = 'left'; g.textBaseline = 'middle';
  g.font = `12px ${UI}`;
  for (const [p, s] of placeLabels) {
    g.fillStyle = p.k === 'm' ? '#6b4a1e' : '#1d2733';
    g.beginPath(); g.arc(s[0], s[1], 2.5, 0, 7); g.fill();
    g.strokeStyle = 'rgba(255,255,255,0.85)'; g.lineWidth = 3; g.strokeText(p.n, s[0] + 5, s[1]); g.fillText(p.n, s[0] + 5, s[1]);
  }
  // landmarks (sightseeing)
  if (opt.labels && zoom >= 6.5 && typeof LANDMARKS !== 'undefined') {
    g.font = `700 12px ${UI}`;
    for (const l of LANDMARKS) {
      const sp = P(l[1], l[2]); if (!inView(sp)) continue;
      const tw = g.measureText('★' + l[0]).width;

      g.fillStyle = '#b07800'; g.fillText('★' + l[0], sp[0] - 6, sp[1]);
    }
  }
  // airports & runways
  for (let i = aps.length - 1; i >= 0; i--) {
    const { ap, s, major } = aps[i];
    if (zoom >= 10) {
      g.strokeStyle = '#2b2f36'; g.lineWidth = Math.max(2, 2 ** (zoom - 13) * 12);
      for (const r of ap.rw) { const a = P(r.le[0], r.le[1]), b = P(r.he[0], r.he[1]); g.beginPath(); g.moveTo(a[0], a[1]); g.lineTo(b[0], b[1]); g.stroke(); }
    }
    const isDest = JW.dest === ap, isDep = JW.dep === ap, isSel = mapv.sel === ap;
    g.fillStyle = isDest ? COL.magenta : isDep ? '#1b8a4c' : '#1f5fae';
    g.strokeStyle = '#fff'; g.lineWidth = 1.5;
    g.beginPath(); g.arc(s[0], s[1], isSel ? 8 : major ? 6 : 4.5, 0, 7); g.fill(); g.stroke();
  }
  g.font = `700 12px ${UI}`;
  for (const { ap, s } of apLabels) {
    g.strokeStyle = 'rgba(255,255,255,0.9)'; g.lineWidth = 3; g.strokeText(ap.name, s[0] + 9, s[1]);
    g.fillStyle = '#0d2a52'; g.fillText(ap.name, s[0] + 9, s[1]);
  }
  // track, destination line, aircraft
  if (JW.active) {
    const [pl, po] = planeGeo();
    if (game.track && game.track.length > 1) {
      g.strokeStyle = 'rgba(227,107,220,0.8)'; g.lineWidth = 2; g.setLineDash([5, 4]); g.beginPath();
      game.track.forEach((q, i) => { const s = P(q[0], q[1]); i ? g.lineTo(s[0], s[1]) : g.moveTo(s[0], s[1]); });
      const s = P(pl, po); g.lineTo(s[0], s[1]); g.stroke(); g.setLineDash([]);
    }
    if (JW.dest) {
      const a = P(pl, po), b = P(JW.dest.lat, JW.dest.lon);
      g.strokeStyle = COL.magenta; g.lineWidth = 2.5; g.beginPath(); g.moveTo(a[0], a[1]); g.lineTo(b[0], b[1]); g.stroke();
    }
    const s = P(pl, po);
    g.save(); g.translate(s[0], s[1]); g.rotate(attitude().hdg * DEG);
    g.fillStyle = '#ffd24a'; g.strokeStyle = '#000'; g.lineWidth = 1.2;
    const k = opt.big ? 1.4 : 1;
    g.beginPath(); g.moveTo(0, -10 * k); g.lineTo(7 * k, 8 * k); g.lineTo(0, 4 * k); g.lineTo(-7 * k, 8 * k); g.closePath(); g.fill(); g.stroke();
    g.restore();
  }
  g.restore();
}
// =====================================================================
//  BUILT-UP AREAS: places made by hand, all in the sim's own world, the
//  way a driving game's city is — clean surfaces, modelled buildings,
//  aircraft at the gates, markings and lights — instead of the
//  photogrammetry, which melts at ground level. Inside a region, round the
//  camera as far as the city is loaded, Google's 3D gives way entirely
//  (and PLATEAU's too); the map's own buildings, roads and trees stand
//  there; outside, everything is as it was.
//  The Tokyo Bay shore from Haneda through Kawasaki to Yokohama.
// =====================================================================
const AUTH_AREAS = [
  // [lat, lon] outline (roughly: Haneda and Ota ward's south, Kawasaki, Yokohama city)
  { id: 'bayshore', poly: [[35.60, 139.70], [35.60, 139.83], [35.50, 139.83], [35.40, 139.72], [35.30, 139.68], [35.30, 139.47], [35.43, 139.45], [35.55, 139.55]] },
];
const AUTH = { on: false, v: [0, 0, 0, 0], area: null, off: false, r: 0 };
function authInPoly(poly, lat, lon) {
  let inside = false;
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    const [ai, bi] = poly[i], [aj, bj] = poly[j];
    if ((ai > lat) !== (aj > lat) && lon < (bj - bi) * (lat - ai) / (aj - ai) + bi) inside = !inside;
  }
  return inside;
}
function authUpdate() {
  AUTH.on = false; AUTH.v = [0, 0, 0, 0]; AUTH.area = null;
  if (typeof JW === 'undefined' || !JW.active || !JW.online || AUTH.off) return;
  const [la, lo] = worldToGeo(R.eye[0], R.eye[2]);
  const agl = R.eye[1] - Math.max(rawHeightGeo(la, lo), 0);
  if (agl > 2500) return;                                             // (from high up the photogrammetry reads better)
  for (const A of AUTH_AREAS) {
    if (!authInPoly(A.poly, la, lo)) continue;
    // the circle round the camera that the city loads (p6_city.js cityUpdate), a little inside it
    const r = game.touch ? clamp(1800 + agl * 1.5, 1800, 3500) : clamp(2800 + agl * 2.2, 2800, 7000);
    AUTH.on = true; AUTH.area = A; AUTH.r = r;
    AUTH.v = [R.eye[0], R.eye[2], r * 0.72, r * 0.9];
    A.lat = la; A.lon = lo; A.r0 = r * 0.72; A.r1 = r * 0.9;
    break;
  }
}
// is a place inside the built-up circle in use (lat, lon)? (the map's own buildings aren't hidden under PLATEAU there)
function authHas(lat, lon) {
  const A = AUTH.area; if (!A) return false;
  return geoDist(lat, lon, A.lat, A.lon) < A.r1;
}
// whether the sim's own world wholly replaces what lies in this box (x0 z0 x1 z1, world metres): inside the
// inner circle Google's and PLATEAU's meshes are all cut, so those tiles needn't be loaded or drawn at all
function authHides(x0, z0, x1, z1) {
  if (!AUTH.on) return false;
  const [cx, cz, r0] = AUTH.v, dx = Math.max(Math.abs(x0 - cx), Math.abs(x1 - cx)), dz = Math.max(Math.abs(z0 - cz), Math.abs(z1 - cz));
  return dx * dx + dz * dz < r0 * r0 * 0.96;
}

// =====================================================================
//  CITY: real 3D buildings and trees from OpenStreetMap (OpenFreeMap
//  vector tiles, zoom 14). A worker decodes each tile, extrudes every
//  building to its mapped height and scatters trees over woods and
//  parks; the main thread only uploads the finished buffers.
// =====================================================================
function cityWorkerMain() {
  const KZ = 110574, DEG = Math.PI / 180;
  const latFromMerc = my => Math.atan(Math.sinh(Math.PI * (1 - 2 * my))) / DEG;
  const clamp = (v, a, b) => v < a ? a : v > b ? b : v;
  const td = new TextDecoder();
  // ---- protobuf / MVT ----
  let B, P;
  const rv = () => { let r = 0, s = 1, c; do { c = B[P++]; r += (c & 0x7f) * s; s *= 128; } while (c >= 0x80); return r; };
  const skip = t => { if (t === 0) rv(); else if (t === 1) P += 8; else if (t === 2) { const l = rv(); P += l; } else if (t === 5) P += 4; };
  function readValue(end) {
    let v = null;
    while (P < end) {
      const k = rv(), f = k >> 3, t = k & 7;
      if (f === 1) { const l = rv(); v = td.decode(B.subarray(P, P + l)); P += l; }
      else if (f === 2) { v = new DataView(B.buffer, B.byteOffset + P, 4).getFloat32(0, true); P += 4; }
      else if (f === 3) { v = new DataView(B.buffer, B.byteOffset + P, 8).getFloat64(0, true); P += 8; }
      else if (f === 4 || f === 5) v = rv();
      else if (f === 6) { const n = rv(); v = n % 2 ? -(n + 1) / 2 : n / 2; }
      else if (f === 7) v = !!rv();
      else skip(t);
    }
    return v;
  }
  function readLayers(buf, wanted) {
    B = buf; P = 0; const out = {};
    while (P < B.length) {
      const k = rv(), f = k >> 3, t = k & 7;
      if (f !== 3 || t !== 2) { skip(t); continue; }
      const end = rv() + P, L = { name: '', keys: [], vals: [], feats: [], extent: 4096 };
      let keep = true;
      while (P < end) {
        const k2 = rv(), f2 = k2 >> 3, t2 = k2 & 7;
        if (f2 === 1) { const l = rv(); L.name = td.decode(B.subarray(P, P + l)); P += l; if (!wanted.includes(L.name)) { keep = false; break; } }
        else if (f2 === 2) { const l = rv(); L.feats.push([P, P + l]); P += l; }
        else if (f2 === 3) { const l = rv(); L.keys.push(td.decode(B.subarray(P, P + l))); P += l; }
        else if (f2 === 4) { const l = rv(); L.vals.push(readValue(P + l)); }
        else if (f2 === 5) L.extent = rv();
        else skip(t2);
      }
      P = end;
      if (keep && L.name) out[L.name] = L;
    }
    return out;
  }
  function readFeature(L, fe) {
    P = fe[0]; const end = fe[1], props = {}; let type = 0, geom = null;
    while (P < end) {
      const k = rv(), f = k >> 3, t = k & 7;
      if (f === 2) { const e = rv() + P; while (P < e) { const ki = rv(), vi = rv(); props[L.keys[ki]] = L.vals[vi]; } }
      else if (f === 3) type = rv();
      else if (f === 4) { const e = rv() + P; geom = []; while (P < e) geom.push(rv()); }
      else skip(t);
    }
    return { props, type, geom };
  }
  function decodeRings(geom) {
    const out = []; let x = 0, y = 0, i = 0, cur = null;
    while (i < geom.length) {
      const c = geom[i++], cmd = c & 7, cnt = c >> 3;
      if (cmd === 7) { if (cur && cur.length >= 6) out.push(cur); cur = null; continue; }
      for (let j = 0; j < cnt; j++) {
        const dx = geom[i++], dy = geom[i++];
        x += (dx >>> 1) ^ -(dx & 1); y += (dy >>> 1) ^ -(dy & 1);
        if (cmd === 1) { if (cur && cur.length >= 6) out.push(cur); cur = [x, y]; } else if (cur) cur.push(x, y);
      }
    }
    if (cur && cur.length >= 6) out.push(cur);
    return out;
  }
  // line strings (roads): a MoveTo starts a new line
  function decodePoints(geom) {
    const out = []; let x = 0, y = 0, i = 0;
    while (i < geom.length) {
      const c = geom[i++], cnt = c >> 3;
      for (let j = 0; j < cnt; j++) { const dx = geom[i++], dy = geom[i++]; x += (dx >>> 1) ^ -(dx & 1); y += (dy >>> 1) ^ -(dy & 1); out.push(x, y); }
    }
    return out;
  }
  function decodeLines(geom) {
    const out = []; let x = 0, y = 0, i = 0, cur = null;
    while (i < geom.length) {
      const c = geom[i++], cmd = c & 7, cnt = c >> 3;
      if (cmd === 7) continue;
      for (let j = 0; j < cnt; j++) {
        const dx = geom[i++], dy = geom[i++];
        x += (dx >>> 1) ^ -(dx & 1); y += (dy >>> 1) ^ -(dy & 1);
        if (cmd === 1) { if (cur && cur.length >= 4) out.push(cur); cur = [x, y]; } else if (cur) cur.push(x, y);
      }
    }
    if (cur && cur.length >= 4) out.push(cur);
    return out;
  }
  // Sutherland-Hodgman against the tile square; points on the border are snapped exactly
  function clipRing(r, E) {
    let pts = r;
    for (let side = 0; side < 4 && pts.length >= 6; side++) {
      const out = [], n = pts.length;
      const ins = (x, y) => side === 0 ? x >= 0 : side === 1 ? x <= E : side === 2 ? y >= 0 : y <= E;
      const cut = (ax, ay, bx, by) => {
        if (side < 2) { const X = side ? E : 0, t = (X - ax) / (bx - ax); out.push(X, ay + (by - ay) * t); }
        else { const Y = side === 3 ? E : 0, t = (Y - ay) / (by - ay); out.push(ax + (bx - ax) * t, Y); }
      };
      for (let i = 0; i < n; i += 2) {
        const sx = pts[(i - 2 + n) % n], sy = pts[(i - 1 + n) % n], ex = pts[i], ey = pts[i + 1];
        const ie = ins(ex, ey), is = ins(sx, sy);
        if (ie) { if (!is) cut(sx, sy, ex, ey); out.push(ex, ey); }
        else if (is) cut(sx, sy, ex, ey);
      }
      pts = out;
    }
    const o = [];
    for (let i = 0; i < pts.length; i += 2) { const l = o.length; if (l && o[l - 2] === pts[i] && o[l - 1] === pts[i + 1]) continue; o.push(pts[i], pts[i + 1]); }
    if (o.length >= 4 && o[0] === o[o.length - 2] && o[1] === o[o.length - 1]) o.length -= 2;
    return o.length >= 6 ? o : null;
  }
  const ringArea = r => { let s = 0; for (let i = 0, n = r.length; i < n; i += 2) { const j = (i + 2) % n; s += r[i] * r[j + 1] - r[j] * r[i + 1]; } return s / 2; };
  function polygons(L, E, filter) {
    const out = [];
    if (!L) return out;
    for (const fe of L.feats) {
      const f = readFeature(L, fe);
      if (f.type !== 3 || !f.geom || !filter(f.props)) continue;
      let cur = null;
      for (const r0 of decodeRings(f.geom)) {
        const a0 = ringArea(r0);
        if (a0 === 0) continue;
        const r = clipRing(r0, E); if (!r) continue;
        const a = ringArea(r);
        if (Math.abs(a) < 1e-3) continue;
        if (a0 > 0) { cur = { rings: [r], props: f.props }; out.push(cur); }
        else if (cur) cur.rings.push(r);
      }
    }
    return out;
  }
  // ---- earcut (after Mapbox earcut, without the z-order hash) ----
  function earcut(data, holes) {
    const outerLen = holes.length ? holes[0] * 2 : data.length;
    let outer = linkedList(data, 0, outerLen, true);
    const tris = [];
    if (!outer || outer.next === outer.prev) return tris;
    if (holes.length) outer = eliminateHoles(data, holes, outer);
    earcutLinked(outer, tris, 0);
    return tris;
  }
  function linkedList(data, start, end, cw) {
    let last;
    if (cw === (signedArea(data, start, end) > 0)) for (let i = start; i < end; i += 2) last = insertNode(i / 2, data[i], data[i + 1], last);
    else for (let i = end - 2; i >= start; i -= 2) last = insertNode(i / 2, data[i], data[i + 1], last);
    if (last && equals(last, last.next)) { removeNode(last); last = last.next; }
    return last;
  }
  function filterPoints(start, end) {
    if (!start) return start;
    if (!end) end = start;
    let p = start, again;
    do {
      again = false;
      if (!p.steiner && (equals(p, p.next) || area(p.prev, p, p.next) === 0)) { removeNode(p); p = end = p.prev; if (p === p.next) break; again = true; }
      else p = p.next;
    } while (again || p !== end);
    return end;
  }
  function earcutLinked(ear, tris, pass) {
    if (!ear) return;
    let stop = ear, guard = 0;
    while (ear.prev !== ear.next && guard++ < 200000) {
      const prev = ear.prev, next = ear.next;
      if (isEar(ear)) { tris.push(prev.i, ear.i, next.i); removeNode(ear); ear = next.next; stop = next.next; continue; }
      ear = next;
      if (ear === stop) {
        if (!pass) earcutLinked(filterPoints(ear), tris, 1);
        else if (pass === 1) { ear = cureLocalIntersections(filterPoints(ear), tris); earcutLinked(ear, tris, 2); }
        else if (pass === 2) splitEarcut(ear, tris);
        break;
      }
    }
  }
  function isEar(ear) {
    const a = ear.prev, b = ear, c = ear.next;
    if (area(a, b, c) >= 0) return false;
    let p = c.next;
    while (p !== a) {
      if (pointInTriangle(a.x, a.y, b.x, b.y, c.x, c.y, p.x, p.y) && area(p.prev, p, p.next) >= 0) return false;
      p = p.next;
    }
    return true;
  }
  function cureLocalIntersections(start, tris) {
    let p = start;
    do {
      const a = p.prev, b = p.next.next;
      if (!equals(a, b) && intersects(a, p, p.next, b) && locallyInside(a, b) && locallyInside(b, a)) {
        tris.push(a.i, p.i, b.i); removeNode(p); removeNode(p.next); p = start = b;
      }
      p = p.next;
    } while (p !== start);
    return filterPoints(p);
  }
  function splitEarcut(start, tris) {
    let a = start;
    do {
      let b = a.next.next;
      while (b !== a.prev) {
        if (a.i !== b.i && isValidDiagonal(a, b)) {
          let c = splitPolygon(a, b);
          a = filterPoints(a, a.next); c = filterPoints(c, c.next);
          earcutLinked(a, tris, 0); earcutLinked(c, tris, 0);
          return;
        }
        b = b.next;
      }
      a = a.next;
    } while (a !== start);
  }
  function eliminateHoles(data, holes, outer) {
    const q = [];
    for (let i = 0; i < holes.length; i++) {
      const s = holes[i] * 2, e = i < holes.length - 1 ? holes[i + 1] * 2 : data.length;
      const list = linkedList(data, s, e, false);
      if (!list) continue;
      if (list === list.next) list.steiner = true;
      q.push(getLeftmost(list));
    }
    q.sort((a, b) => a.x - b.x);
    for (const h of q) { const br = findHoleBridge(h, outer); if (!br) continue; const rev = splitPolygon(br, h); filterPoints(rev, rev.next); outer = filterPoints(br, br.next); }
    return outer;
  }
  function findHoleBridge(hole, outer) {
    let p = outer, qx = -Infinity, m;
    const hx = hole.x, hy = hole.y;
    do {
      if (hy <= p.y && hy >= p.next.y && p.next.y !== p.y) {
        const x = p.x + (hy - p.y) * (p.next.x - p.x) / (p.next.y - p.y);
        if (x <= hx && x > qx) { qx = x; m = p.x < p.next.x ? p : p.next; if (x === hx) return m; }
      }
      p = p.next;
    } while (p !== outer);
    if (!m) return null;
    const stop = m, mx = m.x, my = m.y; let tanMin = Infinity;
    p = m;
    do {
      if (hx >= p.x && p.x >= mx && hx !== p.x && pointInTriangle(hy < my ? hx : qx, hy, mx, my, hy < my ? qx : hx, hy, p.x, p.y)) {
        const tan = Math.abs(hy - p.y) / (hx - p.x);
        if (locallyInside(p, hole) && (tan < tanMin || (tan === tanMin && (p.x > m.x || (p.x === m.x && area(m.prev, m, p.prev) < 0 && area(p.next, m, m.next) < 0))))) { m = p; tanMin = tan; }
      }
      p = p.next;
    } while (p !== stop);
    return m;
  }
  function getLeftmost(s) { let p = s, l = s; do { if (p.x < l.x || (p.x === l.x && p.y < l.y)) l = p; p = p.next; } while (p !== s); return l; }
  function pointInTriangle(ax, ay, bx, by, cx, cy, px, py) {
    return (cx - px) * (ay - py) >= (ax - px) * (cy - py) && (ax - px) * (by - py) >= (bx - px) * (ay - py) && (bx - px) * (cy - py) >= (cx - px) * (by - py);
  }
  function isValidDiagonal(a, b) {
    return a.next.i !== b.i && a.prev.i !== b.i && !intersectsPolygon(a, b) &&
      ((locallyInside(a, b) && locallyInside(b, a) && middleInside(a, b) && (area(a.prev, a, b.prev) || area(a, b.prev, b))) ||
       (equals(a, b) && area(a.prev, a, a.next) > 0 && area(b.prev, b, b.next) > 0));
  }
  const area = (p, q, r) => (q.y - p.y) * (r.x - q.x) - (q.x - p.x) * (r.y - q.y);
  const equals = (a, b) => a.x === b.x && a.y === b.y;
  const sign = v => v > 0 ? 1 : v < 0 ? -1 : 0;
  const onSeg = (p, q, r) => q.x <= Math.max(p.x, r.x) && q.x >= Math.min(p.x, r.x) && q.y <= Math.max(p.y, r.y) && q.y >= Math.min(p.y, r.y);
  function intersects(p1, q1, p2, q2) {
    const o1 = sign(area(p1, q1, p2)), o2 = sign(area(p1, q1, q2)), o3 = sign(area(p2, q2, p1)), o4 = sign(area(p2, q2, q1));
    if (o1 !== o2 && o3 !== o4) return true;
    return (o1 === 0 && onSeg(p1, p2, q1)) || (o2 === 0 && onSeg(p1, q2, q1)) || (o3 === 0 && onSeg(p2, p1, q2)) || (o4 === 0 && onSeg(p2, q1, q2));
  }
  function intersectsPolygon(a, b) {
    let p = a;
    do { if (p.i !== a.i && p.next.i !== a.i && p.i !== b.i && p.next.i !== b.i && intersects(p, p.next, a, b)) return true; p = p.next; } while (p !== a);
    return false;
  }
  const locallyInside = (a, b) => area(a.prev, a, a.next) < 0 ? area(a, b, a.next) >= 0 && area(a, a.prev, b) >= 0 : area(a, b, a.prev) < 0 || area(a, a.next, b) < 0;
  function middleInside(a, b) {
    let p = a, inside = false; const px = (a.x + b.x) / 2, py = (a.y + b.y) / 2;
    do { if (((p.y > py) !== (p.next.y > py)) && p.next.y !== p.y && (px < (p.next.x - p.x) * (py - p.y) / (p.next.y - p.y) + p.x)) inside = !inside; p = p.next; } while (p !== a);
    return inside;
  }
  function splitPolygon(a, b) {
    const a2 = mk(a.i, a.x, a.y), b2 = mk(b.i, b.x, b.y), an = a.next, bp = b.prev;
    a.next = b; b.prev = a; a2.next = an; an.prev = a2; b2.next = a2; a2.prev = b2; bp.next = b2; b2.prev = bp;
    return b2;
  }
  function insertNode(i, x, y, last) {
    const p = mk(i, x, y);
    if (!last) { p.prev = p; p.next = p; } else { p.next = last.next; p.prev = last; last.next.prev = p; last.next = p; }
    return p;
  }
  function removeNode(p) { p.next.prev = p.prev; p.prev.next = p.next; }
  function mk(i, x, y) { return { i, x, y, prev: null, next: null, steiner: false }; }
  function signedArea(d, s, e) { let sum = 0; for (let i = s, j = e - 2; i < e; i += 2) { sum += (d[j] - d[i]) * (d[i + 1] + d[j + 1]); j = i; } return sum; }

  // ---- tile build ----
  const hash = (a, b) => { let h = (Math.imul(a | 0, 73856093) ^ Math.imul(b | 0, 19349663)) >>> 0; h = Math.imul(h ^ (h >>> 13), 1274126177) >>> 0; return (h ^ (h >>> 16)) >>> 0; };
  function build(buf, m) {
    const L = readLayers(buf, ['building', 'landcover', 'landuse', 'transportation', 'poi', 'aeroway']);
    const E = (L.building || L.landcover || { extent: 4096 }).extent;
    const n = 2 ** m.z, dLon = m.lonE - m.lonW, kxN = 111320 * Math.cos(m.latN * DEG);
    const W = dLon * kxN, Hm = (m.latN - m.latS) * KZ;
    const latCache = new Map();
    const lz = py => { let v = latCache.get(py); if (v === undefined) { v = -(latFromMerc((m.y + py / E) / n) - m.latN) * KZ; if (latCache.size < 20000) latCache.set(py, v); } return v; };
    const lx = px => px / E * dLon * kxN;
    const dem = m.dem;
    // the sim levels the ground around runways (runwayFlatten): the same here, so roads and buildings sit on it
    const rws = m.rws || [], latAt = py => m.latN + (m.latS - m.latN) * py / E, lonAt = px => m.lonW + (m.lonE - m.lonW) * px / E;
    const flatten = (px, py, h) => {
      if (!rws.length) return h;
      const lat = latAt(py), lon = lonAt(px);
      for (const r of rws) {
        const dx = (lon - r.lon) * r.kx, dz = -(lat - r.lat) * KZ, u = dx * r.sin - dz * r.cos, v = dx * r.cos + dz * r.sin;
        const e = Math.hypot(Math.max(Math.abs(u) - r.len / 2 - 90, 0), Math.max(Math.abs(v) - r.wid / 2 - 60, 0));
        if (e < 220) { const t = clamp(e / 220, 0, 1), k = t * t * (3 - 2 * t); h = r.elev + (h - r.elev) * k; }
      }
      return h;
    };
    const ground = (px, py) => flatten(px, py, groundRaw(px, py));
    const groundRaw = (px, py) => {
      if (!dem) return m.g0;
      const fx = clamp(px / E * 256 - 0.5, 0, 254.999), fy = clamp(py / E * 256 - 0.5, 0, 254.999);
      const ix = fx | 0, iy = fy | 0, tx = fx - ix, ty = fy - iy, k = iy * 256 + ix;
      const g = v => v !== v ? 0 : v;
      return (g(dem[k]) * (1 - tx) + g(dem[k + 1]) * tx) * (1 - ty) + (g(dem[k + 256]) * (1 - tx) + g(dem[k + 257]) * tx) * ty;
    };
    // ---------- buildings ----------
    const bpolys = polygons(L.building, E, p => !p.hide_3d);
    let nv = 0, ni = 0;
    for (const p of bpolys) for (let r = 0; r < p.rings.length; r++) { const k = p.rings[r].length / 2; nv += k * 4 + (r === 0 || true ? k : 0) + 80; ni += k * 6; }
    ni += nv;                                                         // generous room for roof triangles (and pitched roofs / roof boxes)
    let vb = new ArrayBuffer(nv * 20), i16 = new Int16Array(vb), f32 = new Float32Array(vb), i8 = new Int8Array(vb);
    let idx = new Uint32Array(ni);
    let v = 0, q = 0;
    const put = (x, z, y, nx, ny, nz, u, st, h, yr) => {
      if (v >= nv) { nv *= 2; const g = new ArrayBuffer(nv * 20); new Uint8Array(g).set(new Uint8Array(vb)); vb = g; i16 = new Int16Array(vb); f32 = new Float32Array(vb); i8 = new Int8Array(vb); }
      const s = v * 10;
      i16[s] = Math.round(x * 10); i16[s + 1] = Math.round(z * 10); f32[v * 5 + 1] = y;
      i8[v * 20 + 8] = nx; i8[v * 20 + 9] = ny; i8[v * 20 + 10] = nz;
      i16[s + 6] = clamp(Math.round(u * 10), -32767, 32767); i16[s + 7] = st; i16[s + 8] = clamp(Math.round(h * 10), 0, 32767); i16[s + 9] = clamp(Math.round(yr * 10), -32767, 32767);
      return v++;
    };
    const tri = (a, b, c) => { if (q + 3 > idx.length) { const g = new Uint32Array(idx.length * 2); g.set(idx); idx = g; } idx[q++] = a; idx[q++] = b; idx[q++] = c; };
    // a flat-shaded triangle facing along n (points [x, y, z] in local metres); st/yr as for walls or roofs
    const face = (p1, p2, p3, n, st, H, yr) => {
      const ex = p2[0] - p1[0], ey = p2[1] - p1[1], ez = p2[2] - p1[2], fx = p3[0] - p1[0], fy = p3[1] - p1[1], fz = p3[2] - p1[2];
      const cx = ey * fz - ez * fy, cy = ez * fx - ex * fz, cz = ex * fy - ey * fx;
      if (cx * n[0] + cy * n[1] + cz * n[2] < 0) { const t = p2; p2 = p3; p3 = t; }
      const nb = [Math.round(n[0] * 127), Math.round(n[1] * 127), Math.round(n[2] * 127)];
      const along = p => (p[0] * -n[2] + p[2] * n[0]);
      const a = put(p1[0], p1[2], p1[1], nb[0], nb[1], nb[2], along(p1), st, H, yr), b = put(p2[0], p2[2], p2[1], nb[0], nb[1], nb[2], along(p2), st, H, yr), c = put(p3[0], p3[2], p3[1], nb[0], nb[1], nb[2], along(p3), st, H, yr);
      tri(a, b, c);
    };
    const nrm3 = v => { const l = Math.hypot(v[0], v[1], v[2]) || 1; return [v[0] / l, v[1] / l, v[2] / l]; };
    // a flat quad a b c d (in order round it) facing along n, with its own height value per corner
    const quad = (P, yr, n, st, H, us) => {

    // the smallest rectangle around a footprint (tested along each edge's direction)
    function obb(f) {
      let best = null; const k = f.length / 2;
      for (let i = 0; i < k; i++) {
        const j = (i + 1) % k, dx = f[j * 2] - f[i * 2], dz = f[j * 2 + 1] - f[i * 2 + 1], L = Math.hypot(dx, dz); if (L < 0.5) continue;
        const ux = dx / L, uz = dz / L; let a0 = 1e9, a1 = -1e9, b0 = 1e9, b1 = -1e9;
        for (let m = 0; m < k; m++) { const x = f[m * 2], z = f[m * 2 + 1], a = x * ux + z * uz, b = -x * uz + z * ux; a0 = Math.min(a0, a); a1 = Math.max(a1, a); b0 = Math.min(b0, b); b1 = Math.max(b1, b); }
        const ar = (a1 - a0) * (b1 - b0);
        if (!best || ar < best.ar) best = { ar, ux, uz, a0, a1, b0, b1 };
      }
      if (!best) return null;
      let { ux, uz, a0, a1, b0, b1 } = best;
      if (b1 - b0 > a1 - a0) { const t = ux; ux = -uz; uz = t; [a0, a1, b0, b1] = [b0, b1, -a1, -a0]; }   // u along the long side
      const ca = (a0 + a1) / 2, cb = (b0 + b1) / 2;
      return { cx: ca * ux - cb * uz, cz: ca * uz + cb * ux, ux, uz, l: a1 - a0, w: b1 - b0, fill: Math.abs(ringArea(f)) / Math.max(best.ar, 1e-6) };
    }

    function pitchedRoof(o, top, seed, H, gable) {

      const side = s => nrm3([Vd[0] * s * rh, W2, Vd[2] * s * rh]);
      const rl = gable ? L2 : Math.max(L2 - W2, 0);
      for (const s of [-1, 1]) {

        face(a, b, c, n, rs, H, H); if (rl > 0) face(a, c, d, n, rs, H, H);
        if (gable) {                                                  // the triangular end wall under the roof
          const e = s * o.l / 2, wn = [U[0] * s, 0, U[2] * s];
          face(P(e, -o.w / 2, top), P(e, o.w / 2, top), P(e, 0, R - 0.25), wn, seed, H, H + 2);
        } else {

          face(P(s * L2, -W2, eave), P(s * L2, W2, eave), P(s * rl, 0, R), en, rs, H, H);

      }
      return rh;
    }
