function updateControls(dt) {
  if (CAR.on) { if (game.state === 'flying') carControls(dt); return; }
  const s = sim;
  let tp = (pressedAny('KeyS', 'ArrowDown') ? 1 : 0) - (pressedAny('KeyW', 'ArrowUp') ? 1 : 0);
  let tr = (pressedAny('KeyD', 'ArrowRight') ? 1 : 0) - (pressedAny('KeyA', 'ArrowLeft') ? 1 : 0);
  let ty = (keys.KeyE ? 1 : 0) - (keys.KeyQ ? 1 : 0);
  let analog = false, brake = pressedAny('KeyB', 'Space') || touchIn.brake;
  const gp = padInput(dt);
  if (gp) {
    if (gp.stick) { tr = gp.tr; tp = gp.tp; analog = true; }
    if (gp.ty) ty = gp.ty;
    if (gp.brake) brake = true;
  }
  if (touchIn.stickOn) { tr = touchIn.x; tp = touchIn.y; analog = true; }
  if (analog) { const ex = v => Math.sign(v) * Math.pow(Math.abs(v), 1.2); tr = ex(tr); tp = ex(tp); }   // gentle around the centre
  if (game.invert) tp = -tp;
  if (game.state !== 'flying') { tp = tr = ty = 0; }
  // keys move the stick gradually (about 0.7 s to full) and it springs back to the centre (neutral) when released
  // (about 1.5 s to full, with a curve so that short taps make small corrections)
  const k = (cur, tgt) => analog ? damp(cur, tgt, 12, dt) : cur + clamp(tgt - cur, -(tgt === 0 ? 3.5 : 0.67) * dt, (tgt === 0 ? 3.5 : 0.67) * dt);
  if (game.ap && (tp || tr) && game.state === 'flying') { game.ap = false; toast('操縦したのでオートパイロットを解除しました', 1800); }
  const cv = v => Math.sign(v) * Math.pow(Math.abs(v), 1.3);
  if (analog) { s.pKey = tp; s.rKey = tr; s.pIn = k(s.pIn, tp); s.rIn = k(s.rIn, tr); }
  else { s.pKey = k(s.pKey || 0, tp); s.rKey = k(s.rKey || 0, tr); s.pIn = cv(s.pKey); s.rIn = cv(s.rKey); }
  if (game.levelT > 0) game.levelT -= dt;
  s.rud = s.rud + clamp(ty - s.rud, -3 * dt, 3 * dt);
  if (game.state === 'flying') {
    if (pressedAny('KeyR', 'PageUp')) s.thr = clamp(s.thr + 0.45 * dt, 0, 1);
    if (pressedAny('KeyF', 'PageDown')) s.thr = clamp(s.thr - 0.45 * dt, 0, 1);
  }
  s.brake = brake || (SYS.pbrake && s.onGround);
  if (game.touch) $('thr-fill').style.height = (s.thr * 100).toFixed(1) + '%';
}

function emit(p, v, o) {
  if (particles.length >= PMAX) particles.shift();
  particles.push({ p: p.slice(), v: v.slice(), age: 0, life: o.life, size: o.size, grow: o.grow || 0, col: o.col, a: o.a, add: !!o.add, drag: o.drag || 0.5, rise: o.rise || 0, rot: Math.random() * 6.28 });
}
let smokeAcc = 0;
function updateParticles(dt) {
  for (let i = particles.length - 1; i >= 0; i--) {
    const q = particles[i];
    q.age += dt;
    if (q.age >= q.life) { particles.splice(i, 1); continue; }
    const k = Math.exp(-q.drag * dt);
    q.v = [q.v[0] * k, q.v[1] * k + q.rise * dt, q.v[2] * k];
    q.p = vmadd(q.p, q.v, dt);
    q.size += q.grow * dt;
  }
  const s = sim;
  if (game.state === 'flying' && s.smoke && s.tas > 10) {
    smokeAcc += dt;
    while (smokeAcc > 0.025) {
      smokeAcc -= 0.025;
      const tail = vadd(s.pos, qrot(s.q, [0, 0.1, 4.1]));
      emit(tail, vadd(vscale(s.vel, 0.05), [(Math.random() - 0.5) * 0.6, (Math.random() - 0.5) * 0.6, (Math.random() - 0.5) * 0.6]),
        { life: 9, size: 1.3, grow: 2.2, col: [0.96, 0.96, 0.97], a: 0.6, drag: 0.35 });
    }
  }
  if (s.crashed && game.state !== 'menu' && Math.random() < dt * 25) {
    const wreck = s.pos;
    emit(vadd(wreck, [(Math.random() - 0.5) * 3, 1, (Math.random() - 0.5) * 3]), [(Math.random() - 0.5) * 2, 4 + Math.random() * 3, (Math.random() - 0.5) * 2],
      { life: 9, size: 3, grow: 4, col: [0.18, 0.17, 0.17], a: 0.55, drag: 0.4, rise: 0.5 });
    if (Math.random() < 0.5) emit(vadd(wreck, [(Math.random() - 0.5) * 2, 0.8, (Math.random() - 0.5) * 2]), [0, 3, 0],
      { life: 0.8, size: 2.2, grow: 2, col: [1.0, 0.5, 0.12], a: 0.3, add: true });
  }
}
function explode(p, water) {
  for (let i = 0; i < 70; i++) {
    const d = vnorm([Math.random() - 0.5, Math.random() * 0.8, Math.random() - 0.5]), sp = 6 + Math.random() * 22;
    if (water) emit(p, vscale(d, sp * 0.8), { life: 2 + Math.random() * 2, size: 2, grow: 5, col: [0.92, 0.95, 0.97], a: 0.8, drag: 1.2, rise: -6 });
    else {
      if (i % 2) emit(p, vscale(d, sp), { life: 0.7 + Math.random() * 0.6, size: 2.5, grow: 7, col: [1.0, 0.42 + Math.random() * 0.25, 0.08], a: 0.28, add: true, drag: 2.5 });

    }
  }
}

function crash(reason) {
  const s = sim;
  if (s.crashed) return;
  s.crashed = true; s.crashReason = reason;
  s.vel = [0, 0, 0]; s.w = [0, 0, 0];
  const water = terrainHeight(s.pos[0], s.pos[2]) < 0;
  if (water) s.pos[1] = 0.5;
  explode(s.pos, water);
  audio.crash();
  game.state = 'crashed';
  cam.prevMode = cam.mode; cam.mode = 'orbit'; cam.autoOrbit = true; cam.dist = 45; cam.orbitPitch = 0.35;
  cam.orbitYaw = Math.atan2(cam.eye[0] - s.pos[0], cam.eye[2] - s.pos[2]);
  game.resultTimer = 2.2;
  game.pendingResult = {
    eyebrow: 'CRASH', title: '墜落しました', rows: [['原因', reason], ['飛行時間', fmtTime(game.t)]],
  };
  if (SCHOOL.active) { game.pendingResult = null; schoolCrash(reason); }
}

function finishLanding() {

  const stopOk = onPaved(s.pos[0], s.pos[2]) === 'rwy' ? 10 : 0;
  const total = ft.paved ? sink + zone + centre + stopOk : Math.round((sink + centre) * 0.3);
  const grade = total >= 90 ? 'S' : total >= 75 ? 'A' : total >= 55 ? 'B' : 'C';
  const prev = game.best.landing || 0;
  if (total > prev) { game.best.landing = total; saveBest(); }
  showResult({
    eyebrow: 'LANDING CHALLENGE', title: `着陸 評価 ${grade}`,
    rows: [
      ['接地時の降下率', ft.fpm + ' fpm', sink + ' / 40'],
      ['接地点（目標からの距離）', (ft.u < -RWY.len / 2 ? '滑走路手前' : (d >= 0 ? '+' : '') + Math.round(d) + ' m'), zone + ' / 30'],
      ['センターラインからのずれ', Math.abs(ft.v).toFixed(1) + ' m', centre + ' / 20'],
      ['滑走路上で停止', stopOk ? 'はい' : 'いいえ', stopOk + ' / 10'],
      ['合計', '', total + ' 点', true],
      ['ベスト', '', Math.max(total, prev) + ' 点'],
    ],
  });
}
function updateRings(prevPos) {
  const ring = course.rings[game.ringIdx];
  if (!ring || game.ringDone) return;
  const a = vdot(vsub(prevPos, ring.p), ring.n), b = vdot(vsub(sim.pos, ring.p), ring.n);
  if ((a < 0 && b >= 0) || (a > 0 && b <= 0)) {   // either direction counts
    const hit = vlerp(prevPos, sim.pos, a / (a - b));
    const off = vsub(hit, ring.p), radial = vlen(vsub(off, vscale(ring.n, vdot(off, ring.n))));
    if (radial <= course.radius) {
      game.ringIdx++;
      audio.ring();
      if (game.tour) tourPassed(game.ringIdx - 1);
      if (game.ringIdx >= course.rings.length && game.tour) {
        game.ringDone = true;
        setTimeout(() => showResult({ eyebrow: 'SIGHTSEEING', title: '観光フライト完了', rows: [['めぐった名所', '', course.tour.map(l => l[0]).join('・')], ['飛行時間', '', fmtTime(game.ringTime)]] }), 6500);
      } else if (game.ringIdx >= course.rings.length) {
        game.ringDone = true;
        const t = game.ringTime, prev = game.best.rings;
        if (!prev || t < prev) { game.best.rings = t; saveBest(); }
        showResult({
          eyebrow: 'RING CHALLENGE', title: 'コースクリア',
          rows: [['タイム', '', fmtTime(t), true], ['ベスト', '', fmtTime(Math.min(t, prev || t))], ['通過リング', '', course.rings.length + ' / ' + course.rings.length]],
        });
      } else if (!game.tour) toast(`リング ${game.ringIdx} / ${course.rings.length} 通過`, 1500);
    }
  }
}

// metres of the car are the sim's own (a photogrammetry mesh is a melted blanket at street level)
const ZONE = { on: false, v: [0, 0, 0, 0], side: 0, r0: 240, r1: 340, g: [0, 0, 0, 0], clip: 2.2, gAt: null };
// the ground round the car as a curved surface (h = a + bx + cz + dx² + exz + fz², least squares over rings of
// the map's heights), so the cut under Google's mesh follows valleys and hillsides
function zoneGroundFit() {
  const cx = sim.pos[0], cz = sim.pos[2];
  if (ZONE.gAt && Math.hypot(cx - ZONE.gAt[0], cz - ZONE.gAt[1]) < 10 && R.time - ZONE.gAt[2] < 2) return;
  const A = Array.from({ length: 6 }, () => new Float64Array(7));
  for (const r of [0, 25, 55, 95, 145, 205, 270]) for (let k = 0; k < (r ? 16 : 1); k++) {
    const a = k / 16 * Math.PI * 2, x = Math.cos(a) * r, z = Math.sin(a) * r, y = Math.max(jwHeight(cx + x, cz + z), 0);
    const X = x / 100, Z = z / 100, f = [1, X, Z, X * X, X * Z, Z * Z], w = r < 150 ? 1 : 0.5;   // (the near ground counts most)
    for (let i = 0; i < 6; i++) { for (let j = 0; j < 6; j++) A[i][j] += w * f[i] * f[j]; A[i][6] += w * f[i] * y; }
  }
  for (let i = 0; i < 6; i++) {                                     // Gauss-Jordan
    let p = i; for (let r = i + 1; r < 6; r++) if (Math.abs(A[r][i]) > Math.abs(A[p][i])) p = r;
    [A[i], A[p]] = [A[p], A[i]]; const d = A[i][i] || 1e-9;
    for (let j = i; j < 7; j++) A[i][j] /= d;
    for (let r = 0; r < 6; r++) if (r !== i) { const m = A[r][i]; for (let j = i; j < 7; j++) A[r][j] -= m * A[i][j]; }
  }
  const c = A.map(r => r[6]);
  ZONE.g = [c[0], c[1] / 100, c[2] / 100, ZONE.clip];
  ZONE.g2 = [c[3] / 1e4, c[4] / 1e4, c[5] / 1e4, 0];
  ZONE.gAt = [cx, cz, R.time];
}
// is PLATEAU round the car just plain boxes (no photo textures: LOD1)? Then the sim's own buildings, with
// their windows, doors and roofs, look better at street level (checked once a second)
function platPlainNear() {
  if (typeof PLAT === 'undefined' || !PLAT.active || !PLAT.active.length || !CAR.on) return false;
  if (ZONE.plainT && R.time - ZONE.plainT < 1) return ZONE.plain;
  const [la, lo] = worldToGeo(sim.pos[0], sim.pos[2]), dLa = 400 / 111320, dLo = 400 / (111320 * Math.cos(la * DEG));
  let tex = 0, plain = 0;
  for (const it of PLAT.active) {
    const n = it && it.mesh ? it : it && it[0]; if (!n || !n.mesh || n.mesh.empty || n.s === undefined) continue;
    if (n.n < la - dLa || n.s > la + dLa || n.e < lo - dLo || n.w > lo + dLo) continue;
    if (n.mesh.tex) tex += n.mesh.count; else plain += n.mesh.count;
  }
  ZONE.plainT = R.time; ZONE.plain = plain > 0;            // (any plain boxes near: those are swapped, block by block)
  return ZONE.plain;
}
function zoneUpdate() {
  ZONE.on = !!(CAR.on && GOOG.covering && JW.online && typeof CITY !== 'undefined' && CITY.ready && ZONE.enabled !== false);
  // the same near zone over PLATEAU's city models: their photo textures are a blur at street level, so round
  // the car the map's own buildings (with their modelled windows, shops and balconies) stand in
  // (off by default: PLATEAU's own buildings are real ones, better than the sim's even at street level)
  ZONE.plat = !!((ZONE.platEnabled === true || platPlainNear()) && !ZONE.on && CAR.on && !GOOG.covering && JW.online && typeof CITY !== 'undefined' && CITY.ready &&
    typeof PLAT !== 'undefined' && PLAT.active.length && ZONE.enabled !== false);

  ZONE.v = ZONE.on || ZONE.plat ? [sim.pos[0], sim.pos[2], ZONE.r0, ZONE.r1] : [0, 0, 0, 0];

}

function shadowUniforms(pr) {
  const s = sim, u = pr.u;
  let h = Math.max(terrainHeight(s.pos[0], s.pos[2]), 0), c = [s.pos[0], s.pos[2]];
  for (let i = 0; i < 2; i++) {
    const t = (s.pos[1] - h) / ENV.sunDir[1];
    c = [s.pos[0] - ENV.sunDir[0] * t, s.pos[2] - ENV.sunDir[2] * t];
    h = Math.max(terrainHeight(c[0], c[1]), 0);
    const pv = onPaved(c[0], c[1]); if (pv) h = RWY.elev;
  }
  const above = s.pos[1] - h;
  const f = qrot(s.q, [0, 0, -1]); const fl = Math.hypot(f[0], f[2]) || 1;
  const str = s.crashed ? 0 : 0.6 * (1 - smoothstep(20, 260, above)) * clamp(ENV.sunDir[1] * 3, 0, 1);
  gl.uniform4f(u.uShadowA, c[0], c[1], f[0] / fl, f[2] / fl);
  gl.uniform2f(u.uShadowB, str, 0.12 + above * 0.004);
}
// dynamic resolution: when frames take too long the 3D view is drawn at a lower resolution (the HUD stays
// sharp), and it creeps back up when there is time to spare
const DYN = { k: 1, acc: 0, n: 0, t: 0 };
function dynRes(dt) {
  if (!(dt > 0) || dt > 3) return;                                    // (longer: a pause, not a frame)
  DYN.acc += dt; DYN.n++; DYN.t += dt;
  if (DYN.t < 1.5) return;
  const ft = DYN.acc / DYN.n; DYN.acc = 0; DYN.n = 0; DYN.t = 0;
  DYN.fps = ft > 0.1 ? Math.round(10 / ft) / 10 : Math.round(1 / ft);   // (under 10: to a tenth)
  if (game.state !== 'flying') return;
  if (DYN.lock) { DYN.k = DYN.lock; return; }                         // (held for a benchmark, p_dev.js)
  // (phones may drop further: a sharp half-resolution picture beats a stuttering one)
  if (ft > 1 / 40) DYN.k = Math.max(game.touch ? 0.45 : 0.55, DYN.k - (ft > 1 / 25 ? 0.12 : 0.06));
  else if (ft < 1 / 56) DYN.k = Math.min(1, DYN.k + 0.04);
}
function resize() {
  const dpr = Math.min(window.devicePixelRatio || 1, game.touch ? (CAR.on ? 1.25 : 1.5) : CAR.on ? 2 : 1.75, GOOG.covering && !CAR.on ? 1.25 : 9) * DYN.k;
  const w = Math.max(1, Math.round(glCanvas.clientWidth * dpr)), h = Math.max(1, Math.round(glCanvas.clientHeight * dpr));
  if (glCanvas.width !== w || glCanvas.height !== h) { glCanvas.width = w; glCanvas.height = h; }
  const hd = Math.min(window.devicePixelRatio || 1, 2), hc = hud.cv;
  const hw = Math.round(hc.clientWidth * hd), hh = Math.round(hc.clientHeight * hd);
  if (hc.width !== hw || hc.height !== hh) { hc.width = hw; hc.height = hh; }
  hud.dpr = hd; hud.W = hc.clientWidth; hud.H = hc.clientHeight;
}
// The scene is drawn in two depth ranges (far: 600 m .. 60 km, near: camera .. 600 m) so that a
// 24-bit depth buffer keeps full precision for both the cockpit and distant coastlines.
const DEPTH_SPLIT = 600;
function render() {
  resize();
  const W = glCanvas.width, H = glCanvas.height, asp = W / H;
  gl.viewport(0, 0, W, H);
  gl.clearColor(ENV.horizon[0], ENV.horizon[1], ENV.horizon[2], 1);
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
  let fovY = cam.fov * DEG;
  if (asp < 1.2) fovY = 2 * Math.atan(Math.tan(fovY / 2) * 1.2 / asp);   // keep horizontal view on tall screens
  const v = cam.view;
  R.eye = cam.eye; R.W = W; R.H = H; R.frame = (R.frame || 0) + 1;
  R.camR = [v[0], v[4], v[8]]; R.camU = [v[1], v[5], v[9]]; R.camF = [-v[2], -v[6], -v[10]];
  R.tanY = Math.tan(fovY / 2); R.tanX = R.tanY * asp;
  gl.activeTexture(gl.TEXTURE0); gl.bindTexture(gl.TEXTURE_2D, heightTex);
  gl.activeTexture(gl.TEXTURE1); gl.bindTexture(gl.TEXTURE_2D, normalTex);
  gl.activeTexture(gl.TEXTURE2); gl.bindTexture(gl.TEXTURE_2D, noiseTex);
  gl.activeTexture(gl.TEXTURE3); gl.bindTexture(gl.TEXTURE_2D, puffTex);
  gl.activeTexture(gl.TEXTURE4); gl.bindTexture(gl.TEXTURE_2D, dotTex);
  gl.activeTexture(gl.TEXTURE5); gl.bindTexture(gl.TEXTURE_2D, runwayTex);
  R.nodes = 0;
  if (JW.active) { const pl = frustumPlanes(M4.mul(M4.persp(fovY, asp, 1, 650000), v)); platUpdate(pl); googUpdate(pl); }
  zoneUpdate();
  sunShadowPass(); gl.viewport(0, 0, W, H);
  npUpdate();
  for (const far of [true, false]) {
    const proj = far ? M4.persp(fovY, asp, DEPTH_SPLIT, JW.active ? 650000 : 60000) : M4.persp(fovY, asp, cam.near, DEPTH_SPLIT * 1.02);
    R.VP = M4.mul(proj, v);

  }
}

function drawRings() {
  if (game.mode !== 'rings' || game.state === 'menu') return;
  setCommon(progMesh);
  const u = progMesh.u;
  gl.uniform1i(u.uLivery, 0); gl.uniform1f(u.uSpec, 0.4);
  for (let i = game.ringIdx; i < course.rings.length; i++) {
    const r = course.rings[i], next = i === game.ringIdx;
    const q = quatFromTo([0, 0, 1], r.n);
    const pulse = next ? 0.75 + 0.25 * Math.sin(R.time * 5) : 0;
    gl.uniformMatrix4fv(u.uModel, false, M4.fromQP(q, r.p, course.radius));
    gl.uniform3fv(u.uEmissive, next ? [0.85 * pulse + 0.2, 0.25, 0.8 * pulse + 0.2] : [0.35, 0.45, 0.6]);
    gl.uniform1f(u.uAlpha, next ? 0.95 : 0.35);
    gl.bindVertexArray(course.mesh.vao); gl.drawArrays(gl.TRIANGLES, 0, course.mesh.count);
  }
}
function quatFromTo(a, b) {
  const d = vdot(a, b);
  if (d < -0.9999) return qaxis([0, 1, 0], Math.PI);
  const c = vcross(a, b);
  return qnorm([c[0], c[1], c[2], 1 + d]);
}
function drawClouds() {
  const P = clouds.puffs, n = P.length, eye = R.eye;
  if (clouds.frame !== R.frame) {
  clouds.frame = R.frame;
  const order = clouds.order || (clouds.order = P.map((_, i) => i));
  const d2 = clouds.d2 || (clouds.d2 = new Float32Array(n));
  for (let i = 0; i < n; i++) { const p = P[i].p; d2[i] = (p[0] - eye[0]) ** 2 + (p[1] - eye[1]) ** 2 + (p[2] - eye[2]) ** 2; }
  order.sort((a, b) => d2[b] - d2[a]);
  const lit = vadd(vscale(ENV.sunCol, 0.78), vscale(ENV.ambSky, 0.55)), dark = vadd(vscale(ENV.ambSky, 1.05), vscale(ENV.sunCol, 0.18));
  const D = clouds.data;
  for (let k = 0; k < n; k++) {
    const pf = P[order[k]], o = k * 9, sh = pf.shade;
    D[o] = pf.p[0]; D[o + 1] = pf.p[1]; D[o + 2] = pf.p[2]; D[o + 3] = pf.size;
    D[o + 4] = lerp(dark[0], lit[0], sh); D[o + 5] = lerp(dark[1], lit[1], sh); D[o + 6] = lerp(dark[2], lit[2], sh); D[o + 7] = 0.85;
    D[o + 8] = pf.rot;
  }
  gl.bindBuffer(gl.ARRAY_BUFFER, clouds.vao.bufs[1]); gl.bufferData(gl.ARRAY_BUFFER, D, gl.DYNAMIC_DRAW);
  }
  setCommon(progSprite);
  const u = progSprite.u;
  gl.uniform3fv(u.uCamR, R.camR); gl.uniform3fv(u.uCamU, R.camU);
  gl.uniform1f(u.uNearFade, 1.3); gl.uniform1f(u.uFar, JW.active ? 90000 : 32000); gl.uniform1f(u.uMinPix, 0);
  gl.uniform1i(u.uTex, 3); gl.uniform1f(u.uSilver, 0.45); gl.uniform1i(u.uFog, 1);
  gl.depthMask(false);
  gl.bindVertexArray(clouds.vao.vao); gl.drawArraysInstanced(gl.TRIANGLE_STRIP, 0, 4, n);
}
function drawParticles() {
  if (!particles.length) return;
  const eye = R.eye;
  const alphaP = particles.filter(p => !p.add).sort((a, b) => vdist(b.p, eye) - vdist(a.p, eye));
  const addP = particles.filter(p => p.add);
  setCommon(progSprite);
  const u = progSprite.u;
  gl.uniform3fv(u.uCamR, R.camR); gl.uniform3fv(u.uCamU, R.camU);
  gl.uniform1f(u.uNearFade, 0); gl.uniform1f(u.uFar, 20000); gl.uniform1f(u.uMinPix, 0);
  gl.uniform1i(u.uTex, 3); gl.uniform1f(u.uSilver, 0);
  const lightK = vadd(vscale(ENV.sunCol, 0.7), vscale(ENV.ambSky, 0.6));
  const pass = (list, additive) => {
    if (!list.length) return;
    let k = 0;
    for (const q of list) {
      const o = k * 9, f = q.age / q.life, fade = additive ? (1 - f) : (1 - f) * smoothstep(0, 0.08, f);
      pData[o] = q.p[0]; pData[o + 1] = q.p[1]; pData[o + 2] = q.p[2]; pData[o + 3] = q.size;
      const lk = additive ? [1.1, 1.1, 1.1] : lightK;
      pData[o + 4] = q.col[0] * lk[0]; pData[o + 5] = q.col[1] * lk[1]; pData[o + 6] = q.col[2] * lk[2]; pData[o + 7] = q.a * fade;
      pData[o + 8] = q.rot; k++;
    }
    gl.bindBuffer(gl.ARRAY_BUFFER, pVAO.bufs[1]); gl.bufferData(gl.ARRAY_BUFFER, pData.subarray(0, k * 9), gl.DYNAMIC_DRAW);
    gl.uniform1i(u.uFog, additive ? 0 : 1);
    gl.blendFunc(gl.SRC_ALPHA, additive ? gl.ONE : gl.ONE_MINUS_SRC_ALPHA);
    gl.bindVertexArray(pVAO.vao); gl.drawArraysInstanced(gl.TRIANGLE_STRIP, 0, 4, k);
  };
  gl.depthMask(false);
  pass(alphaP, false); pass(addP, true);
  gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);
}
function drawPapi() {
  const lights = papiLights(), eye = R.eye, D = pVAO.papiData;
  let k = 0;
  for (const l of lights) {
    const [lu] = worldToRwy(l.p[0], l.p[2]), [eu] = worldToRwy(eye[0], eye[2]);
    const facing = l.end ? eu > lu : eu < lu;
    if (!facing) continue;
    const hd = Math.hypot(eye[0] - l.p[0], eye[2] - l.p[2]);
    if (hd > 9000) continue;
    const ang = Math.atan2(eye[1] - l.p[1], hd) / DEG;
    const white = ang > l.ang, o = k * 9;
    D[o] = l.p[0]; D[o + 1] = l.p[1]; D[o + 2] = l.p[2]; D[o + 3] = -0.9;
    D[o + 4] = white ? 1.6 : 1.8; D[o + 5] = white ? 1.6 : 0.15; D[o + 6] = white ? 1.5 : 0.1; D[o + 7] = 1;
    D[o + 8] = 0; k++;
  }
  if (!k) return;
  gl.bindBuffer(gl.ARRAY_BUFFER, pVAO.papi.bufs[1]); gl.bufferData(gl.ARRAY_BUFFER, D, gl.DYNAMIC_DRAW);
  setCommon(progSprite);
  const u = progSprite.u;
  gl.uniform3fv(u.uCamR, R.camR); gl.uniform3fv(u.uCamU, R.camU);
  gl.uniform1f(u.uNearFade, 0); gl.uniform1f(u.uFar, 12000); gl.uniform1f(u.uMinPix, 2.2 * R.tanY / R.H * 2);
  gl.uniform1i(u.uTex, 4); gl.uniform1f(u.uSilver, 0); gl.uniform1i(u.uFog, 0);
  gl.blendFunc(gl.SRC_ALPHA, gl.ONE); gl.depthMask(false);
  gl.bindVertexArray(pVAO.papi.vao); gl.drawArraysInstanced(gl.TRIANGLE_STRIP, 0, 4, k);
  gl.blendFunc(gl.SRC_ALPHA, gl.ONE_MINUS_SRC_ALPHA);
}

function rr(g, x, y, w, h, r) { g.beginPath(); g.moveTo(x + r, y); g.arcTo(x + w, y, x + w, y + h, r); g.arcTo(x + w, y + h, x, y + h, r); g.arcTo(x, y + h, x, y, r); g.arcTo(x, y, x + w, y, r); g.closePath(); }
function project(p) {
  if (JW.active) { const dx = p[0] - R.eye[0], dz = p[2] - R.eye[2]; p = [p[0], p[1] - (dx * dx + dz * dz) / (2 * R_EARTH), p[2]]; }
  const m = R.VP, x = m[0] * p[0] + m[4] * p[1] + m[8] * p[2] + m[12], y = m[1] * p[0] + m[5] * p[1] + m[9] * p[2] + m[13];
  const w = m[3] * p[0] + m[7] * p[1] + m[11] * p[2] + m[15];
  return { x: (x / w * 0.5 + 0.5) * hud.W, y: (1 - (y / w * 0.5 + 0.5)) * hud.H, w, nx: x / w, ny: y / w };
}
function attitude() {
  const f = qrot(sim.q, [0, 0, -1]), r = qrot(sim.q, [1, 0, 0]), up = qrot(sim.q, [0, 1, 0]);
  return {
    pitch: Math.asin(clamp(f[1], -1, 1)) / DEG,
    bank: Math.atan2(-r[1], up[1]) / DEG,
    hdg: (Math.atan2(f[0], -f[2]) / DEG + 360) % 360,
  };
}

function drawRingTarget(g, W, H) {
  const ring = course.rings[game.ringIdx];

    g.beginPath();
    for (const [sx, sy] of [[-1, -1], [1, -1], [1, 1], [-1, 1]]) {

function tape(g, x, y, w, h, val, ppu, labelStep, tickStep, fmt, bands, side, groundVal) {
  const cy = y + h / 2;
  g.save(); rr(g, x, y, w, h, 4); g.fillStyle = 'rgba(0,0,0,0.42)'; g.fill(); g.clip();
  if (groundVal !== undefined) {
    const gy = cy - (groundVal - val) * ppu;
    if (gy < y + h) { g.fillStyle = 'rgba(138,96,56,0.55)'; g.fillRect(x, gy, w, y + h - gy); }
  }
  if (bands) for (const [a, b, c] of bands) {
    const ya = cy - (a - val) * ppu, yb = cy - (b - val) * ppu;
    g.fillStyle = c; g.fillRect(x + w - (c === '#ffffff' ? 9 : 5), yb, c === '#ffffff' ? 3 : 5, ya - yb);
  }
  g.strokeStyle = '#fff'; g.fillStyle = '#fff'; g.lineWidth = 1.2; g.font = `12px ${MONO}`; g.textBaseline = 'middle';
  const range = h / 2 / ppu + tickStep;
  for (let v = Math.floor((val - range) / tickStep) * tickStep; v <= val + range; v += tickStep) {
    const ty = cy - (v - val) * ppu;
    const major = v % labelStep === 0;
    if (side === 'left') {
      g.beginPath(); g.moveTo(x + w, ty); g.lineTo(x + w - (major ? 10 : 6), ty); g.stroke();
      if (major) { g.textAlign = 'right'; g.fillText(fmt(v), x + w - 13, ty); }
    } else {
      g.beginPath(); g.moveTo(x, ty); g.lineTo(x + (major ? 10 : 6), ty); g.stroke();
      if (major) { g.textAlign = 'left'; g.fillText(fmt(v), x + 13, ty); }
    }
  }
  g.restore();
  g.fillStyle = '#000'; g.strokeStyle = '#fff'; g.lineWidth = 1.3;
  const bw = w + 2, bx = side === 'left' ? x - 1 : x - 1;
  g.beginPath();
  if (side === 'left') { g.moveTo(bx, cy - 12); g.lineTo(bx + bw - 6, cy - 12); g.lineTo(bx + bw + 2, cy); g.lineTo(bx + bw - 6, cy + 12); g.lineTo(bx, cy + 12); }
  else { g.moveTo(bx + bw, cy - 12); g.lineTo(bx + 6, cy - 12); g.lineTo(bx - 2, cy); g.lineTo(bx + 6, cy + 12); g.lineTo(bx + bw, cy + 12); }
  g.closePath(); g.fill(); g.stroke();
  g.fillStyle = '#fff'; g.font = `700 14px ${MONO}`; g.textAlign = 'center'; g.textBaseline = 'middle';
  g.fillText(String(Math.round(val)), x + w / 2 + (side === 'left' ? -2 : 3), cy + 0.5);
}

// "targets" panel: throttle and pitch for the phase of flight the aircraft is in (level, climb or descent) at the current bank
function trimTarget() {
  const T = TRIM[ACFT.id]; if (!T) return null;
  const s = sim, a = attitude(), b = Math.abs(a.bank), fpm = s.vs * 196.85, kt = s.ias * 1.94384;
  const slow = s.flapIdx > 0 || (s.gearDown && !ACFT.gear.fixed);
  let key, label;
  if (slow) {
    if (s.flapIdx >= ACFT.v.apprFlaps && ACFT.v.apprFlaps > 0 && fpm < -200) { key = 'a_3deg'; label = '着陸の進入 3°'; }
    else if (fpm > 250) { key = 's_c500'; label = '低速で上昇 500 fpm'; }
    else if (fpm < -250) { key = 's_d500'; label = '低速で降下 500 fpm'; }
    else { key = 's_level'; label = '低速の水平飛行'; }
  } else {
    const vMid = (T.c_level_0[0] + T.y_c500_0[0]) / 2;
    if (fpm > 750) { key = T.y_c1000_0[7] ? 'y_c1000' : 'y_full'; label = key === 'y_full' ? '全開で上昇' : '上昇 1,000 fpm'; }
    else if (fpm > 250) { key = kt > vMid && T.c_c500_0[7] ? 'c_c500' : 'y_c500'; label = '上昇 500 fpm'; }
    else if (fpm < -750) { key = T.c_d1000_0[7] ? 'c_d1000' : 'c_d500'; label = key === 'c_d1000' ? '降下 1,000 fpm' : '降下 500 fpm'; }
    else if (fpm < -250) { key = 'c_d500'; label = '降下 500 fpm'; }
    else { key = 'c_level'; label = '水平飛行'; }
  }
  const banks = key === 'c_level' ? [0, 10, 20, 30, 45, 60] : [0, 10, 20];
  const bb = Math.min(b, banks[banks.length - 1]);
  let i = 0; while (i < banks.length - 2 && bb > banks[i + 1]) i++;
  const r0 = T[key + '_' + banks[i]], r1 = T[key + '_' + banks[i + 1]], f = clamp((bb - banks[i]) / (banks[i + 1] - banks[i]), 0, 1);
  return { key, label, v: r0[0], fpm: r0[3], thr: lerp(r0[1], r1[1], f), pitch: lerp(r0[2], r1[2], f), bank: b, slow,
    over: (slow && b > 22) || (key !== 'c_level' && b > 22), pitchNow: a.pitch, full: key === 'y_full' };
}

  $('tut-next').hidden = !st.next;
  $('tut-next').textContent = game.touch ? '次へ' : '次へ（Enter）';
  $('tut-bar').parentElement.hidden = !!st.next;
  $('tut').classList.remove('ok');

function tutStart() { tut.i = 0; tut.doneT = 0; if (TUT_STEPS[0].setup) TUT_STEPS[0].setup(); tutShow(); }
function tutHide() {
  $('tut').hidden = true; document.body.classList.remove('tut-on');
  document.querySelectorAll('.pulse').forEach(e => e.classList.remove('pulse'));
}
function tutNext() { const st = tutStep(); if (st && st.next) tutAdvance(); }
function tutAdvance() {
  tut.i++; tut.doneT = 0;
  while (TUT_STEPS[tut.i] && TUT_STEPS[tut.i].skipIf && TUT_STEPS[tut.i].skipIf()) tut.i++;
  if (tut.i >= TUT_STEPS.length) {
    tutHide();
    game.best.tutorial = true; saveBest();
    showResult({ eyebrow: 'TUTORIAL', title: 'チュートリアル完了', rows: [
      ['所要時間', '', fmtTime(game.t)],
      ['次のおすすめ', '', '着陸チャレンジ'],
      ['困ったときは', '', game.touch ? '「水平」ボタン' : 'L キーで水平に戻す'],
    ] });
    return;
  }
  const st = tutStep();
  if (st.setup) st.setup();
  tutShow();
}
function tutUpdate(dt) {
  const st = tutStep();
  if (!st || st.next) return;
  if (tut.doneT > 0) {
    tut.doneT -= dt;
    if (tut.doneT <= 0) tutAdvance();
    return;
  }
  if (st.tick) st.tick(dt);
  const pr = st.progress ? clamp(st.progress(), 0, 1) : 0;
  $('tut-bar').style.width = (pr * 100).toFixed(0) + '%';
  if (st.done()) {
    tut.doneT = 1.3;
    $('tut').classList.add('ok'); $('tut-bar').style.width = '100%';
    $('tut-title').textContent = '✓ ' + $('tut-title').textContent;
    audio.ring();
  }

function drawPlaceLabels(g, W, H) {
  const [la, lo] = worldToGeo(R.eye[0], R.eye[2]), alt = R.eye[1];
  const reach = clamp(25000 + alt * 12, 25000, 160000);
  const items = [];
  for (const p of JW.places) {
    if (Math.abs(p.lat - la) > reach / 90000 || Math.abs(p.lon - lo) > reach / 70000) continue;
    const d = geoDist(la, lo, p.lat, p.lon);
    if (d > reach) continue;
    if (p.k === 'c' && d > 12000 && p.p < clamp(d * 4, 50000, 900000)) continue;
    items.push([p.k === 'm' ? '▲ ' + p.n : p.n, p.lat, p.lon, p.k === 'm' ? p.p : Math.max(coarseH(p.lat, p.lon) || 0, 0) + 30, p.k === 'm' ? '#f2dcb4' : '#ffffff', d]);
  }
  for (const l of LANDMARKS) {
    if (Math.abs(l[1] - la) > 1.2 || Math.abs(l[2] - lo) > 1.5) continue;
    const d = geoDist(la, lo, l[1], l[2]);
    if (d > Math.max(reach, l[3] < 0 ? 90000 : 30000)) continue;
    const top = l[3] < 0 ? Math.max(coarseH(l[1], l[2]) || 0, 0) + 60 : Math.max(rawHeightGeo(l[1], l[2]), 0) + l[3] + 25;
    items.push(['★ ' + l[0], l[1], l[2], top, '#ffd66b', d * 0.35]);
  }
  for (const ap of JW.airports) {
    if (Math.abs(ap.lat - la) > 2 || Math.abs(wrapLon(ap.lon - lo)) > 3) continue;
    const d = geoDist(la, lo, ap.lat, ap.lon);
    if (d < reach * 0.8) items.push(['✈ ' + ap.name, ap.lat, ap.lon, ap.rw[0].elev + 20, JW.dest === ap ? COL.magenta : '#9fe0ff', d * 0.5]);
  }
  items.sort((a, b) => a[5] - b[5]);
  const boxes = [];
  g.font = `700 12px ${UI}`; g.textAlign = 'center'; g.textBaseline = 'bottom';
  let n = 0;
  for (const [t, lat, lon, h, col] of items) {
    if (n > 26) break;
    const w = geoToWorld(lat, lon), pp = project([w[0], h, w[1]]);
    if (pp.w <= 0 || pp.x < 0 || pp.x > W || pp.y < 60 || pp.y > H - 40) continue;
    const tw = g.measureText(t).width + 8, bx = [pp.x - tw / 2, pp.y - 18, tw, 16];
    if (boxes.some(b => bx[0] < b[0] + b[2] && b[0] < bx[0] + bx[2] && bx[1] < b[1] + b[3] && b[1] < bx[1] + bx[3])) continue;
    boxes.push(bx); n++;
    g.fillStyle = 'rgba(0,0,0,0.55)'; g.fillText(t, pp.x + 1, pp.y - 3);
    g.fillStyle = col; g.fillText(t, pp.x, pp.y - 4);
    g.fillRect(pp.x - 1, pp.y - 3, 2, 3);
  }
}

function setupMapView() {
  const cv = $('map-cv');
  const zoomAt = (f, sx, sy) => {
    const W = cv.clientWidth, H = cv.clientHeight, z0 = mapv.zoom, z1 = clamp(z0 + f, 2, 16);
    const [cx, cy] = mapPx(mapv.lat, mapv.lon, z0), px = cx + (sx - W / 2), py = cy + (sy - H / 2);
    const [la, lo] = mapGeo(px, py, z0), [qx, qy] = mapPx(la, lo, z1);
    [mapv.lat, mapv.lon] = mapGeo(qx - (sx - W / 2), qy - (sy - H / 2), z1);
    mapv.zoom = z1; mapv.follow = false;
  };
  cv.addEventListener('wheel', e => { e.preventDefault(); zoomAt(e.deltaY < 0 ? 0.35 : -0.35, e.offsetX, e.offsetY); }, { passive: false });
  let moved = 0, pinch = null;
  cv.addEventListener('pointerdown', e => { cv.setPointerCapture(e.pointerId); mapv.pointers.set(e.pointerId, [e.offsetX, e.offsetY]); moved = 0; if (mapv.pointers.size === 2) { const [a, b] = [...mapv.pointers.values()]; pinch = Math.hypot(a[0] - b[0], a[1] - b[1]); } });
  cv.addEventListener('pointermove', e => {
    const prev = mapv.pointers.get(e.pointerId); if (!prev) return;
    const cur = [e.offsetX, e.offsetY]; mapv.pointers.set(e.pointerId, cur);
    if (mapv.pointers.size === 2) {
      const [a, b] = [...mapv.pointers.values()], d = Math.hypot(a[0] - b[0], a[1] - b[1]);
      if (pinch) zoomAt(Math.log2(d / pinch), (a[0] + b[0]) / 2, (a[1] + b[1]) / 2);
      pinch = d; moved += 10; return;
    }
    const dx = cur[0] - prev[0], dy = cur[1] - prev[1]; moved += Math.abs(dx) + Math.abs(dy);
    const [cx, cy] = mapPx(mapv.lat, mapv.lon, mapv.zoom);
    [mapv.lat, mapv.lon] = mapGeo(cx - dx, cy - dy, mapv.zoom); mapv.lon = wrapLon(mapv.lon); mapv.lat = clamp(mapv.lat, -84, 84); mapv.follow = false;
  });
  const up = e => {
    mapv.pointers.delete(e.pointerId); if (mapv.pointers.size < 2) pinch = null;

      const W = cv.clientWidth, H = cv.clientHeight, [cx, cy] = mapPx(mapv.lat, mapv.lon, mapv.zoom);

      mapInfo(best);
    }
  };
  cv.addEventListener('pointerup', up); cv.addEventListener('pointercancel', up);
  $('map-close').addEventListener('click', () => toggleMap(false));
  // search airports and cities
  const sIn = $('map-search'), sOut = $('map-results');
  const norm = t => (t || '').toLowerCase().normalize('NFKC');
  sIn.addEventListener('input', () => {
    const q = norm(sIn.value.trim());
    sOut.textContent = '';
    if (!q) { sOut.hidden = true; return; }
    const res = [];
    for (const a of JW.airports) {
      const hay = [a.name, a.icao, a.iata, a.city, COUNTRY_JA[a.cc]].map(norm);
      const sc = hay.some(h => h === q) ? 0 : hay.some(h => h && h.startsWith(q)) ? 1 : hay.some(h => h && h.includes(q)) ? 2 : 9;
      if (sc < 9) res.push([sc - (a.big ? 0.5 : 0), 'ap', a]);
    }
    for (const p of JW.places) { if (p.k !== 'c') continue; const n = norm(p.n); if (n.startsWith(q) || n.includes(q)) res.push([n === q ? 0.2 : 1.5, 'pl', p]); }
    res.sort((a, b) => a[0] - b[0]);
    for (const [, kind, it] of res.slice(0, 12)) {
      const b = document.createElement('button'); b.type = 'button';
      b.textContent = kind === 'ap' ? `✈ ${it.name}（${it.iata || it.icao}）${it.cc && it.cc !== 'JP' ? '・' + (COUNTRY_JA[it.cc] || it.cc) : ''}` : `● ${it.n}`;
      b.addEventListener('click', () => {

  });
  sIn.addEventListener('keydown', e => { e.stopPropagation(); if (e.key === 'Enter') { const f = sOut.querySelector('button'); if (f) f.click(); } if (e.key === 'Escape') { sIn.value = ''; sOut.hidden = true; sIn.blur(); } });
  $('map-me').addEventListener('click', () => { mapv.follow = true; });

// ---------- main loop ----------
let lastT = performance.now(), acc = 0;
const STEP = 1 / 120;
function frame(now) {
  const rdt = Math.max(0, (now - lastT) / 1000), dt = Math.min(0.1, rdt); lastT = now; dynRes(rdt);   // (the real frame time for the frame rate: the simulation's step is capped)
  R.time += dt;
  if (game.state === 'loading') { requestAnimationFrame(frame); return; }
  devFrameStart(now);                                                 // (the developer panel's timing, p_dev.js)
  try {
    updateControls(dt);
    if (game.state === 'flying' && CAR.on) { carStep(dt); if (JW.active) jwFrame(dt); }
    else if (game.state === 'flying') {
      const ts = game.timeScale || 1;
      acc += dt * ts; let n = 0;
      while (acc >= STEP && n < 14 * ts) {
        const prev = sim.pos.slice();
        game.t += STEP;
        physicsStep(STEP, game.t);
        if (sim.crashed) break;
        if (game.mode === 'rings' && !game.ringDone) { game.ringTime += STEP; updateRings(prev); }
        if (game.state !== 'flying') break;
        acc -= STEP; n++;
      }
      if (n >= 14 * ts) acc = 0;
      if (JW.active) jwFrame(dt * ts);
      if (JW.active && game.state === 'flying') { const hit = cityCollide(); if (hit) crash(hit); }
      if (game.state === 'flying') gameChecks(dt);
      cpFrame(dt * ts);
      if (SCHOOL.active && game.state === 'flying') schoolUpdate(dt * ts);
      if (game.state === 'flying' && game.mode === 'tutorial') tutUpdate(dt);
    } else { acc = 0; cpFrame(0); }
    if (game.state === 'crashed') {
      game.resultTimer -= dt;
      if (game.resultTimer <= 0 && game.pendingResult) { const r = game.pendingResult; game.pendingResult = null; showResult(r); }
    }
    updateParticles(dt);
    updateCamera(dt);
    if (!JW.active) updateTrees(cam.eye); else { jwUpdateClouds(); cityUpdate(); }
    rxRun();                                                          // (the workers' results, a few ms' worth: p6_city.js)
    render();
    googPerf(dt);
    drawHUD();
    if (mapv.open) drawMapView();
    audio.update(dt);
    if ((R.frame & 15) === 0) { carButton(); devHoldArea(); }
    devFrameEnd();
  } catch (err) {
    console.error(err);
    $('load-text').textContent = 'エラーが発生しました: ' + err.message;
    $('loading').hidden = false;
    return;
  }
  requestAnimationFrame(frame);
}
