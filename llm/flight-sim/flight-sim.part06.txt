async function aiAskWithKey(turns, { onText, signal }) {
  let Anthropic;
  // a failed module load is remembered per URL, so a retry asks for a slightly different one
  for (let i = 0; i < 3 && !Anthropic; i++) {
    try { Anthropic = (await import(AI_SDK + (i ? '?retry=' + i : ''))).default; }
    catch (e) { if (i === 2) throw { code: 'sdk_load', message: String(e && e.message || e) }; await new Promise(r => setTimeout(r, 900)); }
  }
  const client = new Anthropic({ apiKey: AI.key, dangerouslyAllowBrowser: true, maxRetries: 1 });
  const run = async fallback => {
    const params = { model: AI_MODEL, max_tokens: 16000, messages: turns };
    if (fallback) { params.betas = ['server-side-fallback-2026-07-01']; params.fallbacks = 'default'; }   // a declined request is re-run on another model
    const stream = client.beta.messages.stream(params, { signal });
    stream.on('text', (delta, snapshot) => onText({ text: snapshot }));
    return stream.finalMessage();
  };
  let msg;
  try {
    try { msg = await run(true); }
    catch (e) { if (e instanceof Anthropic.BadRequestError && /fallback/i.test(e.message || '')) msg = await run(false); else throw e; }
  } catch (e) {
    if (e instanceof Anthropic.APIUserAbortError || (signal && signal.aborted)) throw { code: 'cancelled' };
    if (e instanceof Anthropic.AuthenticationError) throw { code: 'bad_key', message: e.message };
    if (e instanceof Anthropic.PermissionDeniedError) throw { code: 'key_denied', message: e.message };
    if (e instanceof Anthropic.RateLimitError) throw { code: 'rate_limited', message: e.message };
    if (e instanceof Anthropic.APIConnectionError) throw { code: 'network', message: e.message };
    if (e instanceof Anthropic.APIError && e.status === 529) throw { code: 'overloaded', message: e.message };
    if (e instanceof Anthropic.BadRequestError && /credit|balance|billing/i.test(e.message || '')) throw { code: 'no_credit', message: e.message };
    throw { code: 'api_error', message: (e && e.message) || String(e) };
  }
  if (msg.stop_reason === 'refusal') throw { code: 'refused' };
  const text = msg.content.filter(b => b.type === 'text').map(b => b.text).join('');
  return { text, truncated: msg.stop_reason === 'max_tokens' };
}

function aiSearch(q, n = 3) {
  const grams = t => { const g = new Set(); t = t.toLowerCase().replace(/\s+/g, ''); for (let i = 0; i < t.length - 1; i++) g.add(t.slice(i, i + 2)); return g; };
  const qg = grams(q); if (!qg.size) return [];
  const scored = [];
  for (const [ch, s] of bookAll()) {
    const txt = bookText(s), tg = grams(s.t);
    let sc = 0; for (const g of qg) { if (txt.includes(g)) sc += 1; if (tg.has(g)) sc += 2; }
    scored.push([sc / Math.sqrt(qg.size), ch, s]);
  }
  return scored.sort((a, b) => b[0] - a[0]).filter(x => x[0] > 0.9).slice(0, n);
}

function aiEsc(t) { return t.replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c])); }
function aiMarkdown(t) {
  const out = []; let list = null;
  const inline = x => aiEsc(x).replace(/\*\*(.+?)\*\*/g, '<b>$1</b>').replace(/`([^`]+)`/g, '<code>$1</code>');
  for (const raw of t.split('\n')) {
    const line = raw.trimEnd(), ul = line.match(/^\s*[-・*]\s+(.*)/), ol = line.match(/^\s*(\d+)[.)]\s+(.*)/);
    if (ul || ol) { const tag = ul ? 'ul' : 'ol'; if (!list || list !== tag) { if (list) out.push(`</${list}>`); out.push(`<${tag}>`); list = tag; } out.push(`<li>${inline(ul ? ul[1] : ol[2])}</li>`); continue; }
    if (list) { out.push(`</${list}>`); list = null; }
    const h = line.match(/^#+\s*(.*)/);
    if (h) out.push(`<p><b>${inline(h[1])}</b></p>`); else if (line.trim()) out.push(`<p>${inline(line)}</p>`);
  }
  if (list) out.push(`</${list}>`);
  return out.join('');
}

      AI.key = k; try { localStorage.setItem('sorajima-akey', k); } catch (e) { /* ignore */ } aiStatus();
    });
    inp.addEventListener('keydown', e => { if (e.key === 'Enter') { e.preventDefault(); save.click(); } });
    row.append(inp, save); st.append(p, row, msg);
  }
}

async function aiSend() {
  const box = $('ai-in'), q = box.value.trim();
  if (!q || AI.busy) return;
  box.value = '';
  aiBubble('user', `<p>${aiEsc(q)}</p>`);
  const hits = aiSearch(q, 3);
  if (!AI.sample && !AI.key) {
    const d = aiBubble('ai', hits.length ? `<p>AI に接続できないため、教科書から関係のありそうな節を探しました。</p>${aiRefs(hits)}` : '<p>AI に接続できず、教科書にも近い節が見つかりませんでした。言い方を変えるか、教科書の検索欄を試してください。</p>');
    aiWireRefs(d); return;
  }
  // context: the page being read, or the live flight state, plus the best-matching sections
  let ctx = '';
  if (AI.from === 'book' && BK.cur) { const cur = bookAll().find(([, s]) => s.id === BK.cur); if (cur) ctx += `【いま読んでいる教科書の節 ${cur[1].n}「${cur[1].t}」】\n${aiSectionText(cur[1], 5000)}\n\n`; }
  else ctx += `【いまの状況（一時停止中のシミュレータの値）】\n${aiFlightState()}\n\n`;
  for (const [, , s] of hits) if (s.id !== BK.cur || AI.from !== 'book') ctx += `【参考：教科書 ${s.n}「${s.t}」】\n${aiSectionText(s, 2200)}\n\n`;
  const userTurn = `${ctx}【訓練生の質問】\n${q}`;
  AI.turns.push({ role: 'user', content: userTurn });
  while (AI.turns.length > 9) AI.turns.shift();
  // older turns keep only the question itself, so the context of the latest turn dominates
  const turns = AI.turns.map((t, i) => (t.role === 'user' && i < AI.turns.length - 1) ? { role: 'user', content: t.content.replace(/^[\s\S]*【訓練生の質問】\n/, '') } : t);
  while (turns.length && turns[0].role !== 'user') turns.shift();
  const d = aiBubble('ai', '<p class="ai-think">考えています…</p>');
  AI.busy = true; AI.ctl = new AbortController(); $('ai-stop').hidden = false; $('ai-send').disabled = true;
  try {
    const input = turns.map((t, i) => i === 0 ? { role: 'user', content: AI_RULES + '\n\n' + t.content } : t);
    const ask = AI.sample ? (inp, o) => AI.sample(inp, Object.assign({ cache: false }, o)) : aiAskWithKey;
    const { text, truncated } = await ask(input, {
      signal: AI.ctl.signal,
      onText: ({ text }) => { d.innerHTML = aiMarkdown(text); $('ai-log').scrollTop = $('ai-log').scrollHeight; },
    });
    AI.turns.push({ role: 'assistant', content: text });
    d.innerHTML = aiMarkdown(text) + (truncated ? '<p class="ai-note">（長くなったので途中で切れました。質問を分けてください）</p>' : '') + aiRefs(hits);
  } catch (e) {
    AI.turns.pop();
    const keep = e && e.text ? aiMarkdown(e.text) : '';
    const why = { cancelled: '止めました。', not_granted: 'AI の利用が許可されませんでした。この画面では教科書の検索だけ使えます。', sampling_disabled: 'このアカウントでは AI を使えません。', rate_limited: '質問が多すぎるか、使用量の上限に達しました。少し時間をおいてください。', bad_key: 'API キーが無効です（打ち間違い・削除済み）。キーを削除して入れ直してください。', key_denied: 'この API キーではこのモデルを使えません。', no_credit: 'API の残高（クレジット）が足りません。console.anthropic.com で確認してください。', network: 'api.anthropic.com に接続できませんでした（ネットワーク・広告ブロッカー等を確認）。', overloaded: 'いま混み合っています。少し待ってからもう一度。', sdk_load: 'AI の部品（SDK）を読み込めませんでした。ネットワークを確認してください。', api_error: 'AI からエラーが返りました。', session_expired: 'ログインし直してください。', refused: 'この質問には答えられませんでした。言い方を変えてみてください。', prompt_too_large: '質問が長すぎました。短くしてください。' }[e && e.code] || '回答の途中で問題が起きました。もう一度送ってください。';
    if (e && ['not_granted', 'sampling_disabled', 'not_declared', 'capability_disabled', 'capability_removed'].includes(e.code)) { AI.sample = null; aiStatus(); }
    const det = e && e.code === 'api_error' && e.message ? `（${aiEsc(String(e.message).slice(0, 160))}）` : '';
    d.innerHTML = (e && e.code === 'refused' ? '' : keep) + `<p class="ai-note">${why}${det}</p>` + aiRefs(hits);
  } finally {
    AI.busy = false; AI.ctl = null; $('ai-stop').hidden = true; $('ai-send').disabled = false;
    aiWireRefs(d); $('ai-log').scrollTop = $('ai-log').scrollHeight;
  }
}

function sysStep(dt) {
  const s = sim, piston = sysPiston();
  if (SYS.auto) sysAutoStep();
  if (cpAutoAvio()) SYS.avio = SYS.bat;                             // airliners: the displays come on with the battery
  // ---- fuel reaching the engine ----
  const sel = piston ? SYS.fuelSel : 3;
  const avail = sel === 1 ? SYS.fuelL > 0.01 : sel === 2 ? SYS.fuelR > 0.01 : sel === 3 ? SYS.fuelL + SYS.fuelR > 0.02 : false;
  SYS.starvT = avail ? 0 : SYS.starvT + dt;
  const fuelOK = SYS.starvT < 6;                                     // what is left in the lines
  // ---- electrical ----
  const gen = SYS.run && SYS.alt && (piston ? s.rpm > 500 : SYS.spin > 0.9);
  let load = 0;
  if (SYS.bat || gen) {
    load = 3 + (SYS.avio ? 7 : 0) + (SYS.beacon ? 2 : 0) + (SYS.nav ? 3 : 0) + (SYS.strobe ? 3 : 0) + (SYS.land ? 8 : 0) + (SYS.taxi ? 6 : 0) + (SYS.pump ? 4 : 0);
    if (SYS.crank || (SYS.start && !SYS.run)) load += piston ? 140 : 220;
  }
  const capAs = (piston ? 35 : 42) * 3600;
  if (SYS.bat && gen) { const ch = Math.min(30, (1 - SYS.batQ) * 60 + 1.5); SYS.batQ = Math.min(1, SYS.batQ + ch * dt / capAs); SYS.amps = ch; SYS.volts = 28.2; }
  else if (SYS.bat) { SYS.batQ = Math.max(0, SYS.batQ - load * dt / capAs); SYS.amps = -load; SYS.volts = Math.max(0, 23 + 2 * SYS.batQ - load * 0.012) * (SYS.batQ > 0.03 ? 1 : 0.3); }
  else { SYS.amps = 0; SYS.volts = 0; }
  SYS.gen = gen;
  SYS.bus = SYS.bat && SYS.volts > 18;
  // ---- engine ----
  if (piston) {
    SYS.crank = SYS.startHeld && SYS.bus && SYS.volts > 19;
    if (SYS.startHeld && !SYS.bus && dt > 0) sysMsg('スターターが回りません：バッテリー（BAT）を ON に');
    const mixOK = SYS.mix > 0.1;
    // mixture: the fuel/air ratio for a knob position grows as the air thins; best power is a little rich of peak EGT
    const sigma = clamp((s.rho || 1.225) / 1.225, 0.3, 1.3);
    if (game.autoMix && SYS.mix > 0.15) SYS.mix = clamp(Math.sqrt(sigma) * 1.02, 0.2, 1);
    const r = SYS.mix / Math.sqrt(sigma);
    let pf = r >= 1 ? 1 - 2.7 * (r - 1) ** 2 : 1 - 3.2 * (1 - r) ** 2;
    const oneMag = SYS.mags === 1 || SYS.mags === 2;                    // one magneto: slower burn, the rpm drop you check at the run-up
    if (oneMag) pf *= 0.965;
    SYS.rpmK = oneMag ? 0.955 : 1;
    SYS.pf = clamp(pf, 0.3, 1);
    SYS.egt = SYS.run ? (1150 + 330 * Math.exp(-(((r - 0.93) / 0.2) ** 2))) * (0.62 + 0.38 * s.thr) : Math.max(0, SYS.egt - dt * 25);
    if (SYS.run) {
      if (SYS.mags === 0) sysStop('マグネトー（キー）が OFF');
      else if (!mixOK) sysStop('ミクスチャーがカットオフ（燃料が止まった）');
      else if (!fuelOK) sysStop('燃料が来ない（燃料セレクター・残量を確認）');
      else if (r < 0.62) sysStop('混合気が薄すぎる（ミクスチャーを濃く）');
    } else {
      const ready = SYS.mags > 0 && mixOK && fuelOK && r > 0.66;
      SYS.crankT = SYS.crank ? SYS.crankT + dt : 0;
      if (ready && SYS.crank && SYS.crankT > 1.3) { SYS.run = true; SYS.spin = 1; if (game.state === 'flying') toast('エンジン始動！キーを離して BOTH に（始動ボタンを離す）', 2600); }
      else if (ready && s.rpm > 700) SYS.run = true;                    // a windmilling propeller restarts it in flight
      else if (SYS.crank && SYS.crankT > 4 && dt > 0) sysMsg(SYS.mags === 0 ? 'かかりません：キー（マグネトー）を BOTH に' : !mixOK ? 'かかりません：ミクスチャーを RICH（濃く）に' : !fuelOK ? 'かかりません：燃料セレクターを BOTH に' : 'かかりません');
    }
    SYS.flow = SYS.run ? (0.25 * (s.shaftP || 0) / 1000 * Math.max(1, r) + 2 * ACFT.phys.engine.P / 96000) : 0;   // kg/h
  } else {
    // turbine: the starter spins the core to ~25 %, fuel in above ~18 % lights it, then it accelerates to idle
    const starter = SYS.start && SYS.bus && SYS.volts > 20;
    if (SYS.start && !SYS.bus && dt > 0) sysMsg('スタートできません：バッテリー（BAT）を ON に');
    if (SYS.run) {
      if (!SYS.fuelOn) sysStop('燃料レバーが CUTOFF');
      else if (!fuelOK) sysStop('燃料切れ');
      SYS.spin = 1; if (SYS.start) SYS.start = false;
    } else {
      const lit = SYS.fuelOn && fuelOK && SYS.spin > 0.3 && (starter || SYS.spin > 0.45 || s.tas > 120);
      if (lit) { SYS.spin = Math.min(1, SYS.spin + dt / 11); if (SYS.spin >= 1) { SYS.run = true; SYS.start = false; if (game.state === 'flying') toast('エンジン始動完了（回転が安定）', 2200); } }
      else if (starter) SYS.spin = Math.min(0.42, SYS.spin + dt / 9);
      else SYS.spin = Math.max(clamp(s.tas / 400, 0, 0.35), SYS.spin - dt / 14);
    }
    SYS.pf = 1;
    const T = Math.max(sim.thrust || 0, 0);
    SYS.flow = !SYS.run ? (SYS.fuelOn && SYS.spin > 0.3 ? 40 : 0) * (isAirliner() ? 3 : 1)
      : AC.engine.type === 'jet' ? T * (s.ab ? 0.2 : TSFC[ACFT.id] || 0.058) + 150 : ACFT.sys && ACFT.sys.psfc ? ACFT.sys.psfc * ACFT.phys.engine.P / 1000 * (0.3 + 0.7 * s.thr) : 0.33 * ACFT.phys.engine.P / 1000 * (0.12 + 0.88 * s.thr) * 2;
    SYS.egt = SYS.run ? 520 + 330 * s.thr : SYS.spin * 400;
  }
  // ---- burn ----
  const burn = SYS.flow * dt / 3600;
  if (burn > 0) {
    if (sel === 1) SYS.fuelL = Math.max(0, SYS.fuelL - burn);
    else if (sel === 2) SYS.fuelR = Math.max(0, SYS.fuelR - burn);
    else { const l = SYS.fuelL > 0, rr_ = SYS.fuelR > 0, k = l && rr_ ? 0.5 : 1; if (l) SYS.fuelL = Math.max(0, SYS.fuelL - burn * k); if (rr_) SYS.fuelR = Math.max(0, SYS.fuelR - burn * k); }
  }
  if (SYS.pbrake && !s.onGround) SYS.pbrake = false;
  // the aircraft gets lighter as it burns fuel
  if (AC && ACFT.massNow) AC.mass = acOEW(ACFT) + acLoad(ACFT, ACFT.custom !== false).payload + SYS.fuelL + SYS.fuelR;
}
// automatic start (Ctrl+E): flips the switches in order, like a pilot running the checklist
function sysAutoStart() {
  if (SYS.run) { toast('エンジンはもう回っています', 1400); return; }
  SYS.auto = true; SYS.autoT = R.time; toast('自動始動：チェックリストの順にスイッチを操作します', 2200);
}
function sysAutoStep() {
  const t = R.time - SYS.autoT, s = sim;
  const seq = sysPiston()
    ? [() => SYS.bat = true, () => SYS.beacon = true, () => { SYS.fuelSel = 3; if (ACFT.id !== 'trainer') SYS.pump = true; }, () => { SYS.mix = 1; s.thr = Math.max(s.thr, 0.1); }, () => SYS.mags = 3, () => SYS.startHeld = true]
    : [() => SYS.bat = true, () => SYS.beacon = true, () => SYS.pump = true, () => s.thr = 0, () => SYS.start = true];
  const i = Math.min(Math.floor(t / 0.6), seq.length - 1); for (let j = 0; j <= i; j++) seq[j]();
  if (!sysPiston() && SYS.spin > 0.32) SYS.fuelOn = true;
  if (SYS.run) {
    SYS.startHeld = false; SYS.alt = true; SYS.avio = true; SYS.nav = true; SYS.auto = false;
    if (sysPiston()) s.thr = Math.min(s.thr, 0.1);
  } else if (t > 25) { SYS.auto = false; SYS.startHeld = false; toast('自動始動できませんでした（燃料・バッテリーを確認）', 2600); }
}

// navigation display: heading-up arc with the route to the destination (Airbus ARC / Boeing MAP / G1000 map)
function drawND(g, x, y, w, h, style) {
  const s = sim, att = attitude(), ab = style === 'airbus', gm = style === 'g1000';
  const cx = x + w / 2, cy = y + h * 0.84, R = h * 0.72, hdg = att.hdg;
  const trk = Math.atan2(s.vel[0], -s.vel[2]) / DEG, gs = Math.hypot(s.vel[0], s.vel[2]) * 1.94384;
  g.save(); rr(g, x, y, w, h, 3); g.clip();
  const range = gm ? 10 : 40;                                      // NM at the arc
  // compass arc
  g.strokeStyle = '#e8ecf0'; g.lineWidth = 1.5; g.beginPath(); g.arc(cx, cy, R, Math.PI + 0.25, -0.25); g.stroke();
  for (let d = -70; d <= 70; d += 5) {
    const hd = Math.round(hdg / 5) * 5 + d, a = (hd - hdg) * DEG; if (Math.abs(a) > 1.3) continue;
    const px = cx + Math.sin(a) * R, py = cy - Math.cos(a) * R, L = hd % 10 === 0 ? 10 : 5;
    g.beginPath(); g.moveTo(px, py); g.lineTo(cx + Math.sin(a) * (R - L), cy - Math.cos(a) * (R - L)); g.stroke();
    if (((hd % 30) + 360) % 30 === 0) cpText(g, String(((hd / 10) % 36 + 36) % 36), cx + Math.sin(a) * (R - 20), cy - Math.cos(a) * (R - 20), 11, '#e8ecf0');
  }
  g.setLineDash([3, 5]); g.strokeStyle = 'rgba(232,236,240,0.35)'; g.beginPath(); g.arc(cx, cy, R / 2, Math.PI + 0.25, -0.25); g.stroke(); g.setLineDash([]);
  cpText(g, String(range / 2), cx - R / 2 * 0.92 - 8, cy - R / 2 * 0.4, 9, '#5fd4ff');
  // heading pointer and track line
  g.fillStyle = ab ? '#ffe14a' : '#f4f4f4'; g.beginPath(); g.moveTo(cx, cy - R - 2); g.lineTo(cx - 6, cy - R - 12); g.lineTo(cx + 6, cy - R - 12); g.fill();
  const ta = (trk - hdg) * DEG; g.strokeStyle = '#58e08c'; g.lineWidth = 1.5; g.beginPath(); g.moveTo(cx, cy); g.lineTo(cx + Math.sin(ta) * R, cy - Math.cos(ta) * R); g.stroke();
  // destination (magenta route)
  let dtxt = '';
  if (typeof JW !== 'undefined' && JW.active && JW.dest) {
    const [la, lo] = planeGeo(), brg = geoBearing(la, lo, JW.dest.lat, JW.dest.lon), dist = geoDist(la, lo, JW.dest.lat, JW.dest.lon) / 1852;
    const a = (brg - hdg) * DEG, rr_ = Math.min(dist / range, 1.05) * R;
    g.strokeStyle = '#e36bdc'; g.lineWidth = 2; g.beginPath(); g.moveTo(cx, cy); g.lineTo(cx + Math.sin(a) * rr_, cy - Math.cos(a) * rr_); g.stroke();
    if (dist <= range) { g.fillStyle = '#e36bdc'; g.beginPath(); g.arc(cx + Math.sin(a) * rr_, cy - Math.cos(a) * rr_, 5, 0, 7); g.fill(); }
    dtxt = `${JW.dest.icao || JW.dest.name}  ${dist.toFixed(dist < 100 ? 1 : 0)} NM`;
  }
  // own aircraft
  g.strokeStyle = ab ? '#ffe14a' : '#f4f4f4'; g.lineWidth = 3; g.beginPath(); g.moveTo(cx - 14, cy); g.lineTo(cx + 14, cy); g.moveTo(cx, cy - 10); g.lineTo(cx, cy + 12); g.moveTo(cx - 5, cy + 10); g.lineTo(cx + 5, cy + 10); g.stroke();
  // readouts
  cpText(g, `GS ${Math.round(gs)}  TAS ${Math.round(s.tas * 1.94384)}`, x + 8, y + 14, 11, '#e8ecf0', 700, 'left');
  const wv = windAt(s.pos, game.t, s.agl), ws = Math.hypot(wv[0], wv[2]) * 1.94384, wd = (Math.atan2(-wv[0], wv[2]) / DEG + 360) % 360;
  cpText(g, `${String(Math.round(wd)).padStart(3, '0')}°/${Math.round(ws)}`, x + 8, y + 30, 11, '#5fd4ff', 700, 'left');
  if (dtxt) cpText(g, dtxt, x + w - 8, y + 14, 11, '#e36bdc', 700, 'right');
  cpText(g, gm ? 'MAP' : ab ? 'ARC' : 'MAP', x + w - 8, y + h - 12, 10, '#58e08c', 700, 'right');
  g.restore();
}

  ];

    ['燃料', '残量あり・セレクター BOTH', () => SYS.fuelL + SYS.fuelR > SYS.cap * 0.2 && (!piston || SYS.fuelSel === 3)],

    ...(piston ? [['ミクスチャー', 'RICH', () => SYS.mix > 0.9 || game.autoMix]] : []),
    ['燃料セレクター', 'BOTH', () => !piston || SYS.fuelSel === 3],
    ...(ret ? [['ギア', 'DOWN（緑 3 つ）', () => s.gearPos >= 1]] : []),
    ...(hasFlaps ? [['フラップ', '進入位置', () => s.flapIdx >= (V.apprFlaps || 1)]] : []),
    ['着陸灯', 'ON', () => SYS.land],
    ['オートパイロット', 'OFF', () => !game.ap],
  ];

    ['パーキングブレーキ', 'SET', () => SYS.pbrake],
    ['スロットル', 'アイドル', () => s.thr < 0.05],
    ...(cpAutoAvio() ? [] : [['アビオニクス', 'OFF', () => !SYS.avio]]),
    ...(piston ? [['ミクスチャー', 'カットオフ（いちばん下）→ 停止', () => SYS.mix < 0.1]] : [['燃料レバー', 'CUTOFF → 停止', () => !SYS.fuelOn]]),
    ...(piston ? [['キー（マグネトー）', 'OFF', () => SYS.mags === 0]] : []),
    ['灯火（ビーコン以外）', 'OFF', () => !SYS.land && !SYS.strobe && !SYS.taxi],
    ['ビーコン', 'OFF（エンジンが止まってから）', () => !SYS.beacon && !SYS.run],
    ['ALT/GEN・BAT', 'OFF', () => !SYS.alt && !SYS.bat],
  ];
  return [['エンジン始動', start], ['離陸前', before], ['着陸前', landing], ['停止', shutdown]];
}

function clToggle(v) {
  clBuild(); CL.open = v === undefined ? !CL.open : v; CL.el.hidden = !CL.open; CPIT.dirty = true;
  if (CL.open) { if (!SYS.run) CL.tab = 0; clRender(true); }

function clRender(force) {
  if (!CL.open || !CL.el) return;
  if (!force && R.time - CL.t < 0.4) return; CL.t = R.time;
  const lists = clLists(), [name, items] = lists[CL.tab] || lists[0];
  let next = items.findIndex(it => !it[2]());
  const html = [`<h3>チェックリスト（${ACFT.name}）<button data-act="close" aria-label="閉じる">×</button></h3><div class="tabs">`,
    ...lists.map(([n], i) => `<button data-tab="${i}" class="${i === CL.tab ? 'on' : ''}">${n}</button>`), '</div><ol>',
    ...items.map(([a, b, f], i) => `<li class="${f() ? 'ok' : i === next ? 'next' : ''}"><i>${f() ? '✓' : i === next ? '▶' : '・'}</i><div><b>${a}</b><span>${b}</span></div></li>`), '</ol>',
    CL.tab === 0 && !SYS.run ? `<button class="auto" data-act="guide">ガイド付きで始動を習う（1 手順ずつ案内）</button>` : '',
    CL.tab === 0 ? `<button class="auto" data-act="auto">自動で始動する（Ctrl+E）</button>` : '',
    cam.mode !== 'cockpit' ? `<button class="auto" data-act="cockpit">コックピット視点にする（C）</button>` : '',
    `<p>${cpWhere()}上から順に、▶ の項目を操作すると ✓ が付きます。</p>`].join('');
  if (html !== CL.html) { CL.el.innerHTML = html; CL.html = html; }
}

function cpReal(A) {
  const D = cpRealDef(A);
  if (!D || !CPIT.real || (typeof EG !== 'undefined' && EG.on) || typeof glbGet !== 'function') return null;
  const g = cpRealGet(D);
  if (g.state !== 'ok') return null;
  let x = null;
  if (D.extra) {
    x = glbGet(D.extra.url, { xf: cpExtraXf(D) }); if (x.state !== 'ok') return null;
    // its untextured parts (backing and trim, white in the file) painted a dark panel grey
    if (!x.model.tinted) { x.model.tinted = true; for (const d of x.model.draws) if (d.mat.name === 'Material.012') d.mat.cut = 0.5; else if (!d.mat.tex && !d.mat.blend) { d.mat.em = [0, 0, 0]; d.mat.color = [0.07, 0.07, 0.075, 1]; d.mat.metal = 0; d.mat.rough = 0.8; } }
  }
  if ((D.tint || D.glass) && !g.model.tinted) {
    g.model.tinted = true;
    if (D.tint) for (const d of g.model.draws) if (!d.mat.tex) { const c = D.tint[d.mat.name] || D.tint['*']; if (c) d.mat.color = [...c, 1]; d.mat.rough = d.mat.name === 'Material.001' ? 0.1 : 0.75; d.mat.metal = 0; }
    // window glass that the file has as an opaque colour: made clear, with a faint tint
    if (D.glass) for (const d of g.model.draws) if (D.glass.includes(d.mat.name)) { d.mat.color = [0.55, 0.62, 0.66, 0.1]; d.mat.blend = true; d.mat.rough = 0.05; d.mat.metal = 0; }
    g.model.draws.sort((a, b) => a.mat.blend - b.mat.blend);
  }
  return { D, m: g.model, x: x && x.model };
}
// model -> the cockpit's frame with the eye at the origin
function cpRealXf(D) {
  const s = D.s || 1, c = Math.cos(D.rot || 0), n = Math.sin(D.rot || 0), e = D.eye;
  const R = [c * s, 0, -n * s, 0, 0, s, 0, 0, n * s, 0, c * s, 0, 0, 0, 0, 1];
  return M4.mul(R, M4.fromQP([0, 0, 0, 1], [-e[0], -e[1], -e[2]]));
}
// the added model (extra): its own coordinates -> the main model's -> the cockpit's
function cpExtraM(E) {
  const c = Math.cos(E.rot || 0), n = Math.sin(E.rot || 0), s = E.s || 1;
  return M4.mul(M4.fromQP([0, 0, 0, 1], E.to), M4.mul([c * s, 0, -n * s, 0, 0, s, 0, 0, n * s, 0, c * s, 0, 0, 0, 0, 1], M4.fromQP([0, 0, 0, 1], E.at.map(v => -v))));
}

function cpRealGet(D) {
  const k = D.keep, s = D.s || 1, c = Math.cos(D.rot || 0), n = Math.sin(D.rot || 0), e = D.eye;
  // drop: boxes in the model's own coordinates ([x0, y0, z0, x1, y1, z1]) whose parts are left out (an opaque
  // HUD glass in front of the eye)
  // yokes: boxes (the same way) around the control columns, which Y hides
  const inBox = (L, x, y, z) => {
    if (!L) return false;
    const mx = e[0] + (c * x + n * z) / s, my = e[1] + y / s, mz = e[2] + (-n * x + c * z) / s;
    return L.some(b => mx > b[0] && my > b[1] && mz > b[2] && mx < b[3] && my < b[4] && mz < b[5]);
  };
  // each live control is cut out as its own moving part ('h:<index into D.hot>'): by the model's part name,
  // or the triangles within a little of its centre
  const hs = (D.hot || []).map((H, i) => ({ i, n: H.n, c: H.p ? cpRealPt(D, H.p) : null, r2: ((H.r || 0.011) * (D.s || 1) * (H.a === 'fuelSel' ? 0.45 : 0.85)) ** 2, a: H.a })).filter(h => !(D.still || []).includes(h.a));
  const hot = (x, y, z, nm) => {
    for (const h of hs) if (h.n ? nm === h.n : (x - h.c[0]) ** 2 + (y - h.c[1]) ** 2 + (z - h.c[2]) ** 2 < h.r2) return 'h:' + h.i;
    return 0;
  };
  return glbGet(D.url, { key: D.key, split: D.split, xf: cpRealXf(D), pivot: () => [0, 0, 0],
    part: (x, y, z, mat, nm) => ((k && (Math.abs(x) > k || Math.abs(y) > k || z > k || z < -k * 1.4)) || (inBox(D.drop, x, y, z) && !(D.split && D.split.includes(mat))) ? -1 : inBox(D.yokes, x, y, z) ? 1 : D.split && D.split.includes(mat) ? 0 : hot(x, y, z, nm)) });
}

function cpLayout(A) {
  const real = !!cpReal(A);
  if (A.cpL && A.cpL.real === real && A.cpL.av === (typeof acCustom === 'function' ? acCustom(A.id).avionics : '')) return A.cpL;
  const L = Object.assign({}, CP_LAYOUT[A.id] || CP_LAYOUT.trainer), e = A.eye;
  L.av = typeof acCustom === 'function' ? acCustom(A.id).avionics : ''; L.real = real;
  if (L.deck === 'light' && (L.av === 'g1000' || (real && cpRealDef(A) === CP_REAL[A.id])) && ['c172', 'pa28'].includes(A.id)) L.deck = 'glassga';
  L.T = CP_THEME[L.theme] || CP_THEME.cessna;
  L.E = [L.seat, e[1], e[2]];
  L.floor = e[1] - 0.95;
  L.pw = 2 * (L.W - 0.03); L.ph = L.pw * CP_REG.main[1] / CP_REG.main[0];
  const tb = 10 * DEG, c = Math.cos(tb), s = Math.sin(tb);
  L.pz = e[2] - L.dz; L.py = e[1] - L.dy;
  L.top = L.py + L.ph / 2 * c; L.bot = L.py - L.ph / 2 * c; L.zTop = L.pz - L.ph / 2 * s; L.zBot = L.pz + L.ph / 2 * s;
  L.gd = L.deck === 'airliner' ? 0.17 : L.deck === 'turboprop' ? 0.13 : 0.045;
  L.gh = L.glr ? 0.1 : 0.035;
  L.zg = L.zTop + L.gd;                                             // rear face of the glareshield
  const S = L.S = {};
  S.main = { o: [-L.pw / 2, L.top, L.zTop], u: [L.pw, 0, 0], v: [0, L.bot - L.top, L.zBot - L.zTop], shade: 1 };
  if (L.glr) S.glr = { o: [-L.pw / 2 + 0.03, L.top + L.gh - 0.008, L.zg + 0.003], u: [L.pw - 0.06, 0, 0], v: [0, -(L.gh - 0.016), 0], shade: 0 };
  if (L.ped === 'stand') {
    const w = 0.4, len = w * CP_REG.ped[1] / CP_REG.ped[0], dy = -0.08;
    S.ped = { o: [-w / 2, e[1] - 0.64, L.zBot + 0.03], u: [w, 0, 0], v: [0, dy, Math.sqrt(len * len - dy * dy)], shade: 0 };
  } else if (L.ped === 'console') {
    const w = 0.22, len = w * CP_REG.ped[1] / CP_REG.ped[0], dy = -0.3;
    S.ped = { o: [-w / 2, L.bot - 0.02, L.zBot + 0.03], u: [w, 0, 0], v: [0, dy, Math.sqrt(len * len - dy * dy)], shade: 0 };
  }
  if (L.ovh) {
    const w = Math.min(L.pw * 0.58, 1.0), len = w * CP_REG.ovh[1] / CP_REG.ovh[0], rise = 0.13, dz = Math.sqrt(len * len - rise * rise);
    const yF = e[1] + L.roof - 0.1, zF = e[2] - 0.5;
    S.ovh = { o: [-w / 2, yF + rise, zF + dz], u: [w, 0, 0], v: [0, -rise, -dz], shade: 0 };
  }
  // the pilot's position on the main panel canvas (the primary instruments go in front of it)
  L.px = L.seat ? (L.seat + L.pw / 2) / L.pw * CP_REG.main[0] : (L.deck === 'light' ? 760 : 1024);
  // a 3D cabin with the sim's own panel in it: the panel surface where the model says, the pilot's place on it
  const RD = real && cpRealDef(A);
  L.realPanel = !!(RD && RD.panel);
  if (L.realPanel) {
    const P = RD.panel, w = P.w, h = w * CP_REG.main[1] / CP_REG.main[0], c = vadd(cpRealPt(RD, P.c), L.E);
    L.pw = w; L.ph = h;
    S.main = { o: [c[0] - w / 2, c[1] + h / 2, c[2]], u: [w, 0, 0], v: [0, -h, 0], shade: 0 };
    L.px = (L.E[0] - S.main.o[0]) / w * CP_REG.main[0];
  }
  A.cpL = L;
  return L;
}

function cpBuild(A) {
  const L = cpLayout(A), T = L.T, e = A.eye, b = new MeshBuilder();
  const dark = vscale(T.trim, 0.55), wall = T.int, frame = vscale(T.int, 0.7), black = [0.04, 0.04, 0.045];
  const zb = e[2] + 1.05, zf = L.zTop - 0.2, sill = e[1] + L.sill, floor = L.floor;
  // panel housing and the lower (knee) panel
  b.box([0, (L.top + L.bot) / 2, L.zTop - 0.165], [L.pw + 0.06, L.top - L.bot + 0.04, 0.3], dark);   // (its face sits just behind the tilted panel)
  b.box([0, (L.bot + floor + 0.25) / 2, L.zBot - 0.04], [L.pw + 0.04, L.bot - floor - 0.25, 0.1], vscale(T.trim, 0.7));
  // glareshield: a slab over the panel with a rounded, padded lip
  const gTop = L.top + L.gh;
  // glareshield: a padded cowl with a rounded lip, drawn as one smooth section swept across the panel
  {
    const zf = L.zTop - 0.32, R = Math.min(0.03, L.gh * 0.45), prof = [[L.top - 0.004, zf], [gTop, zf]];
    for (let i = 0; i <= 6; i++) { const a = Math.PI / 2 * i / 6; prof.push([gTop - R + Math.cos(a) * R, L.zg - R + Math.sin(a) * R]); }
    prof.push([gTop - R * 2, L.zg + 0.002], [L.top + 0.004, L.zg - 0.004], [L.top - 0.004, L.zg - 0.02]);
    const hw = L.pw / 2 + 0.04, ring = x => prof.map(([y, z]) => [x, y, z]);
    b.loft([ring(-hw), ring(-hw * 0.5), ring(0), ring(hw * 0.5), ring(hw)], T.glare, { capStart: true, capEnd: true, smooth: false });
  }

// baked ambient light: corners, the footwell and the space under the glareshield get less light, and a
// faint per-vertex variation breaks up the flat colours (a real interior is never evenly lit)
function cpBakeAO(b, L, e) {
  const P = b.pos, Cl = b.col, sm = (a, c, x) => { const t = clamp((x - a) / (c - a), 0, 1); return t * t * (3 - 2 * t); };
  for (let i = 0; i < P.length; i += 3) {
    const x = P[i], y = P[i + 1], z = P[i + 2];
    let k = 0.5 + 0.5 * sm(L.floor, L.floor + 0.75, y);                              // dark footwell
    if (z < L.zg + 0.05 && y < L.top + 0.01 && y > L.bot - 0.3) k *= 0.8;            // under the glareshield / in the panel
    k *= 1 - 0.28 * sm(L.W - 0.3, L.W, Math.abs(x)) * (y < e[1] + L.sill + 0.05 ? 1 : 0.4);   // against the walls
    k *= 1 - 0.2 * sm(e[1] + L.roof - 0.25, e[1] + L.roof + 0.1, y);                 // the ceiling
    const n = Math.sin(x * 91.7 + y * 47.3 + z * 13.1) * 0.5 + 0.5;
    k *= 0.96 + 0.08 * n;
    Cl[i] *= k; Cl[i + 1] *= k; Cl[i + 2] *= k;
  }
}

// the pedestal levers: [surface u, value 0..1, mesh] (placed along their slots)
function cpLevers(L) {
  const s = sim, out = [];
  if (!L.S.ped) return out;
  const tp = L.deck === 'turboprop', y0 = PED_SLOT[0], y1 = PED_SLOT[1];
  const at = (x, val) => [x / CP_REG.ped[0], (y1 - (y1 - y0) * val) / CP_REG.ped[1]];
  if (L.ped === 'stand') {
    for (const x of tp ? [150, 205] : [175, 245]) out.push([...at(x, s.thr), 'lever']);
    if (tp) for (const x of [275, 330]) out.push([...at(x, SYS.fuelOn ? 1 : 0.08), 'lever']);
    if (ACFT.wing.flap) out.push([...at(tp ? 390 : 360, 1 - s.flapIdx / 3), 'flap']);
  } else if (L.deck === 'glassga' && ACFT.id === 'sr22') out.push([...at(110, s.thr), 'lever']);
  return out;
}

const CP_RATE = { main: 1 / 20, glr: 1 / 6, ped: 1 / 8, ovh: 1 / 4 };
function cpRegion(k, L) {
  const sc = game.gfx === 'high' ? 1 : game.gfx === 'low' ? 0.6 : 0.8;
  let R = CPIT.reg[k];
  if (!R) R = CPIT.reg[k] = { cv: document.createElement('canvas'), tex: gl.createTexture(), hs: [], t: -1, sc: 0, id: '' };
  const [w, h] = CP_REG[k];
  if (R.sc !== sc) { R.sc = sc; R.cv.width = Math.round(w * sc); R.cv.height = Math.round(h * sc); R.g = R.cv.getContext('2d'); R.t = -1; }
  const id = ACFT.id + L.deck + L.av;
  if (R.id !== id) { R.id = id; R.t = -1; R.bg = null; }
  const now = performance.now() / 1000;
  if (R.t >= 0 && !R.force && now - R.t < CP_RATE[k]) return R;
  R.t = now; R.force = false;
  const g = R.g; g.setTransform(sc, 0, 0, sc, 0, 0);
  HS = R.hs; HS.length = 0;
  cpDrawRegion(k, g, L, R);
  gl.bindTexture(gl.TEXTURE_2D, R.tex);
  gl.pixelStorei(gl.UNPACK_FLIP_Y_WEBGL, false);
  gl.texImage2D(gl.TEXTURE_2D, 0, gl.RGBA, gl.RGBA, gl.UNSIGNED_BYTE, R.cv);
  gl.generateMipmap(gl.TEXTURE_2D);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MIN_FILTER, gl.LINEAR_MIPMAP_LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_MAG_FILTER, gl.LINEAR);
  gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_S, gl.CLAMP_TO_EDGE); gl.texParameteri(gl.TEXTURE_2D, gl.TEXTURE_WRAP_T, gl.CLAMP_TO_EDGE);
  return R;
}

function cpGlass(g, cx, cy, r) {
  g.save(); g.beginPath(); g.arc(cx, cy, r, 0, 7); g.clip();
  const sh = g.createRadialGradient(cx, cy, r * 0.72, cx, cy, r); sh.addColorStop(0, 'rgba(0,0,0,0)'); sh.addColorStop(1, 'rgba(0,0,0,0.45)');
  g.fillStyle = sh; g.fillRect(cx - r, cy - r, 2 * r, 2 * r);
  const hl = g.createLinearGradient(cx - r, cy - r, cx + r * 0.2, cy + r * 0.2); hl.addColorStop(0, 'rgba(255,255,255,0.16)'); hl.addColorStop(0.45, 'rgba(255,255,255,0.03)'); hl.addColorStop(0.46, 'rgba(255,255,255,0)');
  g.fillStyle = hl; g.beginPath(); g.ellipse(cx - r * 0.25, cy - r * 0.35, r * 0.95, r * 0.6, -0.5, 0, 7); g.fill();
  g.restore();
  // a machined bezel ring with a highlight on the top edge
  g.lineWidth = 3; const bz = g.createLinearGradient(cx, cy - r, cx, cy + r); bz.addColorStop(0, '#6d7277'); bz.addColorStop(0.5, '#2a2d31'); bz.addColorStop(1, '#141517');
  g.strokeStyle = bz; g.beginPath(); g.arc(cx, cy, r + 1.5, 0, 7); g.stroke();
}
// the glass of a display: a faint diagonal reflection and a darker edge
function cpScreenGlass(g, x, y, w, h) {
  g.save(); rr(g, x, y, w, h, 3); g.clip();
  const r = g.createLinearGradient(x, y, x + w * 0.6, y + h); r.addColorStop(0, 'rgba(255,255,255,0.07)'); r.addColorStop(0.35, 'rgba(255,255,255,0.015)'); r.addColorStop(0.36, 'rgba(255,255,255,0)');
  g.fillStyle = r; g.fillRect(x, y, w, h);
  g.strokeStyle = 'rgba(0,0,0,0.55)'; g.lineWidth = 6; g.strokeRect(x, y, w, h);
  g.restore();
  // the bezel's bevel
  g.lineWidth = 2; g.strokeStyle = 'rgba(255,255,255,0.12)'; g.beginPath(); g.moveTo(x - 4, y + h + 4); g.lineTo(x - 4, y - 4); g.lineTo(x + w + 4, y - 4); g.stroke();
  g.strokeStyle = 'rgba(0,0,0,0.5)'; g.beginPath(); g.moveTo(x + w + 4, y - 4); g.lineTo(x + w + 4, y + h + 4); g.lineTo(x - 4, y + h + 4); g.stroke();
}
function cpDrawRegion(k, g, L, R) {
  const [w, h] = CP_REG[k], T = L.T;
  CPIT.sheen = []; CPIT.dials = [];
  g.drawImage(cpFace(R, k, w, h, T), 0, 0, w, h);
  if (k === 'main') ({ light: mainLight, glassga: mainGlassGA, turboprop: mainTurboprop, airliner: mainAirliner, jet: mainJet }[L.deck] || mainLight)(g, L, T);
  else if (k === 'glr') glrPanel(g, L, T);
  else if (k === 'ped') (L.ped === 'stand' ? pedStand : pedConsole)(g, L, T);
  else if (k === 'ovh') ovhPanel(g, L, T);
  for (const [x, y, sw, sh] of CPIT.sheen) cpScreenGlass(g, x, y, sw, sh);
  for (const [x, y, r] of CPIT.dials) cpGlass(g, x, y, r);
  CPIT.sheen = null; CPIT.dials = null;
}

function cpClock(g, cx, cy, r, T) {
  cpDial(g, cx, cy, r);
  const d = new Date(), sec = d.getSeconds() + d.getMilliseconds() / 1000, min = d.getMinutes() + sec / 60;
  g.fillStyle = '#e8e8e8'; for (let i = 0; i < 60; i += 5) { const a = i / 60 * Math.PI * 2; g.fillRect(cx + Math.sin(a) * (r - 10) - 1, cy - Math.cos(a) * (r - 10) - 1, 2, 2); }
  for (const [a, l, w, c] of [[min / 60, r - 16, 3, '#e8e8e8'], [sec / 60, r - 8, 1.5, '#f2b33a']]) { const an = a * Math.PI * 2; g.strokeStyle = c; g.lineWidth = w; g.beginPath(); g.moveTo(cx, cy); g.lineTo(cx + Math.sin(an) * l, cy - Math.cos(an) * l); g.stroke(); }
  cpText(g, 'CLOCK', cx, cy + r + 16, 9, T.ink);
}
